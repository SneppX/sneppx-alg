32static int valid_rule(const SNEPPXFirewallRule *rule) {
33 size_t bytes;
34 unsigned bits;
35 if (rule == NULL) return 0;
36 bytes = address_size(rule->family);
37 if (bytes == 0u) return 0;
38 bits = (unsigned)(bytes * 8u);
39 if (rule->source_prefix > bits || rule->destination_prefix > bits) return 0;
40 if (rule->direction > SNEPPX_FW_BOTH_DIRECTIONS) return 0;
41 if (rule->action != SNEPPX_FW_DENY && rule->action != SNEPPX_FW_ALLOW) return 0;
42 if (rule->match_source > 1u || rule->match_destination > 1u ||
43 rule->match_destination_port > 1u || rule->enabled > 1u) return 0;
44 if (rule->match_destination_port &&
45 rule->destination_port_first > rule->destination_port_last) return 0;
46 return 1;
47}

56static int prefix_matches(
57 const uint8_t *address, const uint8_t *network, unsigned prefix_bits) {
58 unsigned full_bytes = prefix_bits / 8u;
59 unsigned remaining_bits = prefix_bits % 8u;
60 if (full_bytes > 0u && memcmp(address, network, full_bytes) != 0) return 0;
61 if (remaining_bits > 0u) {
62 uint8_t mask = (uint8_t)(0xFFu << (8u - remaining_bits));
63 if ((address[full_bytes] & mask) != (network[full_bytes] & mask)) return 0;
64 }
65 return 1;
66}

68static int rule_matches(
69 const SNEPPXFirewallRule *rule, const SNEPPXFirewallPacket *packet) {
70 if (!rule->enabled || rule->family != packet->family) return 0;
71 if (rule->protocol != SNEPPX_FW_PROTOCOL_ANY &&
72 rule->protocol != packet->protocol) return 0;
73 if (rule->direction != SNEPPX_FW_BOTH_DIRECTIONS &&
74 rule->direction != packet->direction) return 0;
75 if (rule->match_source &&
76 !prefix_matches(packet->source, rule->source, rule->source_prefix)) return 0;
77 if (rule->match_destination &&
78 !prefix_matches(packet->destination, rule->destination,
79 rule->destination_prefix)) return 0;
80 if (rule->match_destination_port &&
81 (packet->destination_port < rule->destination_port_first ||
82 packet->destination_port > rule->destination_port_last)) return 0;
83 return 1;
84}

119int SNEPPX_firewall_add_rule(
120 SNEPPXFirewallEngine *engine,
121 const SNEPPXFirewallRule *rule,
122 uint32_t *assigned_rule_id) {
123 SNEPPXFirewallInternal *state;
124 SNEPPXFirewallRule copy;
125 size_t i;
126 if (engine == NULL || engine->internal == NULL || !valid_rule(rule)) {
127 return SNEPPX_FW_ERROR_INVALID;
128 }
129 state = (SNEPPXFirewallInternal *)engine->internal;
130 firewall_lock(state);
131 if (state->rule_count >= SNEPPX_FW_MAX_RULES) {
132 firewall_unlock(state);
133 return SNEPPX_FW_ERROR_CAPACITY;
134 }
135 copy = *rule;
136 if (copy.id == 0u) {
137 uint32_t candidate = state->next_rule_id;
138 int found;
139 do {
140 found = 0;
141 if (candidate == 0u) candidate = 1u;
142 for (i = 0u; i < state->rule_count; ++i) {
143 if (state->rules[i].id == candidate) {
144 found = 1;
145 candidate++;
146 break;
147 }
148 }
149 } while (found);
150 copy.id = candidate;
151 state->next_rule_id = candidate + 1u;
152 } else {
153 for (i = 0u; i < state->rule_count; ++i) {
154 if (state->rules[i].id == copy.id) {
155 firewall_unlock(state);
156 return SNEPPX_FW_ERROR_INVALID;
157 }
158 }
159 }
160 state->rules[state->rule_count++] = copy;
161 if (assigned_rule_id != NULL) *assigned_rule_id = copy.id;
162 firewall_unlock(state);
163 return SNEPPX_FW_OK;
164}

170 if (engine == NULL || engine->internal == NULL || rule_id == 0u) {

172 }
173 state = (SNEPPXFirewallInternal *)engine->internal;
174 firewall_lock(state);
175 for (i = 0u; i < state->rule_count; ++i) {

190int SNEPPX_firewall_evaluate(
191 SNEPPXFirewallEngine *engine,
192 const SNEPPXFirewallPacket *packet,
193 SNEPPXFirewallResult *result) {
194 SNEPPXFirewallInternal *state;
195 const SNEPPXFirewallRule *best = NULL;
196 size_t i;
197 uint8_t action;
198 if (engine == NULL || engine->internal == NULL ||
199 !valid_packet(packet) || result == NULL) {
200 return SNEPPX_FW_ERROR_INVALID;
201 }
202 state = (SNEPPXFirewallInternal *)engine->internal;
203 firewall_lock(state);
204 for (i = 0u; i < state->rule_count; ++i) {
205 const SNEPPXFirewallRule *candidate = &state->rules[i];
206 state->stats.rules_evaluated++;
207 if (!rule_matches(candidate, packet)) continue;
208 if (best == NULL || candidate->priority > best->priority ||
209 (candidate->priority == best->priority &&
210 candidate->action == SNEPPX_FW_DENY &&
211 best->action == SNEPPX_FW_ALLOW)) {
212 best = candidate;
213 }
214 }
215 action = best != NULL ? best->action : state->default_action;
216 result->action = action;
217 result->matched_rule = best != NULL ? 1u : 0u;
218 result->reserved = 0u;
219 result->rule_id = best != NULL ? best->id : 0u;
220 if (action == SNEPPX_FW_ALLOW) {
221 state->stats.allowed_packets++;
222 state->stats.allowed_bytes += packet->payload_bytes;
223 } else {
224 state->stats.blocked_packets++;
225 state->stats.blocked_bytes += packet->payload_bytes;
226 }
227 firewall_unlock(state);
228 return SNEPPX_FW_OK;
229}
