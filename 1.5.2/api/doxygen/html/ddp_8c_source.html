43int sneppx_ddp_init(SNEPPX_DDPState** ddp, int world_size, int rank,
44 int bucket_size_mb, int overlap_comm) {
45 if (!ddp) return -1;
46 SNEPPX_DDPState* d = (SNEPPX_DDPState*)calloc(1, sizeof(SNEPPX_DDPState));
47 if (!d) return -1;
48 d->world_size = world_size;
49 d->rank = rank;
50 d->bucket_size = bucket_size_mb * 1024 * 1024;
51 d->overlap_comm = overlap_comm;
52 sneppx_nccl_initialize();
53 sneppx_pg_create(&d->pg, d->world_size, d->rank);
54 if (overlap_comm) {
55 cudaStreamCreate(&d->comm_stream);
56 d->num_buckets = 4;
57 d->sync_events = (cudaEvent_t*)calloc(d->num_buckets, sizeof(cudaEvent_t));
58 for (int i = 0; i < d->num_buckets; i++) cudaEventCreate(&d->sync_events[i]);
59 }
60 *ddp = d;
61 return 0;
62}

64void sneppx_ddp_destroy(SNEPPX_DDPState* ddp) {
65 if (!ddp) return;
66 if (ddp->pg) sneppx_pg_destroy(ddp->pg);
67 if (ddp->grad_buffer) cudaFree(ddp->grad_buffer);
68 if (ddp->overlap_comm && ddp->sync_events) {
69 for (int i = 0; i < ddp->num_buckets; i++) cudaEventDestroy(ddp->sync_events[i]);
70 free(ddp->sync_events);
71 cudaStreamDestroy(ddp->comm_stream);
72 }
73 free(ddp);
74}

76// Bucket-based gradient all-reduce with optional compute overlap
77int sneppx_ddp_all_reduce_grads(SNEPPX_DDPState* ddp, float** grads,
78 size_t* sizes, int num_grads,
79 cudaStream_t compute_stream) {
80 if (!ddp || !grads || !sizes) return -1;
81 size_t total_bytes = 0;
82 for (int i = 0; i < num_grads; i++) total_bytes += sizes[i];
83 if (!ddp->grad_buffer) cudaMalloc(&ddp->grad_buffer, total_bytes);
84 size_t offset = 0;
85 for (int i = 0; i < num_grads; i++) {
86 cudaMemcpyAsync((char*)ddp->grad_buffer + offset, grads[i],
87 sizes[i], cudaMemcpyDeviceToDevice, compute_stream);
88 offset += sizes[i];
89 }
90 cudaStreamSynchronize(compute_stream);
91 // Real all-reduce across data-parallel ranks via NCCL (CPU fallback if NCCL absent)
92 if (ddp->pg) {
93 int ret = sneppx_pg_all_reduce(ddp->pg, ddp->grad_buffer,
94 total_bytes / sizeof(float),
95 SNEPPX_NCCL_FLOAT, SNEPPX_NCCL_SUM,
96 compute_stream);
97 if (ret != 0) return ret;
98 }
99 offset = 0;
100 for (int i = 0; i < num_grads; i++) {
101 cudaMemcpy(grads[i], (float*)ddp->grad_buffer + offset,
102 sizes[i], cudaMemcpyDeviceToDevice);
103 offset += sizes[i];
104 }
105 return 0;
106}

108// Bucket-level all-reduce with async compute overlap
109int sneppx_ddp_bucket_all_reduce(SNEPPX_DDPState* ddp, int bucket_id,
110 float* data, size_t size,
111 cudaStream_t compute_stream) {
112 if (!ddp || !data) return -1;
113 if (ddp->overlap_comm && bucket_id > 0) {
114 cudaEventSynchronize(ddp->sync_events[bucket_id - 1]);
115 }
116 cudaMemcpyAsync(data, data, size, cudaMemcpyDeviceToDevice, ddp->comm_stream);
117 if (ddp->pg) {
118 int ret = sneppx_pg_all_reduce(ddp->pg, data, size / sizeof(float),
119 SNEPPX_NCCL_FLOAT, SNEPPX_NCCL_SUM,
120 ddp->comm_stream);
121 if (ret != 0) return ret;
122 }
123 if (ddp->overlap_comm) {
124 cudaEventRecord(ddp->sync_events[bucket_id], ddp->comm_stream);
125 }
126 return 0;
127}
