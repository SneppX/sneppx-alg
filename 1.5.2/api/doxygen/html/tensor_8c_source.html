34static int compute_offset_checked(
35 const SNEPPXTensor* tensor,
36 const size_t* indices,
37 size_t* out_offset
38) {
39 if (!tensor || !out_offset || !tensor->data ||
40 (tensor->ndim > 0 && (!indices || !tensor->shape || !tensor->strides))) {
41 return 0;
42 }
43
44 size_t offset = 0;
45 for (size_t i = 0; i < tensor->ndim; ++i) {
46 if (indices[i] >= tensor->shape[i]) return 0;
47 if (tensor->strides[i] != 0 &&
48 indices[i] > (SIZE_MAX - offset) / tensor->strides[i]) return 0;
49 offset += indices[i] * tensor->strides[i];
50 }
51
52 if (tensor->item_size == 0) return 0;
53 if (tensor->storage) {
54 if (tensor->storage->num_bytes % tensor->item_size != 0) return 0;
55 const size_t capacity = tensor->storage->num_bytes / tensor->item_size;
56 if (tensor->offset > capacity || offset >= capacity - tensor->offset) return 0;
57 }
58 *out_offset = offset;
59 return 1;

62static int compute_offset_flat_checked(
63 const SNEPPXTensor* tensor,
64 size_t flat_index,
65 size_t* out_offset
66) {
67 if (!tensor || !out_offset || !tensor->data || flat_index >= tensor->size ||
68 (tensor->ndim > 0 && (!tensor->shape || !tensor->strides))) return 0;
69
70 size_t remaining = flat_index;
71 size_t offset = 0;
72 for (size_t i = tensor->ndim; i > 0; --i) {
73 const size_t dim = tensor->shape[i - 1];
74 if (dim == 0) return 0;
75 const size_t coordinate = remaining % dim;
76 remaining /= dim;

78 if (stride != 0 && coordinate > (SIZE_MAX - offset) / stride) return 0;
79 offset += coordinate * stride;
80 }

85 const size_t capacity = tensor->storage->num_bytes / tensor->item_size;
86 if (tensor->offset > capacity || offset >= capacity - tensor->offset) return 0;
87 }

166SNEPPXTensor* SNEPPX_tensor_as_strided(const SNEPPXTensor* src, size_t offset, const size_t* shape, size_t ndim, const size_t* strides) {
167 if (!src || !src->storage || src->item_size == 0 ||
168 (ndim > 0 && (!shape || !strides))) return NULL;
169 if (src->storage->num_bytes % src->item_size != 0) return NULL;
170
171 size_t view_size = 1;
172 size_t max_offset = offset;
173 int is_empty = 0;
174 for (size_t i = 0; i < ndim; ++i) {
175 if (shape[i] == 0) {
176 is_empty = 1;
177 view_size = 0;
178 } else {
179 if (view_size > SIZE_MAX / shape[i]) return NULL;
180 view_size *= shape[i];
181 const size_t last_index = shape[i] - 1;
182 if (strides[i] != 0 && last_index > SIZE_MAX / strides[i]) return NULL;
183 const size_t delta = last_index * strides[i];
184 if (max_offset > SIZE_MAX - delta) return NULL;
185 max_offset += delta;
186 }
187 }
188
189 const size_t capacity = src->storage->num_bytes / src->item_size;
190 if (offset > capacity) return NULL;
191 if (!is_empty && max_offset >= capacity) return NULL;
192
193 SNEPPXTensor* t = tensor_view_alloc(src, ndim);
194 if (!t) return NULL;
195 for (size_t i = 0; i < ndim; ++i) {
196 t->shape[i] = shape[i];
197 t->strides[i] = strides[i];
198 }
199
200 SNEPPX_storage_retain(src->storage);
201 t->storage = src->storage;
202 t->offset = offset;
203 t->ndim = ndim;
204 t->size = view_size;
205 t->dtype = src->dtype;
206 t->item_size = src->item_size;
207 t->device = src->device;
208 t->device_id = src->device_id;
209 t->layout = src->layout;
210 t->owns_data = 0;
211 t->backend_handle = NULL;
212 t->data = src->storage->data
213 ? (unsigned char*)src->storage->data + offset * t->item_size
214 : NULL;
215 return t;
216}

278SNEPPXTensor* SNEPPX_tensor_create(const size_t* shape, size_t ndim, SNEPPXDtype dtype) {
279 if (ndim > 0 && !shape) return NULL;
280
281 const size_t item_size = SNEPPX_tensor_dtype_size(dtype);
282 if (item_size == 0) return NULL;
283
284 size_t total = 1;
285 for (size_t i = 0; i < ndim; ++i) {
286 if (shape[i] != 0 && total > SIZE_MAX / shape[i]) return NULL;
287 total *= shape[i];
288 }
289
290 /* Validate row-major stride products independently, including zero-sized shapes. */
291 size_t stride = 1;
292 for (size_t i = ndim; i > 0; --i) {
293 const size_t dim = shape[i - 1];
294 if (dim != 0 && stride > SIZE_MAX / dim) return NULL;
295 stride *= dim;
296 }
297
298 if (total != 0 && item_size > SIZE_MAX / total) return NULL;
299 const size_t num_bytes = total * item_size;
300 const size_t safe_ndim = ndim > 0 ? ndim : 1;
301 if (safe_ndim > SIZE_MAX / sizeof(size_t)) return NULL;
302 const size_t metadata_bytes = safe_ndim * sizeof(size_t);
303
304 SNEPPXTensor* tensor = (SNEPPXTensor*)aligned_alloc_wrapper(sizeof(SNEPPXTensor), 64);
305 if (!tensor) return NULL;
306 memset(tensor, 0, sizeof(*tensor));
307
308 tensor->shape = (size_t*)aligned_alloc_wrapper(metadata_bytes, 64);
309 tensor->strides = (size_t*)aligned_alloc_wrapper(metadata_bytes, 64);
310 if (!tensor->shape || !tensor->strides) {
311 SNEPPX_free(tensor->shape, metadata_bytes);
312 SNEPPX_free(tensor->strides, metadata_bytes);
313 SNEPPX_free(tensor, sizeof(SNEPPXTensor));
314 return NULL;
315 }
316
317 tensor->ndim = ndim;
318 tensor->size = total;
319 tensor->dtype = dtype;
320 tensor->item_size = item_size;
321 tensor->device = SNEPPX_DEVICE_CPU;
322 tensor->device_id = 0;
323 tensor->layout = SNEPPX_LAYOUT_ROW_MAJOR;
324 tensor->owns_data = 1;
325
326 stride = 1;
327 for (size_t i = ndim; i > 0; --i) {
328 tensor->shape[i - 1] = shape[i - 1];
329 tensor->strides[i - 1] = stride;
330 stride *= shape[i - 1];
331 }
332
333 tensor->storage = SNEPPX_storage_create(num_bytes);
334 if (!tensor->storage) {
335 SNEPPX_free(tensor->shape, metadata_bytes);
336 SNEPPX_free(tensor->strides, metadata_bytes);
337 SNEPPX_free(tensor, sizeof(SNEPPXTensor));
338 return NULL;
339 }
340 tensor->data = tensor->storage->data;
341 return tensor;
342}

754 float* data = (float*)tensor->data;
755 for (size_t i = 0; i < tensor->size; i += 2) {
756 float u1 = uniform_01();
757 float u2 = uniform_01();
758 float r = sqrtf(-2.0f * logf(u1 + 1e-10f));
759 float theta = 2.0f * 3.14159265f * u2;
760 data[i] = r * cosf(theta);
761 if (i + 1 < tensor->size) {
762 data[i + 1] = r * sinf(theta);
763 }
764 }
765 return tensor;
766}

833 }
834 return SNEPPX_tensor_as_strided(src, src->offset, resolved_shape, new_ndim, new_strides);
835 }
