124int SNEPPX_fv_add_transition(uint32_t from_state, uint32_t to_state) {
125 if (!g_fv_initialized) SNEPPX_fv_model_init();
126 int f = -1, t = -1;
127 for (int i = 0; i < g_fv_model.state_count; i++) {
128 if (g_fv_model.states[i].state_id == from_state) f = i;
129 if (g_fv_model.states[i].state_id == to_state) t = i;
130 }
131 if (f < 0) return -1;
132 if (t < 0) {
133 if (SNEPPX_fv_add_state(to_state, 0, 0) != 0) return -1;
134 for (int i = 0; i < g_fv_model.state_count; i++) {
135 if (g_fv_model.states[i].state_id == to_state) t = i;
136 if (g_fv_model.states[i].state_id == from_state) f = i;
137 }
138 }
139 if (f < 0 || t < 0) return -1;
140 if (g_fv_model.states[f].next_count >= 8) return -1;
141 for (int j = 0; j < (int)g_fv_model.states[f].next_count; j++) {
142 if (g_fv_model.states[f].next_states[j] == to_state) return 0;
143 }
144 g_fv_model.states[f].next_states[g_fv_model.states[f].next_count++] = to_state;
145 return 0;
