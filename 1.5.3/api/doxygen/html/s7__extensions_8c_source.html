25#define SNEPPX_PATCH_HEADER_SIZE 12U
26static void patch_write_u64(uint8_t* out, uint64_t value) {
27 for (int i = 0; i < 8; i++) out[i] = (uint8_t)(value >> (i * 8));
28}
29static uint64_t patch_read_u64(const uint8_t* in) {
30 uint64_t value = 0;
31 for (int i = 0; i < 8; i++) value |= (uint64_t)in[i] << (i * 8);
32 return value;
33}
34static int patch_parse(const uint8_t* old_data, size_t old_len,
35 const uint8_t* patch, size_t patch_len,
36 uint8_t* new_data, size_t new_capacity,
37 size_t* result_len, int emit) {
38 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
39 if (!patch || !result_len || (old_len && !old_data) ||
40 patch_len < SNEPPX_PATCH_HEADER_SIZE + 1 ||
41 memcmp(patch, magic, sizeof(magic)) != 0) return -1;
42 uint64_t expected64 = patch_read_u64(patch + 4);
43 if (expected64 > (uint64_t)SIZE_MAX) return -1;
44 size_t expected = (size_t)expected64;
45 if (expected > new_capacity) return -1;
46 size_t pos = SNEPPX_PATCH_HEADER_SIZE, produced = 0;
47 while (pos < patch_len) {
48 uint8_t opcode = patch[pos++];
49 if (opcode == 0) {
50 if (pos != patch_len || produced != expected) return -1;
51 *result_len = produced;
52 return 0;
53 }
54 if (opcode == 1) {
55 if (patch_len - pos < 16) return -1;
56 uint64_t offset64 = patch_read_u64(patch + pos);
57 uint64_t length64 = patch_read_u64(patch + pos + 8);
58 pos += 16;
59 if (offset64 > (uint64_t)old_len || length64 > (uint64_t)SIZE_MAX ||
60 length64 == 0 || length64 > (uint64_t)old_len - offset64) return -1;
61 size_t offset = (size_t)offset64, length = (size_t)length64;
62 if (produced > expected || length > expected - produced) return -1;
63 if (emit) memcpy(new_data + produced, old_data + offset, length);
64 produced += length;
65 continue;
66 }
67 if (opcode == 2) {
68 if (patch_len - pos < 8) return -1;
69 uint64_t length64 = patch_read_u64(patch + pos);
70 pos += 8;
71 if (length64 == 0 || length64 > (uint64_t)SIZE_MAX ||
72 length64 > (uint64_t)(patch_len - pos)) return -1;
73 size_t length = (size_t)length64;
74 if (produced > expected || length > expected - produced) return -1;
75 if (emit) memcpy(new_data + produced, patch + pos, length);
76 pos += length;
77 produced += length;
78 continue;
79 }
80 return -1;
81 }
82 return -1;
83}
84static int patch_read_file(const char* path, uint8_t** data_out, size_t* len_out) {
85 if (!path || !data_out || !len_out) return -1;
86 *data_out = NULL;
87 *len_out = 0;
88 FILE* file = fopen(path, "rb");
89 if (!file) return -1;
90 if (fseek(file, 0, SEEK_END) != 0) { fclose(file); return -1; }
91 long end = ftell(file);
92 if (end < 0 || fseek(file, 0, SEEK_SET) != 0) { fclose(file); return -1; }
93 size_t len = (size_t)end;
94 uint8_t* data = (uint8_t*)malloc(len ? len : 1);
95 if (!data) { fclose(file); return -1; }
96 if (len && fread(data, 1, len, file) != len) {
97 free(data); fclose(file); return -1;
98 }
99 if (fclose(file) != 0) { free(data); return -1; }
100 *data_out = data;
101 *len_out = len;
102 return 0;
103}
104static int patch_write_file(const char* path, const uint8_t* data, size_t len) {
105 if (!path || (len && !data)) return -1;
106 FILE* file = fopen(path, "wb");
107 if (!file) return -1;
108 int ok = !len || fwrite(data, 1, len, file) == len;
109 if (fclose(file) != 0) ok = 0;
110 if (!ok) { remove(path); return -1; }
111 return 0;
112}

166int SNEPPX_bsdiff(const uint8_t* old_data, size_t old_len,
167 const uint8_t* new_data, size_t new_len,
168 uint8_t* patch, size_t* patch_len) {
169 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
170 if (!patch || !patch_len || (old_len && !old_data) || (new_len && !new_data) ||
171 new_len > SIZE_MAX - 64) return -1;
172 size_t prefix = 0;
173 while (prefix < old_len && prefix < new_len && old_data[prefix] == new_data[prefix]) prefix++;
174 size_t suffix = 0;
175 while (suffix < old_len - prefix && suffix < new_len - prefix &&
176 old_data[old_len - 1 - suffix] == new_data[new_len - 1 - suffix]) suffix++;
177 size_t insert_len = new_len - prefix - suffix;
178 size_t required = SNEPPX_PATCH_HEADER_SIZE + 1;
179 if (prefix) required += 17;
180 if (insert_len) required += 9 + insert_len;
181 if (suffix) required += 17;
182 if (*patch_len < required) { *patch_len = required; return -1; }
183 size_t pos = 0;
184 memcpy(patch + pos, magic, sizeof(magic)); pos += sizeof(magic);
185 patch_write_u64(patch + pos, (uint64_t)new_len); pos += 8;
186 if (prefix) {
187 patch[pos++] = 1;
188 patch_write_u64(patch + pos, 0); pos += 8;
189 patch_write_u64(patch + pos, (uint64_t)prefix); pos += 8;
190 }
191 if (insert_len) {
192 patch[pos++] = 2;
193 patch_write_u64(patch + pos, (uint64_t)insert_len); pos += 8;
194 memcpy(patch + pos, new_data + prefix, insert_len); pos += insert_len;
195 }
196 if (suffix) {
197 patch[pos++] = 1;
198 patch_write_u64(patch + pos, (uint64_t)(old_len - suffix)); pos += 8;
199 patch_write_u64(patch + pos, (uint64_t)suffix); pos += 8;
200 }
201 patch[pos++] = 0;
202 *patch_len = pos;
203 return 0;
204}

851int SNEPPX_bsdiff_create_patch_file(const char* old_path, const char* new_path, const char* patch_path) {
852 if (!old_path || !new_path || !patch_path) return -1;
853 uint8_t *old_data = NULL, *new_data = NULL, *patch = NULL;
854 size_t old_len = 0, new_len = 0, patch_len = 0;
855 int rc = patch_read_file(old_path, &old_data, &old_len);
856 if (rc == 0) rc = patch_read_file(new_path, &new_data, &new_len);
857 if (rc == 0 && new_len > SIZE_MAX - 64) rc = -1;
858 if (rc == 0) {
859 patch_len = new_len + 64;
860 patch = (uint8_t*)malloc(patch_len ? patch_len : 1);
861 if (!patch) rc = -1;
862 }
863 if (rc == 0) rc = SNEPPX_bsdiff(old_data, old_len, new_data, new_len, patch, &patch_len);
864 if (rc == 0) rc = patch_write_file(patch_path, patch, patch_len);
865 free(old_data); free(new_data); free(patch);
866 return rc;

889int SNEPPX_bspatch_apply_patch(const char* old_path, const char* patch_path, const char* new_path) {
890 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
891 if (!old_path || !patch_path || !new_path) return -1;
892 uint8_t *old_data = NULL, *patch = NULL, *new_data = NULL;
893 size_t old_len = 0, patch_len = 0;
894 int rc = patch_read_file(old_path, &old_data, &old_len);
895 if (rc == 0) rc = patch_read_file(patch_path, &patch, &patch_len);
896 if (rc == 0 && (patch_len < SNEPPX_PATCH_HEADER_SIZE + 1 ||
897 memcmp(patch, magic, sizeof(magic)) != 0)) rc = -1;
898 size_t output_len = 0;
899 if (rc == 0) {
900 uint64_t output64 = patch_read_u64(patch + 4);
901 if (output64 > (uint64_t)SIZE_MAX || old_len > SIZE_MAX - patch_len ||
902 output64 > (uint64_t)(old_len + patch_len)) rc = -1;
903 else output_len = (size_t)output64;
904 }
905 if (rc == 0) {
906 new_data = (uint8_t*)malloc(output_len ? output_len : 1);
907 if (!new_data) rc = -1;
908 }
909 if (rc == 0) {
910 size_t capacity = output_len;
911 rc = SNEPPX_bspatch(old_data, old_len, patch, patch_len, new_data, &capacity);
912 if (rc == 0 && capacity != output_len) rc = -1;
913 }
914 if (rc == 0) rc = patch_write_file(new_path, new_data, output_len);
915 free(old_data); free(patch); free(new_data);
916 return rc;
