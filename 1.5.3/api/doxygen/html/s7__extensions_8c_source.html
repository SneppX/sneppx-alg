24static uint32_t hash_djb2(const uint8_t* data, size_t len) {
25 uint32_t h = 5381;
26 for (size_t i = 0; i < len; i++) h = ((h << 5) + h) ^ data[i];
27 return h;

35static uint64_t patch_read_u64(const uint8_t* in) {
36 uint64_t value = 0;
37 for (int i = 0; i < 8; i++) value |= (uint64_t)in[i] << (i * 8);

40static int patch_parse(const uint8_t* old_data, size_t old_len,
41 const uint8_t* patch, size_t patch_len,
42 uint8_t* new_data, size_t new_capacity,
43 size_t* result_len, int emit) {
44 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
45 if (!patch || !result_len || (old_len && !old_data) ||
46 patch_len < SNEPPX_PATCH_HEADER_SIZE + 1 ||
47 memcmp(patch, magic, sizeof(magic)) != 0) return -1;
48 uint64_t expected64 = patch_read_u64(patch + 4);
49 if (expected64 > (uint64_t)SIZE_MAX) return -1;
50 size_t expected = (size_t)expected64;
51 if (expected > new_capacity) return -1;
52 size_t pos = SNEPPX_PATCH_HEADER_SIZE, produced = 0;
53 while (pos < patch_len) {
54 uint8_t opcode = patch[pos++];
55 if (opcode == 0) {
56 if (pos != patch_len || produced != expected) return -1;
57 *result_len = produced;
58 return 0;
59 }
60 if (opcode == 1) {
61 if (patch_len - pos < 16) return -1;
62 uint64_t offset64 = patch_read_u64(patch + pos);
63 uint64_t length64 = patch_read_u64(patch + pos + 8);
64 pos += 16;
65 if (offset64 > (uint64_t)old_len || length64 > (uint64_t)SIZE_MAX ||
66 length64 == 0 || length64 > (uint64_t)old_len - offset64) return -1;
67 size_t offset = (size_t)offset64, length = (size_t)length64;
68 if (produced > expected || length > expected - produced) return -1;
69 if (emit) memcpy(new_data + produced, old_data + offset, length);
70 produced += length;
71 continue;
72 }
73 if (opcode == 2) {
74 if (patch_len - pos < 8) return -1;
75 uint64_t length64 = patch_read_u64(patch + pos);
76 pos += 8;
77 if (length64 == 0 || length64 > (uint64_t)SIZE_MAX ||
78 length64 > (uint64_t)(patch_len - pos)) return -1;
79 size_t length = (size_t)length64;
80 if (produced > expected || length > expected - produced) return -1;
81 if (emit) memcpy(new_data + produced, patch + pos, length);
82 pos += length;
83 produced += length;
84 continue;
85 }
86 return -1;
87 }
88 return -1;
89}
90static int patch_read_file(const char* path, uint8_t** data_out, size_t* len_out) {
91 if (!path || !data_out || !len_out) return -1;
92 *data_out = NULL;
93 *len_out = 0;
94 FILE* file = fopen(path, "rb");
95 if (!file) return -1;
96 if (fseek(file, 0, SEEK_END) != 0) { fclose(file); return -1; }
97 long end = ftell(file);
98 if (end < 0 || fseek(file, 0, SEEK_SET) != 0) { fclose(file); return -1; }
99 size_t len = (size_t)end;
100 uint8_t* data = (uint8_t*)malloc(len ? len : 1);
101 if (!data) { fclose(file); return -1; }
102 if (len && fread(data, 1, len, file) != len) {
103 free(data); fclose(file); return -1;
104 }
105 if (fclose(file) != 0) { free(data); return -1; }
106 *data_out = data;
107 *len_out = len;
108 return 0;
109}

172int SNEPPX_bsdiff(const uint8_t* old_data, size_t old_len,
173 const uint8_t* new_data, size_t new_len,
174 uint8_t* patch, size_t* patch_len) {
175 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
176 if (!patch || !patch_len || (old_len && !old_data) || (new_len && !new_data) ||
177 new_len > SIZE_MAX - 64) return -1;
178 size_t prefix = 0;
179 while (prefix < old_len && prefix < new_len && old_data[prefix] == new_data[prefix]) prefix++;
180 size_t suffix = 0;
181 while (suffix < old_len - prefix && suffix < new_len - prefix &&
182 old_data[old_len - 1 - suffix] == new_data[new_len - 1 - suffix]) suffix++;
183 size_t insert_len = new_len - prefix - suffix;
184 size_t required = SNEPPX_PATCH_HEADER_SIZE + 1;
185 if (prefix) required += 17;
186 if (insert_len) required += 9 + insert_len;
187 if (suffix) required += 17;
188 if (*patch_len < required) { *patch_len = required; return -1; }
189 size_t pos = 0;
190 memcpy(patch + pos, magic, sizeof(magic)); pos += sizeof(magic);
191 patch_write_u64(patch + pos, (uint64_t)new_len); pos += 8;
192 if (prefix) {
193 patch[pos++] = 1;
194 patch_write_u64(patch + pos, 0); pos += 8;
195 patch_write_u64(patch + pos, (uint64_t)prefix); pos += 8;
196 }
197 if (insert_len) {
198 patch[pos++] = 2;
199 patch_write_u64(patch + pos, (uint64_t)insert_len); pos += 8;
200 memcpy(patch + pos, new_data + prefix, insert_len); pos += insert_len;
201 }
202 if (suffix) {
203 patch[pos++] = 1;
204 patch_write_u64(patch + pos, (uint64_t)(old_len - suffix)); pos += 8;
205 patch_write_u64(patch + pos, (uint64_t)suffix); pos += 8;
206 }
207 patch[pos++] = 0;
208 *patch_len = pos;
209 return 0;
210}

562int SNEPPX_bsdiff_total_size(const uint8_t* patch, size_t patch_len) {
563 if (!patch || patch_len < 4) return 0;
564 size_t total = 0;
565 size_t pos = 0;
566 while (pos + 4 <= patch_len) {
567 uint32_t tag = ((uint32_t)patch[pos] << 24) | ((uint32_t)patch[pos + 1] << 16) | ((uint32_t)patch[pos + 2] << 8) | patch[pos + 3];
568 pos += 4;
569 if (tag > 0x00FFFFFF) {
570 if (pos + 2 > patch_len) break;
571 size_t len = ((size_t)patch[pos] << 8) | patch[pos + 1];
572 pos += 2;
573 if (pos + len > patch_len) break;
574 total += len;
575 pos += len;
576 } else {
577 if (pos + tag > patch_len) break;
578 total += tag;
579 pos += tag;
580 }
581 }
582 return (int)total;
583}

787int SNEPPX_manifest_hash(const char* manifest_path, uint8_t* hash_out, size_t hash_len) {
788 if (!manifest_path || !hash_out || hash_len < 4) return -1;
789 FILE* f = fopen(manifest_path, "rb");
790 if (!f) return -1;
791 fseek(f, 0, SEEK_END); long sz = ftell(f); fseek(f, 0, SEEK_SET);
792 uint8_t* buf = (uint8_t*)malloc((size_t)sz);
793 if (!buf) { fclose(f); return -1; }
794 fread(buf, 1, (size_t)sz, f); fclose(f);
795 uint32_t h = hash_djb2(buf, (size_t)sz); free(buf);
796 for (size_t i = 0; i < hash_len && i < 4; i++) hash_out[i] = (uint8_t)(h >> (i * 8));
797 return 0;
798}

875int SNEPPX_bsdiff_create_patch_file(const char* old_path, const char* new_path, const char* patch_path) {
876 if (!old_path || !new_path || !patch_path) return -1;
877 uint8_t *old_data = NULL, *new_data = NULL, *patch = NULL;
878 size_t old_len = 0, new_len = 0, patch_len = 0;
879 int rc = patch_read_file(old_path, &old_data, &old_len);
880 if (rc == 0) rc = patch_read_file(new_path, &new_data, &new_len);
881 if (rc == 0 && new_len > SIZE_MAX - 64) rc = -1;
882 if (rc == 0) {
883 patch_len = new_len + 64;
884 patch = (uint8_t*)malloc(patch_len ? patch_len : 1);
885 if (!patch) rc = -1;
886 }
887 if (rc == 0) rc = SNEPPX_bsdiff(old_data, old_len, new_data, new_len, patch, &patch_len);
888 if (rc == 0) rc = patch_write_file(patch_path, patch, patch_len);
889 free(old_data); free(new_data); free(patch);
890 return rc;
891}

913int SNEPPX_bspatch_apply_patch(const char* old_path, const char* patch_path, const char* new_path) {
914 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
915 if (!old_path || !patch_path || !new_path) return -1;
916 uint8_t *old_data = NULL, *patch = NULL, *new_data = NULL;
917 size_t old_len = 0, patch_len = 0;
918 int rc = patch_read_file(old_path, &old_data, &old_len);
919 if (rc == 0) rc = patch_read_file(patch_path, &patch, &patch_len);
920 if (rc == 0 && (patch_len < SNEPPX_PATCH_HEADER_SIZE + 1 ||
921 memcmp(patch, magic, sizeof(magic)) != 0)) rc = -1;
922 size_t output_len = 0;
923 if (rc == 0) {
924 uint64_t output64 = patch_read_u64(patch + 4);
925 if (output64 > (uint64_t)SIZE_MAX || old_len > SIZE_MAX - patch_len ||
926 output64 > (uint64_t)(old_len + patch_len)) rc = -1;
927 else output_len = (size_t)output64;
928 }
929 if (rc == 0) {
930 new_data = (uint8_t*)malloc(output_len ? output_len : 1);
931 if (!new_data) rc = -1;
932 }
933 if (rc == 0) {
934 size_t capacity = output_len;
935 rc = SNEPPX_bspatch(old_data, old_len, patch, patch_len, new_data, &capacity);
936 if (rc == 0 && capacity != output_len) rc = -1;
937 }
938 if (rc == 0) rc = patch_write_file(new_path, new_data, output_len);
939 free(old_data); free(patch); free(new_data);
940 return rc;
941}

986int SNEPPX_manifest_create(const char* path, const char** files, int file_count, uint8_t* manifest_out, size_t* manifest_len) {
987 if (!path || !files || !manifest_out || !manifest_len) return -1;
988 (void)path;
989 size_t pos = 0;
990 for (int i = 0; i < file_count; i++) {
991 if (!files[i]) continue;
992 size_t flen = strlen(files[i]);
993 if (pos + flen + 5 > *manifest_len) return -1;
994 manifest_out[pos++] = (uint8_t)(i + 1);
995 manifest_out[pos++] = (uint8_t)(flen & 0xFF);
996 manifest_out[pos++] = (uint8_t)((flen >> 8) & 0xFF);
997 memcpy(manifest_out + pos, files[i], flen);
998 pos += flen;
999 uint32_t h = hash_djb2((const uint8_t*)files[i], flen);
1000 for (int b = 0; b < 2; b++) manifest_out[pos++] = (uint8_t)(h >> (b * 8));
1001 }
1002 *manifest_len = pos;
1003 return 0;
1004}
