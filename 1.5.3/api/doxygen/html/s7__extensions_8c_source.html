25static void patch_write_u64(uint8_t* out, uint64_t value) {
26 for (int i = 0; i < 8; i++) out[i] = (uint8_t)(value >> (i * 8));
27}
28static uint64_t patch_read_u64(const uint8_t* in) {
29 uint64_t value = 0;
30 for (int i = 0; i < 8; i++) value |= (uint64_t)in[i] << (i * 8);
31 return value;
32}
33static int patch_parse(const uint8_t* old_data, size_t old_len,
34 const uint8_t* patch, size_t patch_len,
35 uint8_t* new_data, size_t new_capacity,
36 size_t* result_len, int emit) {
37 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
38 if (!patch || !result_len || (old_len && !old_data) ||
39 patch_len < SNEPPX_PATCH_HEADER_SIZE + 1 ||
40 memcmp(patch, magic, sizeof(magic)) != 0) return -1;
41 uint64_t expected64 = patch_read_u64(patch + 4);
42 if (expected64 > (uint64_t)SIZE_MAX) return -1;
43 size_t expected = (size_t)expected64;
44 if (expected > new_capacity) return -1;
45 size_t pos = SNEPPX_PATCH_HEADER_SIZE, produced = 0;
46 while (pos < patch_len) {
47 uint8_t opcode = patch[pos++];
48 if (opcode == 0) {
49 if (pos != patch_len || produced != expected) return -1;
50 *result_len = produced;
51 return 0;
52 }
53 if (opcode == 1) {
54 if (patch_len - pos < 16) return -1;
55 uint64_t offset64 = patch_read_u64(patch + pos);
56 uint64_t length64 = patch_read_u64(patch + pos + 8);
57 pos += 16;
58 if (offset64 > (uint64_t)old_len || length64 > (uint64_t)SIZE_MAX ||
59 length64 == 0 || length64 > (uint64_t)old_len - offset64) return -1;
60 size_t offset = (size_t)offset64, length = (size_t)length64;
61 if (produced > expected || length > expected - produced) return -1;
62 if (emit) memcpy(new_data + produced, old_data + offset, length);
63 produced += length;
64 continue;
65 }
66 if (opcode == 2) {
67 if (patch_len - pos < 8) return -1;
68 uint64_t length64 = patch_read_u64(patch + pos);
69 pos += 8;
70 if (length64 == 0 || length64 > (uint64_t)SIZE_MAX ||
71 length64 > (uint64_t)(patch_len - pos)) return -1;
72 size_t length = (size_t)length64;
73 if (produced > expected || length > expected - produced) return -1;
74 if (emit) memcpy(new_data + produced, patch + pos, length);
75 pos += length;
76 produced += length;
77 continue;
78 }
79 return -1;
80 }
81 return -1;
82}
83static int patch_read_file(const char* path, uint8_t** data_out, size_t* len_out) {
84 if (!path || !data_out || !len_out) return -1;
85 *data_out = NULL;
86 *len_out = 0;
87 FILE* file = fopen(path, "rb");
88 if (!file) return -1;
89 if (fseek(file, 0, SEEK_END) != 0) { fclose(file); return -1; }
90 long end = ftell(file);
91 if (end < 0 || fseek(file, 0, SEEK_SET) != 0) { fclose(file); return -1; }
92 size_t len = (size_t)end;
93 uint8_t* data = (uint8_t*)malloc(len ? len : 1);
94 if (!data) { fclose(file); return -1; }
95 if (len && fread(data, 1, len, file) != len) {
96 free(data); fclose(file); return -1;
97 }
98 if (fclose(file) != 0) { free(data); return -1; }
99 *data_out = data;
100 *len_out = len;
101 return 0;
102}

108 if (fclose(file) != 0) ok = 0;
109 if (!ok) { remove(path); return -1; }
110 return 0;

165int SNEPPX_bsdiff(const uint8_t* old_data, size_t old_len,
166 const uint8_t* new_data, size_t new_len,
167 uint8_t* patch, size_t* patch_len) {
168 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
169 if (!patch || !patch_len || (old_len && !old_data) || (new_len && !new_data) ||
170 new_len > SIZE_MAX - 64) return -1;
171 size_t prefix = 0;
172 while (prefix < old_len && prefix < new_len && old_data[prefix] == new_data[prefix]) prefix++;
173 size_t suffix = 0;
174 while (suffix < old_len - prefix && suffix < new_len - prefix &&
175 old_data[old_len - 1 - suffix] == new_data[new_len - 1 - suffix]) suffix++;
176 size_t insert_len = new_len - prefix - suffix;
177 size_t required = SNEPPX_PATCH_HEADER_SIZE + 1;
178 if (prefix) required += 17;
179 if (insert_len) required += 9 + insert_len;
180 if (suffix) required += 17;
181 if (*patch_len < required) { *patch_len = required; return -1; }
182 size_t pos = 0;
183 memcpy(patch + pos, magic, sizeof(magic)); pos += sizeof(magic);
184 patch_write_u64(patch + pos, (uint64_t)new_len); pos += 8;
185 if (prefix) {
186 patch[pos++] = 1;
187 patch_write_u64(patch + pos, 0); pos += 8;
188 patch_write_u64(patch + pos, (uint64_t)prefix); pos += 8;
189 }
190 if (insert_len) {
191 patch[pos++] = 2;
192 patch_write_u64(patch + pos, (uint64_t)insert_len); pos += 8;
193 memcpy(patch + pos, new_data + prefix, insert_len); pos += insert_len;
194 }
195 if (suffix) {
196 patch[pos++] = 1;
197 patch_write_u64(patch + pos, (uint64_t)(old_len - suffix)); pos += 8;
198 patch_write_u64(patch + pos, (uint64_t)suffix); pos += 8;
199 }
200 patch[pos++] = 0;
201 *patch_len = pos;
202 return 0;
203}

377static void dep_dfs(SNEPPXDepResolver* dr, int idx, int* dep_graph, int n) {
378 if (dep_cycle_detected) return;

547int SNEPPX_bsdiff_total_size(const uint8_t* patch, size_t patch_len) {
548 if (!patch || patch_len < 4) return 0;
549 size_t total = 0;
550 size_t pos = 0;
551 while (pos + 4 <= patch_len) {
552 uint32_t tag = ((uint32_t)patch[pos] << 24) | ((uint32_t)patch[pos + 1] << 16) | ((uint32_t)patch[pos + 2] << 8) | patch[pos + 3];
553 pos += 4;
554 if (tag > 0x00FFFFFF) {
555 if (pos + 2 > patch_len) break;
556 size_t len = ((size_t)patch[pos] << 8) | patch[pos + 1];
557 pos += 2;
558 if (pos + len > patch_len) break;
559 total += len;
560 pos += len;
561 } else {
562 if (pos + tag > patch_len) break;
563 total += tag;
564 pos += tag;
565 }
566 }
567 return (int)total;
568}

891int SNEPPX_bspatch_apply_patch(const char* old_path, const char* patch_path, const char* new_path) {
892 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
893 if (!old_path || !patch_path || !new_path) return -1;
894 uint8_t *old_data = NULL, *patch = NULL, *new_data = NULL;
895 size_t old_len = 0, patch_len = 0;
896 int rc = patch_read_file(old_path, &old_data, &old_len);
897 if (rc == 0) rc = patch_read_file(patch_path, &patch, &patch_len);
898 if (rc == 0 && (patch_len < SNEPPX_PATCH_HEADER_SIZE + 1 ||
899 memcmp(patch, magic, sizeof(magic)) != 0)) rc = -1;
900 size_t output_len = 0;
901 if (rc == 0) {
902 uint64_t output64 = patch_read_u64(patch + 4);
903 if (output64 > (uint64_t)SIZE_MAX || old_len > SIZE_MAX - patch_len ||
904 output64 > (uint64_t)(old_len + patch_len)) rc = -1;
905 else output_len = (size_t)output64;
906 }
907 if (rc == 0) {
