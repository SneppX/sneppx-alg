23static void patch_write_u64(uint8_t* out, uint64_t value)
24{
25 for (int i = 0; i < 8; i++)
26 out[i] = (uint8_t) (value >> (i * 8));
27}
28static uint64_t patch_read_u64(const uint8_t* in)
29{
30 uint64_t value = 0;
31 for (int i = 0; i < 8; i++)
32 value |= (uint64_t) in[i] << (i * 8);
33 return value;
34}
35static int patch_parse(const uint8_t* old_data, size_t old_len, const uint8_t* patch, size_t patch_len,
36 uint8_t* new_data, size_t new_capacity, size_t* result_len, int emit)
37{
38 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
39 if (!patch || !result_len || (old_len && !old_data) || patch_len < SNEPPX_PATCH_HEADER_SIZE + 1 ||
40 memcmp(patch, magic, sizeof(magic)) != 0)
41 return -1;
42 uint64_t expected64 = patch_read_u64(patch + 4);
43 if (expected64 > (uint64_t) SIZE_MAX)
44 return -1;
45 size_t expected = (size_t) expected64;
46 if (expected > new_capacity)
47 return -1;
48 size_t pos = SNEPPX_PATCH_HEADER_SIZE, produced = 0;
49 while (pos < patch_len)
50 {
51 uint8_t opcode = patch[pos++];
52 if (opcode == 0)
53 {
54 if (pos != patch_len || produced != expected)
55 return -1;
56 *result_len = produced;
57 return 0;
58 }
59 if (opcode == 1)
60 {
61 if (patch_len - pos < 16)
62 return -1;
63 uint64_t offset64 = patch_read_u64(patch + pos);
64 uint64_t length64 = patch_read_u64(patch + pos + 8);
65 pos += 16;
66 if (offset64 > (uint64_t) old_len || length64 > (uint64_t) SIZE_MAX || length64 == 0 ||
67 length64 > (uint64_t) old_len - offset64)
68 return -1;
69 size_t offset = (size_t) offset64, length = (size_t) length64;
70 if (produced > expected || length > expected - produced)
71 return -1;
72 if (emit)
73 memcpy(new_data + produced, old_data + offset, length);
74 produced += length;
75 continue;
76 }
77 if (opcode == 2)
78 {
79 if (patch_len - pos < 8)
80 return -1;
81 uint64_t length64 = patch_read_u64(patch + pos);
82 pos += 8;
83 if (length64 == 0 || length64 > (uint64_t) SIZE_MAX || length64 > (uint64_t) (patch_len - pos))
84 return -1;
85 size_t length = (size_t) length64;
86 if (produced > expected || length > expected - produced)
87 return -1;
88 if (emit)
89 memcpy(new_data + produced, patch + pos, length);
90 pos += length;
91 produced += length;
92 continue;
93 }
94 return -1;
95 }
96 return -1;
97}
98static int patch_read_file(const char* path, uint8_t** data_out, size_t* len_out)
99{
100 if (!path || !data_out || !len_out)
101 return -1;
102 *data_out = NULL;
103 *len_out = 0;
104 FILE* file = fopen(path, "rb");
105 if (!file)
106 return -1;
107 if (fseek(file, 0, SEEK_END) != 0)
108 {
109 fclose(file);
110 return -1;
111 }
112 long end = ftell(file);
113 if (end < 0 || fseek(file, 0, SEEK_SET) != 0)
114 {
115 fclose(file);
116 return -1;
117 }
118 size_t len = (size_t) end;
119 uint8_t* data = (uint8_t*) malloc(len ? len : 1);
120 if (!data)
121 {
122 fclose(file);
123 return -1;
124 }
125 if (len && fread(data, 1, len, file) != len)
126 {
127 free(data);
128 fclose(file);
129 return -1;
130 }
131 if (fclose(file) != 0)
132 {
133 free(data);
134 return -1;
135 }
136 *data_out = data;
137 *len_out = len;
138 return 0;
139}
140static int patch_write_file(const char* path, const uint8_t* data, size_t len)
141{
142 if (!path || (len && !data))
143 return -1;
144 FILE* file = fopen(path, "wb");
145 if (!file)
146 return -1;
147 int ok = !len || fwrite(data, 1, len, file) == len;
148 if (fclose(file) != 0)
149 ok = 0;
150 if (!ok)
151 {
152 remove(path);
153 return -1;
154 }
155 return 0;

219int SNEPPX_bsdiff(const uint8_t* old_data, size_t old_len, const uint8_t* new_data, size_t new_len, uint8_t* patch,
220 size_t* patch_len)
221{
222 static const uint8_t magic[4] = {'S', 'X', 'P', '1'};
223 if (!patch || !patch_len || (old_len && !old_data) || (new_len && !new_data) || new_len > SIZE_MAX - 64)
224 return -1;
225 size_t prefix = 0;
226 while (prefix < old_len && prefix < new_len && old_data[prefix] == new_data[prefix])
227 prefix++;
228 size_t suffix = 0;
229 while (suffix < old_len - prefix && suffix < new_len - prefix &&
230 old_data[old_len - 1 - suffix] == new_data[new_len - 1 - suffix])
231 suffix++;
232 size_t insert_len = new_len - prefix - suffix;
233 size_t required = SNEPPX_PATCH_HEADER_SIZE + 1;
234 if (prefix)
235 required += 17;
236 if (insert_len)
237 required += 9 + insert_len;
238 if (suffix)
239 required += 17;
240 if (*patch_len < required)
241 {
242 *patch_len = required;
243 return -1;
244 }
245 size_t pos = 0;
246 memcpy(patch + pos, magic, sizeof(magic));
247 pos += sizeof(magic);
248 patch_write_u64(patch + pos, (uint64_t) new_len);
249 pos += 8;
250 if (prefix)
251 {
252 patch[pos++] = 1;
253 patch_write_u64(patch + pos, 0);
254 pos += 8;
255 patch_write_u64(patch + pos, (uint64_t) prefix);
256 pos += 8;
257 }
258 if (insert_len)
259 {
260 patch[pos++] = 2;
261 patch_write_u64(patch + pos, (uint64_t) insert_len);
262 pos += 8;
263 memcpy(patch + pos, new_data + prefix, insert_len);
264 pos += insert_len;
265 }
266 if (suffix)
267 {
268 patch[pos++] = 1;
269 patch_write_u64(patch + pos, (uint64_t) (old_len - suffix));
270 pos += 8;
271 patch_write_u64(patch + pos, (uint64_t) suffix);
272 pos += 8;
273 }
274 patch[pos++] = 0;
275 *patch_len = pos;
276 return 0;
277}

1047int SNEPPX_bsdiff_create_patch_file(const char* old_path, const char* new_path, const char* patch_path)
1048{
1049 if (!old_path || !new_path || !patch_path)
1050 return -1;
1051 uint8_t *old_data = NULL, *new_data = NULL, *patch = NULL;
1052 size_t old_len = 0, new_len = 0, patch_len = 0;
1053 int rc = patch_read_file(old_path, &old_data, &old_len);
1054 if (rc == 0)
1055 rc = patch_read_file(new_path, &new_data, &new_len);
1056 if (rc == 0 && new_len > SIZE_MAX - 64)
1057 rc = -1;
1058 if (rc == 0)
1059 {
1060 patch_len = new_len + 64;
1061 patch = (uint8_t*) malloc(patch_len ? patch_len : 1);
1062 if (!patch)
1063 rc = -1;
1064 }
1065 if (rc == 0)
1066 rc = SNEPPX_bsdiff(old_data, old_len, new_data, new_len, patch, &patch_len);

1102 size_t old_len = 0, patch_len = 0;

1104 if (rc == 0)

1117 }
1118 if (rc == 0)
1119 {
1120 new_data = (uint8_t*) malloc(output_len ? output_len : 1);
1121 if (!new_data)
1122 rc = -1;
1123 }
1124 if (rc == 0)
1125 {
1126 size_t capacity = output_len;
1127 rc = SNEPPX_bspatch(old_data, old_len, patch, patch_len, new_data, &capacity);
1128 if (rc == 0 && capacity != output_len)
1129 rc = -1;

1131 if (rc == 0)
