63static uint32_t fuzz_crc32(const uint8_t* data, size_t len)
64{
65 uint32_t crc = 0xFFFFFFFF;
66 static uint32_t table[256];
67 static int table_init = 0;
68 if (!table_init)
69 {
70 for (int i = 0; i < 256; i++)
71 {
72 uint32_t c = i;
73 for (int j = 0; j < 8; j++)
74 c = (c >> 1) ^ (c & 1 ? 0xEDB88320 : 0);
75 table[i] = c;
76 }
77 table_init = 1;
78 }
79 for (size_t i = 0; i < len; i++)
80 crc = table[(crc ^ data[i]) & 0xFF] ^ (crc >> 8);
81 return crc ^ 0xFFFFFFFF;
82}

85{
86 if (*len == 0)

98{
99 if (*len < 2)

121static int fuzz_mutate_splice(uint8_t* buf, size_t* len, size_t max_len)
122{
123 if (corpus_count < 2 || *len < 4)
124 return 0;
125 int src_idx = rand() % corpus_count;

127 return 0;

129 int pos2 = rand() % (corpus[src_idx].len - 2) + 1;
130 int count = (rand() % ((*len - pos1 > corpus[src_idx].len - pos2) ? corpus[src_idx].len - pos2 : *len - pos1)) + 1;
131 memcpy(buf + pos1, corpus[src_idx].data + pos2, count);
132 return 1;

135static int fuzz_mutate_havoc(uint8_t* buf, size_t* len, size_t max_len)
136{
137 int ops = (rand() % 8) + 1;
138 for (int i = 0; i < ops; i++)
139 {
140 int choice = rand() % 4;
141 switch (choice)
142 {
143 case 0:
144 fuzz_mutate_flip(buf, len, max_len);
145 break;
146 case 1:
147 fuzz_mutate_erase(buf, len);
148 break;
149 case 2:
150 fuzz_mutate_insert(buf, len, max_len);
151 break;
152 case 3:
153 if (*len > 0)
154 buf[rand() % (*len)] = rand() & 0xFF;
155 break;
156 }
157 }
158 return 1;
159}

277int SNEPPX_fuzzer_generate(uint8_t* buf, size_t* len, size_t max_len)
278{
279 if (!buf || !len || max_len < 16)
280 return -1;
281 if (corpus_count > 0 && rand() % 3 > 0)
282 {
283 int idx = rand() % corpus_count;
284 if (corpus[idx].len > 0 && corpus[idx].len <= max_len)
285 {
286 memcpy(buf, corpus[idx].data, corpus[idx].len);
287 *len = corpus[idx].len;
288 double r = (double) rand() / RAND_MAX;
289 if (r < fuzzer_config.mutate_prob_flip)
290 fuzz_mutate_flip(buf, len, max_len);
291 else if (r < fuzzer_config.mutate_prob_flip + fuzzer_config.mutate_prob_erase)
292 fuzz_mutate_erase(buf, len);
293 else if (r < fuzzer_config.mutate_prob_flip + fuzzer_config.mutate_prob_erase +
294 fuzzer_config.mutate_prob_insert)
295 fuzz_mutate_insert(buf, len, max_len);
296 else if (r < fuzzer_config.mutate_prob_flip + fuzzer_config.mutate_prob_erase +
297 fuzzer_config.mutate_prob_insert + fuzzer_config.mutate_prob_splice)
298 fuzz_mutate_splice(buf, len, max_len);
299 else
300 fuzz_mutate_havoc(buf, len, max_len);
301 total_mutations++;
302 return 0;
303 }
304 }
305 switch (current_protocol)
306 {
307 case FUZZ_PROTO_TLS:
308 return fuzz_generate_tls_client_hello(buf, len, max_len);
309 case FUZZ_PROTO_HTTP:
310 return fuzz_generate_http_request(buf, len, max_len);
311 case FUZZ_PROTO_QUIC:
312 return fuzz_generate_quic_packet(buf, len, max_len);
313 default:
314 for (size_t i = 0; i < 64 && i < max_len; i++)
315 buf[i] = rand() & 0xFF;
316 *len = 64;
317 return 0;
318 }
319}

329int SNEPPX_fuzzer_execute(const uint8_t* input, size_t len, fuzz_result_t* result)
330{
331 if (!input || !result)
332 return -1;
333 total_executions++;
334 memset(result, 0, sizeof(fuzz_result_t));
335 result->input_crc = fuzz_crc32(input, len);
336 result->execution_id = total_executions;
337 int crash_detected = 0;
338 if (fuzzer_config.enable_crash_detection)
339 {
340 if (len > 0 && input[len - 1] == 0xFF)
341 crash_detected = 1;
342 for (size_t i = 0; i < len - 1 && !crash_detected; i++)
343 if (input[i] == 'C' && input[i + 1] == 'R' && input[i + 2] == 'A' && input[i + 3] == 'S' &&
344 input[i + 4] == 'H')
345 crash_detected = 1;
346 }
347 if (crash_detected)
348 {
349 result->crashed = 1;
350 result->crash_type = FUZZ_CRASH_SEGFAULT;
351 total_crashes++;
352 return 0;
353 }
354 if (fuzzer_config.enable_coverage && coverage_count < 65536)
355 {
356 uint32_t hash = fuzz_crc32(input, len > 16 ? 16 : len);
357 int found = 0;
358 for (int i = 0; i < coverage_count; i++)
359 if (coverage_map[i].pc == hash)
360 {
361 coverage_map[i].hit_count++;
362 found = 1;
363 break;
364 }
365 if (!found)
366 {
367 coverage_map[coverage_count].pc = hash;
368 coverage_map[coverage_count].hit_count = 1;
369 coverage_count++;
370 }
371 result->new_coverage = !found;
372 }
373 if (corpus_count < FUZZER_CORPUS_SIZE && (result->new_coverage || rand() % 100 < 5))
374 {
375 corpus[corpus_count].data = (uint8_t*) malloc(len);
376 if (corpus[corpus_count].data)
377 {
378 memcpy(corpus[corpus_count].data, input, len);
379 corpus[corpus_count].len = len;
380 corpus[corpus_count].crc = result->input_crc;
381 corpus[corpus_count].fitness = result->new_coverage ? 10 : 1;
382 corpus_count++;
383 }
384 }
385 return 0;
386}
