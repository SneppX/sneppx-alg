203 for (size_t i = 0; i < n; i++)
204 {
205 if (cum >= cfg->top_p)
206 probs[indices[i]] = 0.0f;
207 cum += probs[indices[i]];
208 }
209 float new_sum = 0.0f;
210 for (size_t i = 0; i < n; i++)
211 new_sum += probs[i];
212 inv_sum = 1.0f / (new_sum + 1e-10f);
213 for (size_t i = 0; i < n; i++)
214 probs[i] *= inv_sum;
215 }

218 float cumulative = 0.0f;
219 int result = 0;
220 for (size_t i = 0; i < n; i++)
221 {
222 cumulative += probs[i];
223 if (r < cumulative)
224 {
225 result = (int) i;

227 }
228 }
