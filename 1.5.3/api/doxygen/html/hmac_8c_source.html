23/*
24 * HMAC-SHA256 uses SHA-256 for both hash passes. The SHA-512 context used by
25 * the generic path writes a 64-byte digest, which overflowed the 32-byte
26 * output buffer expected by SNEPPX_hmac_sha256 callers.
27 */
28static int hmac_sha256_compute(const uint8_t* key, size_t key_len, const uint8_t* data, size_t data_len,
29 uint8_t out[32])
30{
31 uint8_t k[64], ipad[64], opad[64], inner_hash[32], outer[96];
32 if (key_len > sizeof(k))
33 {
34 SNEPPX_sha256(k, key, key_len);
35 memset(k + 32, 0, sizeof(k) - 32);
36 }
37 else
38 {
39 memcpy(k, key, key_len);
40 memset(k + key_len, 0, sizeof(k) - key_len);
41 }
42 for (size_t i = 0; i < sizeof(k); i++)
43 {
44 ipad[i] = (uint8_t) (k[i] ^ 0x36);
45 opad[i] = (uint8_t) (k[i] ^ 0x5C);
46 }
47 memset(k, 0, sizeof(k));
48
49 if (data_len > SIZE_MAX - sizeof(ipad))
50 return -1;
51 uint8_t* inner = (uint8_t*) malloc(sizeof(ipad) + data_len);
52 if (!inner)
53 return -1;
54 memcpy(inner, ipad, sizeof(ipad));
55 if (data_len > 0)
56 memcpy(inner + sizeof(ipad), data, data_len);
57 SNEPPX_sha256(inner_hash, inner, sizeof(ipad) + data_len);
58 memset(inner, 0, sizeof(ipad) + data_len);
59 free(inner);
60
61 memcpy(outer, opad, sizeof(opad));
62 memcpy(outer + sizeof(opad), inner_hash, sizeof(inner_hash));
63 SNEPPX_sha256(out, outer, sizeof(outer));
64 memset(outer, 0, sizeof(outer));
65 memset(inner_hash, 0, sizeof(inner_hash));
66 memset(ipad, 0, sizeof(ipad));
67 memset(opad, 0, sizeof(opad));
68 return 0;
69}

71static int hmac_compute(const uint8_t* key, size_t key_len, const uint8_t* data, size_t data_len, uint8_t* out,
72 size_t out_len, int use_sha512)
73{
74 if (!use_sha512)
75 {
76 uint8_t digest[32];
77 if (hmac_sha256_compute(key, key_len, data, data_len, digest) != 0)
78 return -1;
79 size_t copy_len = out_len < sizeof(digest) ? out_len : sizeof(digest);
80 memcpy(out, digest, copy_len);
81 memset(digest, 0, sizeof(digest));
82 return 0;
83 }
84
85 (void) out_len;
86 uint8_t k[128], ipad[128], opad[128];
87 size_t block_size = 128;
88 memset(k, 0, block_size);
89 if (key_len > block_size)
90 {
91 SNEPPXSHA512Context ctx;
92 SNEPPX_sha512_init(&ctx);
93 SNEPPX_sha512_update(&ctx, key, key_len);
94 uint8_t h[64];
95 SNEPPX_sha512_finish(&ctx, h);
96 memcpy(k, h, 64);
97 }
98 else
99 memcpy(k, key, key_len);
100 for (size_t i = 0; i < block_size; i++)
101 {
102 ipad[i] = k[i] ^ 0x36;
103 opad[i] = k[i] ^ 0x5C;
104 }
105
106 uint8_t inner_hash[64];
107 {
108 SNEPPXSHA512Context ctx;
109 SNEPPX_sha512_init(&ctx);
110 SNEPPX_sha512_update(&ctx, ipad, block_size);
111 SNEPPX_sha512_update(&ctx, data, data_len);
112 SNEPPX_sha512_finish(&ctx, inner_hash);
113 }
114 {
115 SNEPPXSHA512Context ctx;
116 SNEPPX_sha512_init(&ctx);
117 SNEPPX_sha512_update(&ctx, opad, block_size);
118 SNEPPX_sha512_update(&ctx, inner_hash, sizeof(inner_hash));
119 SNEPPX_sha512_finish(&ctx, out);
120 }
121 memset(k, 0, sizeof(k));
122 memset(ipad, 0, sizeof(ipad));
123 memset(opad, 0, sizeof(opad));
124 return 0;
125}
