51/*
52 * One-way absorption: derive a SipHash output from the current pool and the
53 * incoming bytes, then mix that output into the pool. Using the previous pool
54 * as the key means an observer cannot recover earlier state from the new pool.
55 */
56static void entropy_absorb(SNEPPXEntropyPool* ep, const uint8_t* data, size_t len)
57{
58 uint8_t key[SNEPPX_SIPHASH_KEY_SIZE];
59 uint8_t block[8 + 16];
60 size_t off = 0;
61 uint64_t counter = 0;
62
63 while (off < len)
64 {
65 size_t take = len - off;
66 if (take > 16)
67 take = 16;
68 memcpy(key, ep->pool, sizeof(key));
69 store64_le(block, counter++);
70 memcpy(block + 8, data + off, take);
71 if (take < 16)
72 memset(block + 8 + take, 0, 16 - take);
73 uint64_t h = SNEPPX_siphash(key, block, sizeof(block));
74 for (unsigned int i = 0; i < 8; ++i)
75 ep->pool[(ep->pool_index + (int) i) % SNEPPX_ENTROPY_POOL_SIZE] ^= (uint8_t) (h >> (8U * i));
76 ep->pool_index = (ep->pool_index + 8) % SNEPPX_ENTROPY_POOL_SIZE;
77 off += take;
78 }
79}

81/* Counter-mode SipHash stream. No timing source is used after seeding. */
82static void entropy_extract(const SNEPPXEntropyPool* ep, uint8_t* out, size_t out_len)
83{
84 uint8_t key[SNEPPX_SIPHASH_KEY_SIZE];
85 uint8_t msg[8 + SNEPPX_ENTROPY_POOL_SIZE];
86 uint8_t block[8];
87 uint64_t counter = 0;
88
89 memcpy(key, ep->pool, sizeof(key));
90 memcpy(msg + 8, ep->pool, SNEPPX_ENTROPY_POOL_SIZE);
91 while (out_len > 0)
92 {
93 store64_le(msg, counter++);
94 uint64_t h = SNEPPX_siphash(key, msg, sizeof(msg));
95 store64_le(block, h);
96 size_t take = out_len < 8 ? out_len : 8;
97 memcpy(out, block, take);
98 out += take;
99 out_len -= take;
100 for (unsigned int i = 0; i < 8; ++i)
101 key[i] ^= block[i];
102 for (size_t i = 8; i < sizeof(key); ++i)
103 key[i] ^= key[i - 8];
104 }
105}

235int SNEPPX_entropy_pool_get(SNEPPXEntropyPool* ep, uint8_t* out, size_t out_len)
236{
237 if (!ep || !out)
238 return -1;
239 if (out_len == 0)
240 return 0;
241 if (out_len > INT_MAX)

244 return -1;

246 entropy_extract(ep, out, out_len);
247 /* One-way rekey after generation: old state cannot be recovered. */
248 entropy_absorb(ep, out, out_len);
249
250 ep->entropy_estimate -= (int) out_len;
251 if (ep->entropy_estimate < 0)
