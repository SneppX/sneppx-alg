25static uint8_t gf_log[256];
26static int gf_tables_initialized = 0;

28static void gf_init(void)
29{
30 if (gf_tables_initialized)
31 return;
32 int x = 1;
33 for (int i = 0; i < 255; i++)
34 {
35 gf_exp[i] = (uint8_t) x;
36 gf_log[x] = (uint8_t) i;
37 /* GF(256) modulus 0x11b is irreducible but NOT primitive for
38 * generator 2: x has order 51, leaving 204 field elements with a
39 * zero log entry (and repeatedly clobbering gf_log[1]). 3 is a
40 * primitive element, so enumerate with x*3 = xtime(x) ^ x. Keep
41 * x masked to 8 bits: an unmasked shift made gf_log[x] write far
42 * past its 256-entry bounds. */
43 int doubled = (x << 1) ^ ((x & 0x80) ? 0x1b : 0);
44 x = (doubled ^ x) & 0xff;
45 }
46 for (int i = 255; i < 512; i++)
47 gf_exp[i] = gf_exp[i - 255];
48 gf_tables_initialized = 1;
49}

51static uint8_t gf_mul(uint8_t a, uint8_t b)
52{
53 if (a == 0 || b == 0)
54 return 0;
55 return gf_exp[gf_log[a] + gf_log[b]];
56}

58static uint8_t gf_inv(uint8_t a)
59{
60 if (a == 0)
61 return 0;
62 return gf_exp[255 - gf_log[a]];

74static uint8_t gf_poly_eval(uint8_t* coeff, int k, uint8_t x)
75{
76 uint8_t result = coeff[k - 1];
77 for (int i = k - 2; i >= 0; i--)
78 {
79 result = gf_add(gf_mul(result, x), coeff[i]);
80 }
81 return result;

287{

289 return -1;

337int SNEPPX_shamir_reconstruct(uint8_t** shares, size_t* share_lens, int k, uint8_t* secret, size_t* secret_len)
338{
339 if (!shares || !share_lens || k < 2 || !secret || !secret_len)
340 return -1;
341 gf_init();
342 size_t len = share_lens[0] - 1;
343 if (*secret_len < len)
344 return -1;
345 *secret_len = len;
346 uint8_t x_vals[16];
347 for (int i = 0; i < k; i++)
348 x_vals[i] = shares[i][0];
349 for (size_t pos = 0; pos < len; pos++)
350 {
351 uint8_t result = 0;
352 for (int j = 0; j < k; j++)
353 {
354 uint8_t num = 1, den = 1;
355 uint8_t yj = shares[j][1 + pos];
356 for (int m = 0; m < k; m++)
357 {
358 if (m == j)
359 continue;
360 num = gf_mul(num, x_vals[m]);
361 den = gf_mul(den, gf_add(x_vals[j], x_vals[m]));
362 }
363 result = gf_add(result, gf_mul(yj, gf_mul(num, gf_inv(den))));
364 }
365 secret[pos] = result;
366 }
367 return 0;
368}

474static void dash_parse_json(const char* json)
475{
476 if (!json)
477 return;
478 int brace_depth = 0;
479 int key_count = 0;
480 int arr_count = 0;
481 int colon_count = 0;
482 int comma_count = 0;
483 int in_string = 0;
484 char prev = 0;
485 while (*json)
486 {
487 char c = *json;
488 if (c == '"' && prev != '\\')
489 in_string = !in_string;
490 if (!in_string)
491 {
492 if (c == '{')
493 brace_depth++;
494 if (c == '}')
495 brace_depth--;
496 if (c == '[')
497 arr_count++;
498 if (c == ']')
499 arr_count--;
500 if (c == ':')
501 {
502 colon_count++;
503 if (brace_depth > 0)
504 key_count++;
505 }
506 if (c == ',')
507 comma_count++;
508 }
509 prev = c;
510 json++;
511 }
512 dash_state.total_requests++;
513 dash_state.active_connections = arr_count > 0 ? arr_count : dash_state.active_connections;
514 if (comma_count > 20)
515 dash_state.errors_logged++;
516 if (key_count > 8)
517 dash_state.critical_alerts++;
518 else if (key_count > 4)
519 dash_state.warnings++;
520 else if (key_count > 0)
521 dash_state.info_count++;
522}

811 if (slot < 0)
812 return -1;
813 uint8_t plain[HSM_MAX_DATA];
814 size_t plain_len = hsm_slots[slot].data_len;
815 hsm_xor_crypt(hsm_slots[slot].encrypted, plain, plain_len, hsm_xor_key);
816 int wrap_slot = hsm_find_slot(wrapping_key_id, NULL);
817 if (wrap_slot < 0)
818 return -1;
819 uint8_t wk[32];
820 hsm_xor_crypt(hsm_slots[wrap_slot].encrypted, wk, 32, hsm_xor_key);
821 if (*wrapped_len < plain_len + 8)
822 return -1;
823 *wrapped_len = plain_len + 8;
824 for (size_t i = 0; i < plain_len; i++)
825 wrapped_out[i] = plain[i] ^ wk[i % 32];
826 for (size_t i = 0; i < 4; i++)
827 wrapped_out[plain_len + i] = wk[i];
828 wrapped_out[plain_len + 4] = 0xAB;
829 wrapped_out[plain_len + 5] = 0xCD;
830 wrapped_out[plain_len + 6] = 0xEF;
831 wrapped_out[plain_len + 7] = 0x01;
832 return 0;
833}

844int SNEPPX_hsm_unwrap_key(const uint8_t* wrapped, size_t wrapped_len, const uint8_t* wrapping_key_id,
845 uint8_t* key_id_out)
846{
847 if (!wrapped || !wrapping_key_id || !key_id_out || wrapped_len < 8)
848 return -1;
849 if (!hsm_initialized)
850 return -1;
851 int wrap_slot = hsm_find_slot(wrapping_key_id, NULL);
852 if (wrap_slot < 0)
853 return -1;
854 uint8_t wk[32];
855 hsm_xor_crypt(hsm_slots[wrap_slot].encrypted, wk, 32, hsm_xor_key);
856 size_t plain_len = wrapped_len - 8;
857 uint8_t plain[HSM_MAX_DATA];
858 for (size_t i = 0; i < plain_len; i++)
859 plain[i] = wrapped[i] ^ wk[i % 32];
860 uint8_t new_id[32];
861 if (SNEPPX_random_bytes(new_id, sizeof(new_id)) != 0)
862 return -1;
863 int free_slot = -1;
864 hsm_find_slot(new_id, &free_slot);
865 if (free_slot < 0)
866 return -1;
867 memcpy(hsm_slots[free_slot].key_id, new_id, 32);
868 hsm_xor_crypt(plain, hsm_slots[free_slot].encrypted, plain_len, hsm_xor_key);
869 hsm_slots[free_slot].data_len = plain_len;
870 hsm_slots[free_slot].occupied = 1;
871 memcpy(key_id_out, new_id, 32);
872 return 0;
873}
