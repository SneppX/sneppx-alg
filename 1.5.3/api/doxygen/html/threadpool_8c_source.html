88typedef struct WorkNode
89{
90 SNEPPXTask task;
91 struct WorkNode* next;
92} WorkNode;

94typedef struct
95{
96 WorkNode* head; /* front (stolen from) */
97 WorkNode* tail; /* back (owner pushes) */
98 SNEPPX_mutex_t lock;
99} WorkDeque;

101static void deque_init(WorkDeque* q)
102{
103 q->head = q->tail = NULL;
104 SNEPPX_mutex_init(&q->lock);
105}
106
107static void deque_destroy(WorkDeque* q)
108{
109 /* Drain remaining */
110 SNEPPX_mutex_lock(&q->lock);
111 WorkNode* n = q->head;
112 while (n)
113 {
114 WorkNode* next = n->next;
115 free(n);
116 n = next;
117 }
118 q->head = q->tail = NULL;
119 SNEPPX_mutex_unlock(&q->lock);
120 SNEPPX_mutex_destroy(&q->lock);
121}
122
123/* Owner pushes to the back (LIFO for locality). */
124static int deque_push(WorkDeque* q, SNEPPXTask task)
125{
126 WorkNode* n = (WorkNode*) malloc(sizeof(WorkNode));
127 if (!n)
128 return -1;
129 n->task = task;
130 n->next = NULL;
131 SNEPPX_mutex_lock(&q->lock);
132 if (q->tail)
133 {
134 q->tail->next = n;
135 q->tail = n;
136 }
137 else
138 {
139 q->head = q->tail = n;
140 }
141 SNEPPX_mutex_unlock(&q->lock);
142 return 0;
143}
144
145/* Owner pops from the back. Returns 0 on success, -1 if empty. */
146static int deque_pop(WorkDeque* q, SNEPPXTask* out)
147{
148 SNEPPX_mutex_lock(&q->lock);
149 WorkNode* prev = NULL;
150 WorkNode* cur = q->head;
151 while (cur && cur != q->tail)
152 {
153 prev = cur;
154 cur = cur->next;
155 }
156 if (!cur)
157 {
158 SNEPPX_mutex_unlock(&q->lock);
159 return -1;
160 }
161 *out = cur->task;
162 if (prev)
163 prev->next = NULL;
164 else
165 q->head = NULL;
166 q->tail = prev;
167 free(cur);
168 SNEPPX_mutex_unlock(&q->lock);
169 return 0;
170}
171
172/* Steal from the front. Returns 0 on success, -1 if empty. */
173static int deque_steal(WorkDeque* q, SNEPPXTask* out)
174{
175 SNEPPX_mutex_lock(&q->lock);
176 WorkNode* n = q->head;
177 if (!n)
178 {
179 SNEPPX_mutex_unlock(&q->lock);
180 return -1;
181 }
182 *out = n->task;
183 q->head = n->next;
184 if (!q->head)
185 q->tail = NULL;
186 free(n);
187 SNEPPX_mutex_unlock(&q->lock);
188 return 0;
189}
190
191/* Number of tasks currently in the deque. */
192static int deque_count(WorkDeque* q)
193{
194 SNEPPX_mutex_lock(&q->lock);
195 int cnt = 0;
196 for (WorkNode* n = q->head; n; n = n->next)
197 cnt++;
198 SNEPPX_mutex_unlock(&q->lock);
199 return cnt;
200}

335/* Worker thread entry point */
336#ifdef _WIN32
337static DWORD WINAPI worker_entry(LPVOID arg)
338#else
339static void* worker_entry(void* arg)
340#endif
341{
342 SNEPPXThreadPool* pool = (SNEPPXThreadPool*) ((void**) arg)[0];
343 int tid = (int) (intptr_t) ((void**) arg)[1];
344 set_thread_id(tid);
345 free(arg);
346
347 SNEPPXTask task;
348 for (;;)
349 {
350 int got = 0;
351
352 /* 1. Try own queue (pop back — LIFO) */
353 if (deque_pop(&pool->queues[tid], &task) == 0)
354 {
355 got = 1;
356 }
357 else
358 {
359 /* 2. Steal from other queues (front — FIFO) */
360 for (size_t i = 1; i < pool->num_threads; i++)
361 {
362 size_t victim = (tid + i) % pool->num_threads;
363 if (deque_steal(&pool->queues[victim], &task) == 0)
364 {
365 got = 1;
366 break;
367 }
368 }
369 }
370
371 if (got)
372 {
373 task.func(task.arg);
374 SNEPPX_mutex_lock(&pool->global_lock);
375 pool->active_tasks--;
376 SNEPPX_mutex_unlock(&pool->global_lock);
377 continue;
378 }
379
380 /*
381 * Check shutdown and wait under the same mutex used by destroy().
382 * This prevents a broadcast from being lost between the predicate
383 * check and the condition-variable wait.
384 */
385 SNEPPX_mutex_lock(&pool->global_lock);
386 pool->waiting_count++;
387 while (!pool->shutdown && pool->active_tasks == 0)
388 {
389 SNEPPX_cond_wait(&pool->global_cond, &pool->global_lock);
390 }
391 int should_shutdown = pool->shutdown;
392 pool->waiting_count--;
393 SNEPPX_mutex_unlock(&pool->global_lock);
394 if (should_shutdown)
395 break;
396 }
397 return 0;
398}

406{

416 pool->active_tasks = 0;

497/* Submit a task to a thread's queue (round-robin via atomic index or random). */
498static volatile long g_rr_counter = 0;
499
507int SNEPPX_threadpool_submit(SNEPPXThreadPool* pool, SNEPPXTask task)
508{
509 size_t idx;
510 if (g_thread_id >= 0)
511 {
512 idx = (size_t) g_thread_id; /* own queue */
513 }
514 else
515 {
516#ifdef _WIN32
517 idx = (size_t) (_InterlockedExchangeAdd(&g_rr_counter, 1) % (long) pool->num_threads);
518#else
519 idx = (size_t) (__sync_fetch_and_add(&g_rr_counter, 1) % (long) pool->num_threads);
520#endif
521 }
522 if (idx >= pool->num_threads)
523 idx = 0;
524
525 int ret = deque_push(&pool->queues[idx], task);
526 if (ret != 0)
527 return ret;
528
529 SNEPPX_mutex_lock(&pool->global_lock);
530 pool->active_tasks++;
531 /* Wake one sleeper */
532 if (pool->waiting_count > 0)
533 SNEPPX_cond_signal(&pool->global_cond);
534 SNEPPX_mutex_unlock(&pool->global_lock);
535 return 0;
536}

578void SNEPPX_threadpool_wait(SNEPPXThreadPool* pool)
579{
580 /* Busy-wait until all tasks are done. Workers are guaranteed to wake
581 * for every submitted task because the predicate-based cond_wait loop
582 * cannot lose signals. */
583 while (1)
584 {
585 int all_empty = 1;
586 for (size_t i = 0; i < pool->num_threads; i++)
587 {
588 if (deque_count(&pool->queues[i]) > 0)
589 {
590 all_empty = 0;
591 break;
592 }
593 }
594 SNEPPX_mutex_lock(&pool->global_lock);
595 int active_tasks = pool->active_tasks;
596 SNEPPX_mutex_unlock(&pool->global_lock);
597 if (all_empty && active_tasks == 0)
598 break;
599#ifdef _WIN32
600 SwitchToThread();
601#else
602 sched_yield();
603#endif
604 }
605}

671{
672 size_t range = end - start;
673 size_t nthreads = pool->num_threads;
674 if (range == 0)
675 return;
676 if (nthreads <= 1 || range < 64)

678 func(start, end, arg);
679 return;
680 }

682 size_t chunk_size = (range + nthreads - 1) / nthreads;
683 size_t num_tasks = 0;

686 {

690 ForTaskArg* fta = (ForTaskArg*) malloc(sizeof(ForTaskArg));
691 if (!fta)

703 task.func = for_task_func;
704 task.arg = fta;
