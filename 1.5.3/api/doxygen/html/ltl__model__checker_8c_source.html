147{

166 if (g_fv_model.states[i].state_id == to_state)
167 t = i;
168 if (g_fv_model.states[i].state_id == from_state)
169 f = i;
170 }

175 return -1;
176 for (int j = 0; j < (int) g_fv_model.states[f].next_count; j++)
177 {
178 if (g_fv_model.states[f].next_states[j] == to_state)
179 return 0;
180 }
