142SNEPPXFilterResult SNEPPX_prompt_filter_scan(SNEPPXPromptFilter* pf, const char* prompt, size_t len)
143{
144 if (!pf || !pf->enabled || !prompt)
145 return SNEPPX_FILTER_CLEAN;
146 total_scans++;
147 char lower[SNEPPX_PATTERN_MAX_LEN];
148 size_t plen = (len < SNEPPX_PATTERN_MAX_LEN - 1) ? len : SNEPPX_PATTERN_MAX_LEN - 1;
149 for (size_t i = 0; i < plen; i++)
150 lower[i] = (char) tolower((unsigned char) prompt[i]);
151 lower[plen] = '\0';
152
153 SNEPPXFilterResult highest = SNEPPX_FILTER_CLEAN;
154 int highest_prio = -1;
155 for (int i = 0; i < pf->pattern_count; i++)
156 {
157 if (!pf->patterns[i].is_active)
158 continue;
159 if (strstr(lower, pf->patterns[i].pattern))
160 {
161 if (pattern_priorities[i] > highest_prio)
162 {
163 highest_prio = pattern_priorities[i];
164 highest = pf->patterns[i].classification;
165 }
166 }
167 }
168 if (highest != SNEPPX_FILTER_CLEAN)
169 total_blocks++;
170 return highest;
171}

248int SNEPPX_prompt_filter_match_wildcard(const char* text, const char* pattern)
249{
250 if (!text || !pattern)
251 return 0;
252 if (pattern[0] == '\0')
253 return text[0] == '\0';
254 if (pattern[0] == '*')
255 {
256 for (int i = 0; text[i]; i++)
257 if (SNEPPX_prompt_filter_match_wildcard(text + i, pattern + 1))
258 return 1;
259 return SNEPPX_prompt_filter_match_wildcard(text, pattern + 1);
260 }
261 if (pattern[0] == '?' && text[0])
262 {
263 return SNEPPX_prompt_filter_match_wildcard(text + 1, pattern + 1);
264 }
265 if (pattern[0] == text[0])
266 {
267 return SNEPPX_prompt_filter_match_wildcard(text + 1, pattern + 1);
268 }
269 return 0;

498static void to_lower_utf8(char* dst, const char* src, size_t max_len)
499{
500 if (!dst || !src)
501 return;
502 size_t i = 0, j = 0;
503 while (src[i] && j < max_len - 1)
504 {
505 int clen = utf8_char_len((unsigned char) src[i]);
506 if (clen == 1)
507 {
508 dst[j++] = (char) tolower((unsigned char) src[i++]);
509 }
510 else
511 {
512 for (int k = 0; k < clen && src[i + k] && j < max_len - 1; k++)
513 dst[j++] = src[i + k];
514 i += clen;
515 }
516 }
517 dst[j] = '\0';
518}

689static size_t hex_decode_local(const char* in, size_t in_len, char* out)
690{
691 size_t out_pos = 0;
692 for (size_t i = 0; i + 1 < in_len; i += 2)
693 {
694 unsigned char c1 = (unsigned char) in[i], c2 = (unsigned char) in[i + 1];
695 unsigned char v1 = (c1 >= 'a' ? c1 - 'a' + 10 : (c1 >= 'A' ? c1 - 'A' + 10 : c1 - '0'));
696 unsigned char v2 = (c2 >= 'a' ? c2 - 'a' + 10 : (c2 >= 'A' ? c2 - 'A' + 10 : c2 - '0'));
697 out[out_pos++] = (char) ((v1 << 4) | v2);
698 }
699 return out_pos;
700}
701static const char* b64_alphabet_local = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
702static int b64_rev_local[256];
703static int b64_init_local = 0;
704static void b64_build_rev_local(void)
705{
706 if (b64_init_local)
707 return;
708 b64_init_local = 1;
709 for (int i = 0; i < 256; i++)
710 b64_rev_local[i] = -1;
711 for (int i = 0; i < 64; i++)
712 b64_rev_local[(int) b64_alphabet_local[i]] = i;
713}
714static size_t b64_decode_local(const char* in, size_t in_len, char* out)
715{
716 b64_build_rev_local();
717 size_t out_pos = 0;
718 int val = 0, valb = -8;
719 for (size_t i = 0; i < in_len; i++)
720 {
721 unsigned char c = (unsigned char) in[i];
722 if (c == '=')
723 break;
724 int d = b64_rev_local[c];
725 if (d == -1)
726 continue;
727 val = (val << 6) | d;
728 valb += 6;
729 if (valb >= 0)
730 {
731 out[out_pos++] = (char) ((val >> valb) & 0xFF);
732 valb -= 8;
733 }
734 }
735 return out_pos;
736}
