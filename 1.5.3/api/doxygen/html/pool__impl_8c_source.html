89void* SNEPPX_mem_chunk_carve(SNEPPXMemChunk* chunk, size_t block_size, size_t alignment)
90{
91 if (!chunk || !chunk->mem || block_size == 0)
92 return NULL;
93 if (alignment == 0)
94 alignment = 16;
95 if (alignment & (alignment - 1))
96 alignment = 16;
97 uintptr_t base = (uintptr_t) chunk->mem;
98 size_t used = 0;
99 for (int i = 0; i < chunk->slab_count; i++)
100 {
101 size_t slab_sz = chunk->size_class_idx > 0 ? (1 << (chunk->size_class_idx + 4)) : 64;
102 used += slab_sz;
103 }
104 uintptr_t candidate = base + used;
105 uintptr_t aligned = (candidate + alignment - 1) & ~(uintptr_t) (alignment - 1);
106 size_t needed = (aligned - candidate) + block_size;
107 if (used + needed > chunk->size)
108 return NULL;
109 chunk->slab_count++;
110 return (void*) aligned;
111}

135void SNEPPX_lockfree_stack_push(void* stack_ptr, void* node_ptr)
136{
137 if (!stack_ptr || !node_ptr)
