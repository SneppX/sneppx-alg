201int SNEPPX_profiler_record(SNEPPX_Profiler* prof, const char* name, float elapsed_ms)
202{
203 if (!prof || !name || !prof->enabled)
204 return -1;
205 for (int i = 0; i < prof->num_entries; i++)
206 {
207 if (strcmp(prof->entries[i].name, name) == 0)
208 {
209 SNEPPX_ProfilerEntry* e = &prof->entries[i];
210 e->num_calls++;
211 e->total_time_ms += elapsed_ms;
212 if (elapsed_ms < e->min_time_ms)
213 e->min_time_ms = elapsed_ms;
214 if (elapsed_ms > e->max_time_ms)
215 e->max_time_ms = elapsed_ms;
216 e->avg_time_ms = e->total_time_ms / e->num_calls;
217 return 0;
218 }
219 }
220 if (prof->num_entries >= SNEPPX_PROFILER_MAX_ENTRIES)
221 return -1;
222 SNEPPX_ProfilerEntry* e = &prof->entries[prof->num_entries++];
223 strncpy(e->name, name, SNEPPX_PROFILER_NAME_MAX - 1);
224 e->name[SNEPPX_PROFILER_NAME_MAX - 1] = '\0';
225 e->num_calls = 1;
226 e->total_time_ms = elapsed_ms;
227 e->min_time_ms = elapsed_ms;
