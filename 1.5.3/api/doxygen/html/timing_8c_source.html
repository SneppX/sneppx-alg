107void SNEPPX_timing_random_delay(uint32_t min_ns, uint32_t max_ns)
108{
109 if (max_ns <= min_ns)
110 return;
111 uint32_t range = max_ns - min_ns;
112 uint32_t r;
113 SNEPPX_random_bytes((uint8_t*) &r, sizeof(r));
114 uint32_t delay = min_ns + (r % range);
115 uint64_t start = SNEPPX_timing_start();
116 while (1)
117 {
118 uint64_t elapsed = SNEPPX_timing_end() - start;
119 if (elapsed >= delay)
