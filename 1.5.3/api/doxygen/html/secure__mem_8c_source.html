196void* SNEPPX_secure_malloc(SNEPPXSecurePool* pool, size_t size, size_t alignment)
197{
198 if (!pool || !size)
199 return NULL;
200 if (alignment < 16)
201 alignment = 16;
202 size_t header = pool->use_canaries ? SNEPPX_CANARY_SIZE : 0;
203 size_t aligned_offset = (pool->used + alignment - 1) & ~(alignment - 1);
204 size_t needed = aligned_offset + header + size + header;
205 if (needed > pool->capacity)
206 return NULL;
207 uint8_t* ptr = pool->base + aligned_offset + header;
208 if (pool->use_canaries)
209 {
210 SNEPPXCanary canary;
211 SNEPPX_canary_generate(&canary);
212 SNEPPX_canary_write(&canary, ptr - SNEPPX_CANARY_SIZE);
213 SNEPPX_canary_write(&canary, ptr + size);
214 }
215 pool->used = needed;
216 if (pool->used > pool->peak)
217 pool->peak = pool->used;
218 pool->alloc_count++;
219 return ptr;
220}

228void SNEPPX_secure_pool_free(SNEPPXSecurePool* pool, void* ptr, size_t size)
229{
230 if (!pool || !ptr)
231 return;
232 if (pool->use_canaries)
233 {
234 uint8_t* mem = (uint8_t*) ptr;
235 SNEPPXCanary* lead = (SNEPPXCanary*) (mem - SNEPPX_CANARY_SIZE);
236 SNEPPXCanary* trail = (SNEPPXCanary*) (mem + size);
237 if (!SNEPPX_canary_verify(lead, lead->value))
238 {
239 fprintf(stderr, "HEAP CORRUPTION: leading canary mismatch at %p\n", ptr);
240 abort();
241 }
242 if (!SNEPPX_canary_verify(trail, trail->value))
243 {
244 fprintf(stderr, "HEAP CORRUPTION: trailing canary mismatch at %p\n", ptr);
245 abort();
246 }
247 }
248 SNEPPX_secure_zero(ptr, size);
249}
