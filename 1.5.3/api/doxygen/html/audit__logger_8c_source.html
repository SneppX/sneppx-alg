24static uint32_t audit_crc32(const void* data, size_t len)
25{
26 const unsigned char* buf = (const unsigned char*) data;
27 uint32_t crc = 0xFFFFFFFF;
28 for (size_t i = 0; i < len; i++)
29 {
30 crc ^= buf[i];
31 for (int j = 0; j < 8; j++)
32 crc = (crc >> 1) ^ (0xEDB88320 & -(crc & 1));
33 }
34 return ~crc;
35}

79int SNEPPX_audit_log(SNEPPXAuditLogger* audit, int event_type, const char* description, uint64_t related_address)
80{
81 if (!audit || !audit->enabled || !description)
82 return -1;
83 if (audit->entry_count >= SNEPPX_AUDIT_MAX_ENTRIES)
84 return -1;
85 SNEPPXAuditEntry* e = &audit->entries[audit->entry_count++];
86 e->timestamp = (uint64_t) time(NULL);
87 e->event_type = event_type;
88 strncpy(e->description, description, SNEPPX_AUDIT_DESC_LEN - 1);
89 e->related_address = related_address;
90 e->crc = audit_crc32(e, offsetof(SNEPPXAuditEntry, crc));
91 audit->chain_crc = audit_crc32(&audit->chain_crc, sizeof(audit->chain_crc)) ^ e->crc;
92 return 0;
93}

125int SNEPPX_audit_verify_chain(SNEPPXAuditLogger* audit)
126{
127 if (!audit)
128 return 0;
129 uint32_t expected_chain = 0;
130 for (int i = 0; i < audit->entry_count; i++)
131 {
132 uint32_t entry_crc = audit_crc32(&audit->entries[i], offsetof(SNEPPXAuditEntry, crc));
133 if (entry_crc != audit->entries[i].crc)
134 return 0;
135 expected_chain = audit_crc32(&expected_chain, sizeof(expected_chain)) ^ entry_crc;
136 }
137 return (expected_chain == audit->chain_crc) ? 1 : 0;
138}

221int SNEPPX_audit_sign_entry(SNEPPXAuditLogger* audit, int index, const uint8_t* key)
222{
223 if (!audit || !key)
224 return -1;
225 if (index < 0 || index >= audit->entry_count)
226 return -1;
227 SNEPPXAuditEntry* e = &audit->entries[index];
228 uint32_t sig = 0;
229 for (size_t i = 0; i < SNEPPX_AUDIT_HMAC_KEY_LEN; i++)
230 sig ^= ((uint32_t) key[i] << ((i % 4) * 8));
231 sig ^= e->crc;
232 e->crc = sig;
233 return 0;
