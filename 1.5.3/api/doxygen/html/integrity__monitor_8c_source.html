300static uint32_t crc32_c(const void* data, size_t len)
301{
302 const unsigned char* buf = (const unsigned char*) data;
303 uint32_t crc = 0xFFFFFFFF;
304 for (size_t i = 0; i < len; i++)
305 {
306 crc ^= buf[i];
307 for (int j = 0; j < 8; j++)
308 crc = (crc >> 1) ^ (0xEDB88320 & -(crc & 1));
309 }
310 return ~crc;
311}

447int SNEPPX_monitor_timing_check(uint64_t elapsed_us)
448{
449 if (g_timing_samples < 5)
450 {
451 g_timing_samples++;
452 return 0;
453 }
454 if (g_sliding_window_count < SNEPPX_MONITOR_SLIDING_WINDOW_SIZE)
455 {
456 g_sliding_window[g_sliding_window_count++] = (double) elapsed_us;
457 }
458 else
459 {
460 g_sliding_window[g_sliding_window_index] = (double) elapsed_us;
461 g_sliding_window_index = (g_sliding_window_index + 1) % SNEPPX_MONITOR_SLIDING_WINDOW_SIZE;
462 }
463 double sum = 0;
464 int n = g_sliding_window_count > 0 ? g_sliding_window_count : 1;
465 for (int i = 0; i < g_sliding_window_count; i++)
466 sum += g_sliding_window[i];
467 double avg = sum / n;
468 double variance = 0;
469 for (int i = 0; i < g_sliding_window_count; i++)
470 {
471 double diff = g_sliding_window[i] - avg;
472 variance += diff * diff;
473 }
474 double effective_stddev = (n > 1) ? sqrt(variance / (n - 1)) : g_timing_stddev;
475 if (effective_stddev < 0.001)
476 effective_stddev = g_timing_stddev;
477 double z = (fabs((double) elapsed_us - g_timing_baseline)) / effective_stddev;
478 if (z > 3.0)
479 {
480 char desc[256];
481 snprintf(desc, sizeof(desc), "Timing anomaly: z-score %.2f (elapsed %llu us, baseline %.0f us)", z,
482 (unsigned long long) elapsed_us, g_timing_baseline);
483 fire_event(SNEPPX_MONITOR_EVENT_CANARY_TRIGGERED, desc, 0, 0);
484 return 1;
485 }
486 return 0;
487}

785 if (g_region_freq[i].min_interval_us == 0 || interval < g_region_freq[i].min_interval_us)
786 g_region_freq[i].min_interval_us = interval;
787 if (interval > g_region_freq[i].max_interval_us)
788 g_region_freq[i].max_interval_us = interval;
789 }
790 g_region_freq[i].last_check_time = now_us;
791 if (interval > 0 && interval < 100 && g_region_freq[i].check_count > 5)
792 {
793 char desc[256];
794 snprintf(desc, sizeof(desc), "Anti-fuzzing: region '%s' checked too often (interval %llu us)",
795 g_regions[i].name, (unsigned long long) interval);
796 fire_event(SNEPPX_MONITOR_EVENT_FREQ_ANOMALY, desc, 0, 0);
797 violations++;
798 }
799 if (g_region_freq[i].check_count > 50 && g_last_verify_time > 0)
800 {
801 uint64_t total_span = now_us - g_last_verify_time;
802 if (total_span > 0)
803 {
804 double checks_per_sec = (double) g_region_freq[i].check_count / ((double) total_span / 1000000.0);
805 if (checks_per_sec < 0.01 && g_region_freq[i].check_count > 0)
806 {
807 char desc[256];
808 snprintf(desc, sizeof(desc), "Evasion suspect: region '%s' checked only %.2f/sec (%d checks)",
809 g_regions[i].name, checks_per_sec, g_region_freq[i].check_count);
810 fire_event(SNEPPX_MONITOR_EVENT_FREQ_ANOMALY, desc, 0, 0);
811 violations++;
812 }
813 }
814 }
815 uint32_t current = crc32_c(g_regions[i].addr, g_regions[i].size);
816 if (current != g_regions[i].baseline_crc)
817 {
818 char desc[256];
819 snprintf(desc, sizeof(desc), "Region '%s' modified: CRC mismatch (expected 0x%08X, got 0x%08X)",
820 g_regions[i].name, g_regions[i].baseline_crc, current);
821 fire_event(SNEPPX_MONITOR_EVENT_TEXT_MODIFIED, desc, (uint64_t) (uintptr_t) g_regions[i].addr,
822 g_regions[i].size);
823 violations++;
824 }
825 }
826
827 if (!violations)
828 {
829 int canary_ok = SNEPPX_monitor_check_canary();
830 if (!canary_ok)
831 {
832 fire_event(SNEPPX_MONITOR_EVENT_CANARY_TRIGGERED, "Stack canary corrupted", 0, 0);
833 violations++;
834 }
835 }
836
837 violations += SNEPPX_monitor_freq_analyze();
838 violations += SNEPPX_monitor_api_hook_check();
839
840 if (g_auto_learn_enabled)
841 {
842 uint64_t now = (uint64_t) time(NULL);
843 if (now - g_auto_learn_start < g_auto_learn_duration)
844 {
845 for (int i = 0; i < g_region_count; i++)
846 {
847 if (g_regions[i].active)
848 {
849 uint32_t current = crc32_c(g_regions[i].addr, g_regions[i].size);
850 if (current != g_regions[i].baseline_crc)
851 {
852 g_regions[i].baseline_crc = current;
853 }
854 }
855 }
856 memcpy(g_event_type_baselines, g_event_type_counts, sizeof(g_event_type_baselines));
857 }
858 else
859 {
860 g_auto_learn_enabled = 0;
861 }
862 }
863
864 return violations;
865}
