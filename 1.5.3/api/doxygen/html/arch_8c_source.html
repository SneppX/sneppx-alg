336int SNEPPX_model_forward(SNEPPXModel* model, const SNEPPXTensor* input, SNEPPXTensor** output)
337{
338 if (!model || !input || !output)
339 return 1;
340
341 SNEPPXTensor* current = NULL;
342
343 /* For attention models with embedding, do embedding lookup on token indices */
344 if (model->attention && model->embed_weight && input->ndim == 2)
345 {
346 /* Input: [B, S] token IDs stored as floats */
347 size_t B = input->shape[0], S = input->shape[1];
348 size_t D = model->attention->config.d_model;
349 (void) D;
350 SNEPPXTensor* idx_t = SNEPPX_tensor_create(input->shape, input->ndim, SNEPPX_INT32);
351 if (!idx_t)
352 return 1;
353 size_t n = B * S;
354 int* id = (int*) idx_t->data;
355 float* fd = (float*) input->data;
356 for (size_t i = 0; i < n; i++)
357 id[i] = (int) fd[i];
358 current = SNEPPX_tensor_embedding(model->embed_weight, idx_t);
359 SNEPPX_tensor_destroy(idx_t);
360 if (current)
361 {
362 size_t sh3[] = {B, S, D};
363 SNEPPXTensor* r = SNEPPX_tensor_reshape(current, sh3, 3);
364 if (r)
365 {
366 SNEPPX_tensor_destroy(current);
367 current = r;
368 }
369 }
370 }
371 else if (input->ndim == 3)
372 {
373 size_t b = input->shape[0], s = input->shape[1], d = input->shape[2];
374 size_t sh[] = {b * s, d};
375 current = SNEPPX_tensor_create(sh, 2, SNEPPX_FLOAT32);
376 if (current)
377 memcpy(current->data, input->data, b * s * d * sizeof(float));
378 }
379 else if (input->ndim == 2)
380 {
381 current = SNEPPX_tensor_create(input->shape, input->ndim, SNEPPX_FLOAT32);
382 if (current)
383 memcpy(current->data, input->data, input->size * sizeof(float));
384 }
385 if (!current)
386 return 1;
387
388 if (model->attention && model->embed_weight)
389 {
390 size_t S = current->ndim >= 3 ? current->shape[1] : current->shape[0];
391 SNEPPXTensor* cos_full =
392 SNEPPX_rope_precompute(S, model->attention->config.head_dim, model->attention->config.rope_base);
393 if (cos_full)
394 {
395 SNEPPXTensor* attn_out = SNEPPX_attn_forward(model->attention, current, cos_full, cos_full);
396 SNEPPX_tensor_destroy(cos_full);
397 if (attn_out)
398 {
399 SNEPPX_tensor_destroy(current);
400 current = attn_out;
401 }
402 }
403 }
404
405 if (model->hss_model)
406 {
407 SNEPPXTensor* hss_out = NULL;
408 if (SNEPPX_hss_forward(model->hss_model, current, &hss_out) == 0 && hss_out)
409 {
410 SNEPPX_tensor_destroy(current);
411 current = hss_out;
412 }
413 }
414
415 if (model->ser_model)
416 {
417 size_t seq = current->ndim == 2 ? current->shape[0] : current->shape[1];
418 size_t dim = current->shape[current->ndim - 1];
419 SNEPPXTensor flat;
420 size_t fsh[] = {seq, dim};
421 flat.data = current->data;
422 flat.shape = fsh;
423 flat.ndim = 2;
424 flat.size = seq * dim;
425 flat.item_size = sizeof(float);
426 flat.dtype = SNEPPX_FLOAT32;
427 flat.strides = NULL;
428 SNEPPXTensor* ser_out = NULL;
429 SNEPPX_ser_forward(model->ser_model->layers[0], &flat, &ser_out);
430 if (ser_out)
431 {
432 SNEPPX_tensor_destroy(current);
433 current = ser_out;
434 }
435 }
436
437 if (model->arc_layer)
438 {
439 float metrics[4];
440 SNEPPXTensor* arc_out = NULL;
441 SNEPPX_arc_forward(model->arc_layer, current, &arc_out, metrics);
442 if (arc_out)
443 {
444 SNEPPX_tensor_destroy(current);
445 current = arc_out;
446 }
447 }
448
449 if (model->npe_vm && model->npe_program)
450 {
451 SNEPPXTensor* npe_out = NULL;
452 SNEPPX_npe_vm_run(model->npe_vm, current, &npe_out);
453 if (npe_out)
454 {
455 SNEPPX_tensor_destroy(current);
456 current = npe_out;
457 }
458 }
459
460 if (model->fm_controller)
461 {
462 SNEPPXTensor* fm_out = NULL;
463 int ret = SNEPPX_fm_forward(model->fm_controller, 0, current, &fm_out);
464 if (ret == 0 && fm_out)
465 {
466 SNEPPX_tensor_destroy(current);
467 current = fm_out;
468 }
469 }
470
471 *output = current;
472 return current ? 0 : 1;
473}

484{

541 idx += n;
