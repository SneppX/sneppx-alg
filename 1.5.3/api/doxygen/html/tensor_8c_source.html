35static int compute_offset_checked(const SNEPPXTensor* tensor, const size_t* indices, size_t* out_offset)
36{
37 if (!tensor || !out_offset || !tensor->data ||
38 (tensor->ndim > 0 && (!indices || !tensor->shape || !tensor->strides)))
39 {
40 return 0;
41 }
42
43 size_t offset = 0;
44 for (size_t i = 0; i < tensor->ndim; ++i)
45 {
46 if (indices[i] >= tensor->shape[i])
47 return 0;
48 if (tensor->strides[i] != 0 && indices[i] > (SIZE_MAX - offset) / tensor->strides[i])
49 return 0;
50 offset += indices[i] * tensor->strides[i];
51 }
52
53 if (tensor->item_size == 0)
54 return 0;
55 if (tensor->storage)
56 {
57 if (tensor->storage->num_bytes % tensor->item_size != 0)
58 return 0;
59 const size_t capacity = tensor->storage->num_bytes / tensor->item_size;
60 if (tensor->offset > capacity || offset >= capacity - tensor->offset)
61 return 0;
62 }
63 *out_offset = offset;
64 return 1;
65}

67static int compute_offset_flat_checked(const SNEPPXTensor* tensor, size_t flat_index, size_t* out_offset)
68{
69 if (!tensor || !out_offset || !tensor->data || flat_index >= tensor->size ||
70 (tensor->ndim > 0 && (!tensor->shape || !tensor->strides)))
71 return 0;
72
73 size_t remaining = flat_index;
74 size_t offset = 0;
75 for (size_t i = tensor->ndim; i > 0; --i)
76 {
77 const size_t dim = tensor->shape[i - 1];
78 if (dim == 0)
79 return 0;
80 const size_t coordinate = remaining % dim;
81 remaining /= dim;
82 const size_t stride = tensor->strides[i - 1];
83 if (stride != 0 && coordinate > (SIZE_MAX - offset) / stride)
84 return 0;
85 offset += coordinate * stride;
86 }
87
88 if (tensor->item_size == 0)
89 return 0;
90 if (tensor->storage)
91 {
92 if (tensor->storage->num_bytes % tensor->item_size != 0)
93 return 0;
94 const size_t capacity = tensor->storage->num_bytes / tensor->item_size;
95 if (tensor->offset > capacity || offset >= capacity - tensor->offset)
96 return 0;
97 }
98 *out_offset = offset;
99 return 1;

190SNEPPXTensor* SNEPPX_tensor_as_strided(const SNEPPXTensor* src, size_t offset, const size_t* shape, size_t ndim,
191 const size_t* strides)
192{
193 if (!src || !src->storage || src->item_size == 0 || (ndim > 0 && (!shape || !strides)))
194 return NULL;
195 if (src->storage->num_bytes % src->item_size != 0)
196 return NULL;
197
198 size_t view_size = 1;
199 size_t max_offset = offset;
200 int is_empty = 0;
201 for (size_t i = 0; i < ndim; ++i)
202 {
203 if (shape[i] == 0)
204 {
205 is_empty = 1;
206 view_size = 0;
207 }
208 else
209 {
210 if (view_size > SIZE_MAX / shape[i])
211 return NULL;
212 view_size *= shape[i];
213 const size_t last_index = shape[i] - 1;
214 if (strides[i] != 0 && last_index > SIZE_MAX / strides[i])
215 return NULL;
216 const size_t delta = last_index * strides[i];
217 if (max_offset > SIZE_MAX - delta)
218 return NULL;
219 max_offset += delta;
220 }
221 }
222
223 const size_t capacity = src->storage->num_bytes / src->item_size;
224 if (offset > capacity)
225 return NULL;
226 if (!is_empty && max_offset >= capacity)
227 return NULL;
228
229 SNEPPXTensor* t = tensor_view_alloc(src, ndim);
230 if (!t)
231 return NULL;
232 for (size_t i = 0; i < ndim; ++i)
233 {
234 t->shape[i] = shape[i];
235 t->strides[i] = strides[i];
236 }
237
238 SNEPPX_storage_retain(src->storage);
239 t->storage = src->storage;
240 t->offset = offset;
241 t->ndim = ndim;
242 t->size = view_size;
243 t->dtype = src->dtype;
244 t->item_size = src->item_size;
245 t->device = src->device;
246 t->device_id = src->device_id;
247 t->layout = src->layout;
248 t->owns_data = 0;
249 t->backend_handle = NULL;
250 t->data = src->storage->data ? (unsigned char*) src->storage->data + offset * t->item_size : NULL;
251 return t;
252}

324SNEPPXTensor* SNEPPX_tensor_create(const size_t* shape, size_t ndim, SNEPPXDtype dtype)
325{
326 if (ndim > 0 && !shape)
327 return NULL;
328
329 const size_t item_size = SNEPPX_tensor_dtype_size(dtype);
330 if (item_size == 0)
331 return NULL;
332
333 size_t total = 1;
334 for (size_t i = 0; i < ndim; ++i)
335 {
336 if (shape[i] != 0 && total > SIZE_MAX / shape[i])
337 return NULL;
338 total *= shape[i];
339 }
340
341 /* Validate row-major stride products independently, including zero-sized shapes. */
342 size_t stride = 1;
343 for (size_t i = ndim; i > 0; --i)
344 {
345 const size_t dim = shape[i - 1];
346 if (dim != 0 && stride > SIZE_MAX / dim)
347 return NULL;
348 stride *= dim;
349 }
350
351 if (total != 0 && item_size > SIZE_MAX / total)
352 return NULL;
353 const size_t num_bytes = total * item_size;
354 const size_t safe_ndim = ndim > 0 ? ndim : 1;
355 if (safe_ndim > SIZE_MAX / sizeof(size_t))
356 return NULL;
357 const size_t metadata_bytes = safe_ndim * sizeof(size_t);
358
359 SNEPPXTensor* tensor = (SNEPPXTensor*) aligned_alloc_wrapper(sizeof(SNEPPXTensor), 64);
360 if (!tensor)
361 return NULL;
362 memset(tensor, 0, sizeof(*tensor));
363
364 tensor->shape = (size_t*) aligned_alloc_wrapper(metadata_bytes, 64);
365 tensor->strides = (size_t*) aligned_alloc_wrapper(metadata_bytes, 64);
366 if (!tensor->shape || !tensor->strides)
367 {
368 SNEPPX_free(tensor->shape, metadata_bytes);
369 SNEPPX_free(tensor->strides, metadata_bytes);
370 SNEPPX_free(tensor, sizeof(SNEPPXTensor));
371 return NULL;
372 }
373
374 tensor->ndim = ndim;
375 tensor->size = total;
376 tensor->dtype = dtype;
377 tensor->item_size = item_size;
378 tensor->device = SNEPPX_DEVICE_CPU;
379 tensor->device_id = 0;
380 tensor->layout = SNEPPX_LAYOUT_ROW_MAJOR;
381 tensor->owns_data = 1;
382
383 stride = 1;
384 for (size_t i = ndim; i > 0; --i)
385 {
386 tensor->shape[i - 1] = shape[i - 1];
387 tensor->strides[i - 1] = stride;
388 stride *= shape[i - 1];
389 }
390
391 tensor->storage = SNEPPX_storage_create(num_bytes);
392 if (!tensor->storage)
393 {
394 SNEPPX_free(tensor->shape, metadata_bytes);
395 SNEPPX_free(tensor->strides, metadata_bytes);
396 SNEPPX_free(tensor, sizeof(SNEPPXTensor));
397 return NULL;
398 }
399 tensor->data = tensor->storage->data;
400 return tensor;
401}
