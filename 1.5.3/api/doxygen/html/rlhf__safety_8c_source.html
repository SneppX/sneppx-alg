179int SNEPPX_rlhf_score(const char* input, const char* output, rlhf_score_t* score)
180{
181 if (!input || !output || !score)
182 return -1;
183 memset(score, 0, sizeof(rlhf_score_t));
184 score->harmfulness = rlhf_detect_harm(output, strlen(output));
185 score->bias = rlhf_detect_bias(output, strlen(output));
186 score->factuality = rlhf_check_factual(output, strlen(output));
187 score->helpfulness = 1.0 - score->factuality * 0.3;
188 score->honesty = 1.0 - score->factuality * 0.2;
189 score->overall = (score->helpfulness * rlhf_config.helpfulness_weight) +
190 ((1.0 - score->harmfulness) * rlhf_config.harmlessness_weight) +
191 (score->honesty * rlhf_config.honesty_weight);
192 score->should_refuse =
193 (score->harmfulness > rlhf_config.harm_threshold || score->bias > rlhf_config.bias_threshold);
194 if (score->should_refuse && rlhf_config.enable_refusal)
195 total_refusals++;
196 return 0;
197}

207int SNEPPX_rlhf_check_policies(const char* text, rlhf_violation_t* violations, int max_violations)
208{
209 if (!text || !violations)
210 return -1;
211 int found = 0;
212 for (int i = 0; i < policy_count && found < max_violations; i++)
213 {
214 if (!policies[i].active)
215 continue;
216 double score = 0;
217 switch (policies[i].rule_type)
218 {
219 case RLHF_RULE_HARM:
220 score = rlhf_detect_harm(text, strlen(text));
221 break;
222 case RLHF_RULE_BIAS:
223 score = rlhf_detect_bias(text, strlen(text));
224 break;
225 case RLHF_RULE_FACTUAL:
226 score = rlhf_check_factual(text, strlen(text));
227 break;
228 case RLHF_RULE_CUSTOM:
229 score = rlhf_detect_harm(text, strlen(text)) * 0.5;
230 break;
231 }
232 if (score > policies[i].threshold)
233 {
234 strncpy(violations[found].policy_name, policies[i].name, 63);
235 violations[found].score = score;
236 violations[found].threshold = policies[i].threshold;
237 violations[found].severity = (score > 0.8) ? RLHF_SEV_CRITICAL
238 : (score > 0.6) ? RLHF_SEV_HIGH
239 : RLHF_SEV_MEDIUM;
240 policies[i].violations++;
241 found++;
242 }
243 }
244 return found;
245}
