781 for (size_t i = 0; i < len; i++)
782 {

798int SNEPPX_quic_stream_recv(SNEPPXQUICConn* qc, int stream_id, uint8_t* data, size_t* len)

805 if (qc->stream_buffers[idx] == NULL)

811 uint32_t avail = *wp - *rp;
812 if (avail == 0)
813 return -1;
814 size_t to_read = *len;
815 if (to_read > avail)
816 to_read = avail;
817 size_t orig_rp = *rp;
818 uint32_t data_off = 8;
819 for (size_t i = 0; i < to_read; i++)
820 {
821 data[i] = buf[data_off + (*rp % cap)];
822 (*rp)++;
823 }
824 if (*rp - orig_rp != to_read)
825 *rp = (uint32_t) (orig_rp + to_read);
826 *len = to_read;
827 return (int) to_read;
828}

943/* ---- mTLS ---- */
953int SNEPPX_mtls_authenticate(const uint8_t* cert_der, size_t cert_len, const uint8_t* key_der, size_t key_len)
954{
955 if (!cert_der || !key_der || cert_len < 10 || key_len < 4)
956 return 0;
957 if (cert_der[0] != 0x30)
958 return 0;
959 size_t off = 2;
960 size_t outer_len = 0;
961 if (cert_der[off] < 0x80)
962 {
963 outer_len = cert_der[off];
964 off++;
965 }
966 else
967 {
968 int lbytes = cert_der[off] & 0x7f;
969 off++;
970 for (int i = 0; i < lbytes && i < 4; i++)
971 outer_len = (outer_len << 8) | cert_der[off++];
972 }
973 if (off + outer_len > cert_len)
974 return 0;
975 if (off + 2 > cert_len)
976 return 0;
977 if (cert_der[off] != 0x30)
978 return 0;
979 off++;
980 size_t tbs_len = 0;
981 if (cert_der[off] < 0x80)
982 {
983 tbs_len = cert_der[off];
984 off++;
985 }
986 else
987 {
988 int lbytes = cert_der[off] & 0x7f;
989 off++;
990 for (int i = 0; i < lbytes; i++)
991 tbs_len = (tbs_len << 8) | cert_der[off++];
992 }
993 if (off + tbs_len > cert_len)
994 return 0;
995 size_t tbs_end = off + tbs_len;
996 int found_key = 0;
997 for (size_t i = off; i + key_len <= tbs_end; i++)
998 {
999 if (memcmp(cert_der + i, key_der, key_len) == 0)
1000 {
1001 found_key = 1;
1002 break;
1003 }
1004 }
1005 if (!found_key)
1006 return 0;
1007 size_t sig_start = tbs_end;
1008 if (sig_start + 2 > cert_len)
1009 return 0;
1010 if (cert_der[sig_start] != 0x30)
1011 return 0;
1012 size_t sig_algo_len = 0;
1013 if (cert_der[sig_start + 1] < 0x80)
1014 sig_algo_len = cert_der[sig_start + 1];
1015 sig_start += 2 + sig_algo_len;
1016 if (sig_start + 4 > cert_len)
1017 return 0;
1018 if (cert_der[sig_start] != 0x03 && cert_der[sig_start] != 0x04)
1019 return 0;
1020 return 1;
1021}

1091 }

1101 tmp[pos++] = 0x03;
1102 tmp[pos++] = 0x02;
1103 tmp[pos++] = 0x01;
1104 tmp[pos++] = 0x00;
1105 tmp[pos++] = 0x30;
1106 size_t rl_len_off = pos;
1107 pos++;
1108 tmp[pos++] = 0x30;
1109 size_t req_len_off = pos;
1110 pos++;
1111 tmp[pos++] = 0x30;

1113 pos++;
1114 tmp[pos++] = 0x30;
1115 tmp[pos++] = 0x09;
1116 tmp[pos++] = 0x06;
1117 tmp[pos++] = 0x05;
1118 tmp[pos++] = 0x2b;
1119 tmp[pos++] = 0x0e;

1121 tmp[pos++] = 0x02;
1122 tmp[pos++] = 0x1a;
1123 tmp[pos++] = 0x05;
1124 tmp[pos++] = 0x00;
1125 tmp[pos++] = 0x04;
1126 tmp[pos++] = 0x14;
1127 memcpy(tmp + pos, ih, 20);
1128 pos += 20;
1129 tmp[pos++] = 0x04;
1130 tmp[pos++] = 0x14;
1131 memcpy(tmp + pos, kh, 20);
1132 pos += 20;
1133 tmp[pos++] = 0x02;
1134 tmp[pos++] = 0x01;
1135 tmp[pos++] = 0x01;
1136 tmp[cid_len_off] = (uint8_t) (pos - cid_len_off - 1);
1137 tmp[req_len_off] = (uint8_t) (pos - req_len_off - 1);

1142 return -1;

1145 return 0;
