29int SNEPPX_act_forward(const float* x, float* output, SNEPPXActivationType act, size_t n)
30{
31 if (!x || !output || n == 0)
32 return -1;

34 {
35 case SNEPPX_ACT_RELU:
36 for (size_t i = 0; i < n; i++)
37 output[i] = x[i] > 0.0f ? x[i] : 0.0f;
38 return 0;
39 case SNEPPX_ACT_LEAKY_RELU:
40 return SNEPPX_act_leaky_relu(x, output, 0.01f, n);
41 case SNEPPX_ACT_PRELU:

82int SNEPPX_act_backward(const float* x, const float* grad_out, float* grad_in, SNEPPXActivationType act, size_t n)
83{
84 if (!x || !grad_out || !grad_in || n == 0)
85 return -1;
86 for (size_t i = 0; i < n; i++)
87 {
88 float xv = x[i], g = grad_out[i];
89 switch (act)
90 {
91 case SNEPPX_ACT_RELU:
92 grad_in[i] = xv > 0.0f ? g : 0.0f;
93 break;
94 case SNEPPX_ACT_LEAKY_RELU:
95 grad_in[i] = xv > 0.0f ? g : g * 0.01f;
96 break;
97 case SNEPPX_ACT_GELU_TANH:
98 {
99 float c = 0.7978845608028654f;
100 float x3 = xv * xv * xv;
101 float t = tanhf(c * (xv + 0.044715f * x3));
102 float sech2 = 1.0f - t * t;
103 float dg = 0.5f * (1.0f + t) + 0.5f * xv * c * (1.0f + 3.0f * 0.044715f * xv * xv) * sech2;
104 grad_in[i] = g * dg;
105 break;
106 }
107 case SNEPPX_ACT_SILU:
108 {
109 float s = 1.0f / (1.0f + expf(-xv));
110 grad_in[i] = g * s * (1.0f + xv * (1.0f - s));
111 break;
112 }
113 case SNEPPX_ACT_SIGMOID:
114 {
115 float s = 1.0f / (1.0f + expf(-xv));
116 grad_in[i] = g * s * (1.0f - s);
117 break;
118 }
119 case SNEPPX_ACT_TANH:
120 {
121 float t = tanhf(xv);
122 grad_in[i] = g * (1.0f - t * t);
123 break;
124 }
125 default:
126 grad_in[i] = g;
127 }
128 }
129 return 0;
130}

140int SNEPPX_act_softmax_stable(const float* x, float* output, size_t n)
141{
142 if (!x || !output || n == 0)
143 return -1;
144 float maxv = -FLT_MAX;
145 for (size_t i = 0; i < n; i++)
146 if (x[i] > maxv)
147 maxv = x[i];
148 float sum = 0.0f;
149 for (size_t i = 0; i < n; i++)
150 {
151 float e = expf(x[i] - maxv);
152 output[i] = e;
153 sum += e;
154 }
155 if (sum > 0.0f)
156 for (size_t i = 0; i < n; i++)
157 output[i] /= sum;
158 return 0;
159}
