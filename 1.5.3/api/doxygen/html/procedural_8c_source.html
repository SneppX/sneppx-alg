81static int find_matching_skill(SNEPPXProceduralCache* cache, const SNEPPXTensor* state)
82{
83 float* sd = (float*) state->data;
84 size_t slen = state->size;
85 float s_norm = 0.0f;
86 for (size_t i = 0; i < slen; i++)
87 s_norm += sd[i] * sd[i];
88 s_norm = sqrtf(s_norm + 1e-10f);
89 int best_idx = -1;
90 float best_sim = -1.0f;
91 for (size_t i = 0; i < cache->num_skills; i++)
92 {
93 if (!cache->skills[i].valid || !cache->skills[i].state_pattern)
94 continue;
95 float* pd = (float*) cache->skills[i].state_pattern->data;
96 size_t plen = cache->skills[i].state_pattern->size;
97 size_t min_dim = slen < plen ? slen : plen;
98 float dot = 0.0f, p_norm = 0.0f;
99 for (size_t j = 0; j < min_dim; j++)
100 dot += sd[j] * pd[j];
101 for (size_t j = 0; j < plen; j++)
102 p_norm += pd[j] * pd[j];
103 p_norm = sqrtf(p_norm + 1e-10f);
104 float sim = dot / (s_norm * p_norm + 1e-10f);
105 if (sim > best_sim)
106 {
107 best_sim = sim;
108 best_idx = (int) i;
109 }
110 }
111 if (best_sim > 0.85f)
112 return best_idx;
113 return -1;
114}

152int sneppx_procedural_learn(SNEPPXProceduralCache* cache, const SNEPPXTensor* state, const SNEPPXTensor* skill)
153{
154 if (!cache || !state || !skill)
155 return -1;
156 int idx = find_matching_skill(cache, state);
157 if (idx >= 0)
158 {
159 SNEPPXProceduralSkill* sk = &cache->skills[idx];
160 if (sk->skill_output)
161 SNEPPX_tensor_destroy(sk->skill_output);
162 sk->skill_output = SNEPPX_tensor_copy(skill);
163 if (!sk->skill_output)
164 return -1;
165 sk->frequency++;
166 sk->last_used = ++cache->access_counter;
167 return 0;
168 }
169 idx = evict_lru(cache);
170 if (idx < 0)
171 return -1;
172 SNEPPXProceduralSkill* sk = &cache->skills[idx];
173 sk->state_pattern = SNEPPX_tensor_copy(state);
174 if (!sk->state_pattern)
175 return -1;
176 sk->skill_output = SNEPPX_tensor_copy(skill);
177 if (!sk->skill_output)
178 {
179 SNEPPX_tensor_destroy(sk->state_pattern);
180 sk->state_pattern = NULL;
181 return -1;
182 }
183 sk->frequency = 1;
184 sk->last_used = ++cache->access_counter;
185 sk->sequence_length = 1;
186 sk->valid = 1;
187 return 0;
