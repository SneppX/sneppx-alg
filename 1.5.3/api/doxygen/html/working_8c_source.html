120int sneppx_working_attend(SNEPPXWorkingMemory* wm, const SNEPPXTensor* query, SNEPPXTensor** output)
121{
122 if (!wm || !query || !output)
123 return -1;
124 float* qd = (float*) query->data;
125 size_t qlen = query->size;
126 float query_norm = 0.0f;
127 for (size_t i = 0; i < qlen; i++)
128 query_norm += qd[i] * qd[i];
129 query_norm = sqrtf(query_norm + 1e-10f);
130 float total_weight = 0.0f;
131 for (size_t i = 0; i < wm->num_slots; i++)
132 {
133 if (wm->slots[i].is_free || !wm->slots[i].content)
134 {
135 wm->slots[i].attention_weight = 0.0f;
136 continue;
137 }
138 float* sd = (float*) wm->slots[i].content->data;
139 size_t slen = wm->slots[i].content->size;
140 size_t min_dim = qlen < slen ? qlen : slen;
141 float dot = 0.0f, s_norm = 0.0f;
142 for (size_t j = 0; j < min_dim; j++)
143 dot += qd[j] * sd[j];
144 for (size_t j = 0; j < slen; j++)
145 s_norm += sd[j] * sd[j];
146 s_norm = sqrtf(s_norm + 1e-10f);
147 float sim = dot / (query_norm * s_norm + 1e-10f);
148 float recency = 1.0f / (1.0f + (float) (wm->clock - wm->slots[i].last_modified));
149 wm->slots[i].attention_weight = sim * 0.7f + recency * 0.3f;
150 if (wm->slots[i].attention_weight < 0.0f)
151 wm->slots[i].attention_weight = 0.0f;
152 total_weight += wm->slots[i].attention_weight;
153 }
154 size_t shape[2] = {1, wm->slot_dim};
155 SNEPPXTensor* result = SNEPPX_tensor_zeros(shape, 2, SNEPPX_FLOAT32);
156 if (!result)
157 return -1;
158 float* rd = (float*) result->data;
159 for (size_t i = 0; i < wm->num_slots; i++)
160 {
161 if (wm->slots[i].is_free || !wm->slots[i].content || wm->slots[i].attention_weight <= 0.0f)
162 continue;
163 float w = wm->slots[i].attention_weight / (total_weight + 1e-10f);
164 float* sd = (float*) wm->slots[i].content->data;
165 size_t copy_dim = wm->slot_dim < wm->slots[i].content->size ? wm->slot_dim : wm->slots[i].content->size;
166 for (size_t j = 0; j < copy_dim; j++)
167 rd[j] += w * sd[j];
168 wm->slots[i].last_modified = ++wm->clock;
169 }
170 *output = result;
171 return 0;
172}
