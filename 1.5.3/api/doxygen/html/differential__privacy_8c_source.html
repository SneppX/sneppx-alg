34/* M-10: DP noise must come from the CSPRNG, never rand(). Returning an error
35 * lets each caller fail closed instead of emitting predictable noise. */
36static int dp_random_unit(double* out)
37{
38 uint64_t bits = 0;
39 if (!out || SNEPPX_random_bytes((uint8_t*) &bits, sizeof(bits)) != 0)
40 return -1;
41 /* 53 random mantissa bits + 0.5 gives an open interval (0,1), so log()
42 * never sees zero. */
43 *out = ((double) (bits >> 11) + 0.5) / 9007199254740992.0;
44 return 0;
45}

55double SNEPPX_dp_laplace_mech(double value, double sensitivity, double epsilon)
56{
57 double u;
58 if (dp_random_unit(&u) != 0)
59 return NAN;
60 double scale = sensitivity / epsilon;
61 u -= 0.5;
62 return value - scale * (u > 0 ? log(1 - 2 * u) : -log(1 + 2 * u));
63}

74double SNEPPX_dp_gaussian_mech(double value, double sensitivity, double epsilon, double delta)
75{
76 double u1, u2;
77 if (dp_random_unit(&u1) != 0 || dp_random_unit(&u2) != 0)
78 return NAN;
79 double sigma = sensitivity * sqrt(2 * log(1.25 / delta)) / epsilon;
80 double z = sqrt(-2 * log(u1)) * cos(2 * 3.141592653589793 * u2);
81 return value + sigma * z;
82}

102int SNEPPX_dp_compose(dp_composition_t* comp)
103{
104 if (comp_count >= DP_MAX_COMPOSITIONS)
105 return -1;
106 double eps_sum = 0.0, del_sum = 0.0;
107 for (int i = 0; i < comp->num_mechanisms; i++)
108 {
109 eps_sum += mechanisms[comp->indices[i]].epsilon;
110 del_sum += mechanisms[comp->indices[i]].delta;
111 }
112 comp->total_epsilon = eps_sum;
113 comp->total_delta = del_sum;
114 comp->id = comp_count;
115 compositions[comp_count++] = *comp;
116 total_epsilon_consumed += eps_sum;
117 total_delta_consumed += del_sum;
118 return comp->id;
119}

220int SNEPPX_dp_dpsgd_step(double* params, const double* gradients, int n, double lr, double epsilon, double delta,
221 double clip_norm)
222{
223 double norm = 0;
224 for (int i = 0; i < n; i++)
225 norm += gradients[i] * gradients[i];
226 norm = sqrt(norm);
227 double scale = (norm > clip_norm) ? clip_norm / norm : 1.0;
228 double sigma = clip_norm * sqrt(2 * log(1.25 / delta)) / epsilon;
229 for (int i = 0; i < n; i++)
230 {
231 double u1, u2;
232 if (dp_random_unit(&u1) != 0 || dp_random_unit(&u2) != 0)
233 return -1;
234 double noise = sqrt(-2 * log(u1)) * cos(2 * 3.141592653589793 * u2) * sigma;
235 params[i] -= lr * (gradients[i] * scale + noise);
236 }
237 return 0;
238}
