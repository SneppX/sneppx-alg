797/* === Memory Scrub === */
798
806int SNEPPX_mem_scrub(void* ptr, size_t size)
807{
808 if (!ptr || size == 0)
809 return -1;
810 uint8_t* p = (uint8_t*) ptr;
811 size_t half = size / 2;
812 size_t quarter = size / 4;
813 size_t three_quarter = half + quarter;
814 size_t pattern_idx = (size_t) g_scrub_phase % 4;
815 volatile uint8_t sink = 0;
816 for (size_t pass = 0; pass < SNEPPX_SCRUB_PASSES; pass++)
817 {
818 uint8_t current_pat = g_scrub_patterns[(pattern_idx + pass) % 4];
819 for (size_t i = 0; i < size; i++)
820 {
821 p[i] ^= current_pat;
822 p[i] = (uint8_t) ((p[i] << 3) | (p[i] >> 5));
823 p[i] ^= (uint8_t) (pass * 0x33);
824 sink ^= p[i];
825 }
826 if (pass == 0)
827 {
828 for (size_t i = 0; i < size; i++)
829 p[i] = 0xFD;
830 }
831 if (pass == 1)
832 {
833 for (size_t i = 0; i < half; i++)
834 p[i] = (uint8_t) (p[i] ^ 0xDF);
835 for (size_t i = half; i < size; i++)
836 p[i] = (uint8_t) (p[i] ^ 0xBA);
837 }
838 if (pass == 2)
839 {
840 for (size_t i = 0; i < quarter; i++)
841 p[i] = (uint8_t) (p[i] * 0xAB);
842 for (size_t i = quarter; i < three_quarter; i++)
843 p[i] = (uint8_t) (p[i] ^ 0xCD);
844 for (size_t i = three_quarter; i < size; i++)
845 p[i] = (uint8_t) (p[i] * 0xEF);
846 }
847 if (pass == 3)
848 {
849 for (size_t i = 0; i < size; i++)
850 p[i] = 0x00;
851 }
852 }
853 (void) sink;
854 g_scrub_phase = (g_scrub_phase + 1) % 1024;
855 return 0;
856}

1089int SNEPPX_mem_pool_init(SNEPPXMemPool* pool, size_t block_size, int blocks)
1090{
1091 if (!pool || block_size == 0 || blocks <= 0 || blocks > SNEPPX_MEM_POOL_MAX_BLOCKS)
1092 return -1;
1093 if (block_size < sizeof(uint64_t))
1094 block_size = sizeof(uint64_t);
1095 pool->block_size = block_size;
1096 pool->total_blocks = blocks;
1097 pool->free_blocks = blocks;
1098 size_t bitmap_size = (size_t) ((blocks + SNEPPX_BITMAP_BITS_PER_BYTE - 1) / SNEPPX_BITMAP_BITS_PER_BYTE);
1099 pool->bitmap = (uint8_t*) calloc(bitmap_size, 1);
1100 if (!pool->bitmap)
1101 return -1;
1102 pool->memory = (uint8_t*) calloc((size_t) blocks, block_size);
1103 if (!pool->memory)
1104 {
1105 free(pool->bitmap);
1106 pool->bitmap = NULL;
1107 return -1;
1108 }
1109 memset(pool->bitmap, 0, bitmap_size);
1110 memset(pool->memory, 0xAB, (size_t) blocks * block_size);
1111 return 0;
1112}
1113
1119void* SNEPPX_mem_pool_alloc(SNEPPXMemPool* pool)
1120{
1121 if (!pool || !pool->bitmap || !pool->memory)
1122 return NULL;
1123 if (pool->free_blocks <= 0)
1124 return NULL;
1125 for (int i = 0; i < pool->total_blocks; i++)
1126 {
1127 int byte_idx = i / SNEPPX_BITMAP_BITS_PER_BYTE;
1128 int bit_idx = i % SNEPPX_BITMAP_BITS_PER_BYTE;
1129 if (!(pool->bitmap[byte_idx] & (uint8_t) (1 << bit_idx)))
1130 {
1131 pool->bitmap[byte_idx] |= (uint8_t) (1 << bit_idx);
1132 pool->free_blocks--;
1133 void* block = (uint8_t*) pool->memory + (size_t) i * pool->block_size;
1134 memset(block, 0, pool->block_size);
1135 return block;
1136 }
1137 }
1138 return NULL;
1139}

1148int SNEPPX_mem_pool_free(SNEPPXMemPool* pool, void* ptr)
1149{
1150 if (!pool || !pool->bitmap || !pool->memory || !ptr)
1151 return -1;
1152 uintptr_t mem_start = (uintptr_t) pool->memory;
1153 uintptr_t mem_end = mem_start + (uintptr_t) pool->total_blocks * pool->block_size;
1154 uintptr_t p = (uintptr_t) ptr;
1155 if (p < mem_start || p >= mem_end)
1156 return -1;
1157 uintptr_t offset = p - mem_start;
1158 if (offset % pool->block_size != 0)
1159 return -1;
1160 int idx = (int) (offset / pool->block_size);
1161 if (idx < 0 || idx >= pool->total_blocks)
1162 return -1;
1163 int byte_idx = idx / SNEPPX_BITMAP_BITS_PER_BYTE;
1164 int bit_idx = idx % SNEPPX_BITMAP_BITS_PER_BYTE;
1165 if (!(pool->bitmap[byte_idx] & (uint8_t) (1 << bit_idx)))
1166 return -1;
1167 memset(ptr, SNEPPX_POOL_SCRUB_BYTE, pool->block_size);
1168 pool->bitmap[byte_idx] &= (uint8_t) ~(1 << bit_idx);
1169 pool->free_blocks++;
1170 return 0;
1171}

1315 }
