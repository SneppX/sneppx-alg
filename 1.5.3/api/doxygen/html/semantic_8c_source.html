66static int find_or_create_concept(SNEPPXSemanticGraph* graph, const char* key)
67{
68 int idx = find_concept_idx(graph, key);
69 if (idx >= 0)
70 return idx;
71 if (graph->num_concepts >= graph->max_concepts)
72 {
73 uint64_t oldest = graph->access_counter;
74 int lru_idx = -1;
75 for (size_t i = 0; i < graph->num_concepts; i++)
76 {
77 if (graph->concepts[i].valid && graph->concepts[i].last_access < oldest)
78 {
79 oldest = graph->concepts[i].last_access;
80 lru_idx = (int) i;
81 }
82 }
83 if (lru_idx >= 0)
84 {
85 if (graph->concepts[lru_idx].embedding)
86 {
87 SNEPPX_tensor_destroy(graph->concepts[lru_idx].embedding);
88 graph->concepts[lru_idx].embedding = NULL;
89 }
90 for (size_t e = 0; e < graph->num_edges; e++)
91 {
92 if (graph->edges[e].valid &&
93 (graph->edges[e].from_idx == (size_t) lru_idx || graph->edges[e].to_idx == (size_t) lru_idx))
94 {
95 graph->edges[e].valid = 0;
96 }
97 }
98 memset(graph->concepts[lru_idx].key, 0, SNEPPX_MAX_CONCEPT_KEY_LEN);
99 graph->concepts[lru_idx].valid = 0;
100 idx = lru_idx;
101 }
102 else
103 {
104 return -1;
105 }
106 }
107 else
108 {
109 idx = (int) graph->num_concepts;
110 graph->num_concepts++;
111 }
112 return idx;
113}

158{

164 SNEPPXSemanticConcept* conc = &graph->concepts[idx];

172 conc->ref_count = 0;

178int sneppx_semantic_retrieve(SNEPPXSemanticGraph* graph, const SNEPPXTensor* query, size_t top_k, SNEPPXTensor** output,
179 char*** out_keys)
180{
181 if (!graph || !query || !output)
182 return -1;
183 if (graph->num_concepts == 0)
184 {
185 *output = NULL;
186 return 0;
187 }
188 size_t n = graph->num_concepts;
189 float* scores = SNEPPX_malloc(n * sizeof(float), 64);
190 size_t* indices = SNEPPX_malloc(n * sizeof(size_t), 64);
191 if (!scores || !indices)
192 {
193 if (scores)
194 SNEPPX_free(scores, n * sizeof(float));
195 if (indices)
196 SNEPPX_free(indices, n * sizeof(size_t));
197 return -1;
198 }
199 size_t valid_count = 0;
200 float query_norm = 0.0f;
201 float* qd = (float*) query->data;
202 size_t qlen = query->size;
203 for (size_t i = 0; i < qlen; i++)
204 query_norm += qd[i] * qd[i];
205 query_norm = sqrtf(query_norm + 1e-10f);
206 for (size_t i = 0; i < graph->num_concepts; i++)
207 {
208 if (!graph->concepts[i].valid || !graph->concepts[i].embedding)
209 continue;
210 float* ed = (float*) graph->concepts[i].embedding->data;
211 size_t elen = graph->concepts[i].embedding->size;
212 size_t min_dim = qlen < elen ? qlen : elen;
213 float dot = 0.0f, norm = 0.0f;
214 for (size_t j = 0; j < min_dim; j++)
215 dot += qd[j] * ed[j];
216 for (size_t j = 0; j < elen; j++)
217 norm += ed[j] * ed[j];
218 norm = sqrtf(norm + 1e-10f);
219 scores[valid_count] = dot / (query_norm * norm + 1e-10f);
220 indices[valid_count] = i;
221 graph->concepts[i].last_access = ++graph->access_counter;
222 graph->concepts[i].ref_count++;
223 valid_count++;
224 }
225 if (valid_count == 0)
226 {
227 *output = NULL;
228 SNEPPX_free(scores, n * sizeof(float));
229 SNEPPX_free(indices, n * sizeof(size_t));
230 return 0;
231 }
232 for (size_t i = 0; i < valid_count - 1; i++)
233 {
234 for (size_t j = 0; j < valid_count - 1 - i; j++)
235 {
236 if (scores[j] < scores[j + 1])
237 {
238 float ts = scores[j];
239 scores[j] = scores[j + 1];
240 scores[j + 1] = ts;
241 size_t ti = indices[j];
242 indices[j] = indices[j + 1];
243 indices[j + 1] = ti;
244 }
245 }
246 }
247 size_t actual_k = top_k < valid_count ? top_k : valid_count;
248 size_t shape[2] = {actual_k, graph->embedding_dim};
249 SNEPPXTensor* result = SNEPPX_tensor_zeros(shape, 2, SNEPPX_FLOAT32);
250 if (!result)
251 {
252 SNEPPX_free(scores, n * sizeof(float));
253 SNEPPX_free(indices, n * sizeof(size_t));
254 return -1;
255 }
256 float* rd = (float*) result->data;
257 if (out_keys)
258 {
259 *out_keys = SNEPPX_malloc(actual_k * sizeof(char*), 64);
260 if (!*out_keys)
261 {
262 SNEPPX_tensor_destroy(result);
263 SNEPPX_free(scores, n * sizeof(float));
264 SNEPPX_free(indices, n * sizeof(size_t));
265 return -1;
266 }
267 }
268 for (size_t i = 0; i < actual_k; i++)
269 {
270 SNEPPXSemanticConcept* c = &graph->concepts[indices[i]];
271 if (c->embedding)
272 {
273 memcpy(rd + i * graph->embedding_dim, c->embedding->data,
274 (graph->embedding_dim < c->embedding->size ? graph->embedding_dim : c->embedding->size) *
275 sizeof(float));
276 }
277 if (out_keys)
278 {
279 (*out_keys)[i] = SNEPPX_malloc(SNEPPX_MAX_CONCEPT_KEY_LEN, 64);
280 if ((*out_keys)[i])
281 {
282 strncpy((*out_keys)[i], c->key, SNEPPX_MAX_CONCEPT_KEY_LEN - 1);
283 (*out_keys)[i][SNEPPX_MAX_CONCEPT_KEY_LEN - 1] = '\0';
284 }
285 }
286 }
287 *output = result;
288 SNEPPX_free(scores, n * sizeof(float));
289 SNEPPX_free(indices, n * sizeof(size_t));
290 return (int) actual_k;
291}

294{
