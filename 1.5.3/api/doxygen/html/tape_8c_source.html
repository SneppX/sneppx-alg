68void SNEPPX_tape_record(SNEPPXTape* tape, SNEPPXVariable* var)
69{
70 if (!tape || !var)
71 return;
72 /* The tape OWNS every recorded variable (destroyed in tape_destroy).
73 * Recording the same variable twice would register two destroy entries
74 * for one pointer -> double free at teardown. Skip duplicates. */
75 for (size_t i = 0; i < tape->num_vars; i++)
76 {
77 if (tape->vars[i] == var)
78 return;
79 }
80 if (tape->num_vars >= tape->capacity)
81 {
82 tape->capacity *= 2;
83 SNEPPXVariable** new_vars = (SNEPPXVariable**) SNEPPX_malloc(tape->capacity * sizeof(SNEPPXVariable*), 64);
84 if (!new_vars)
85 return;
86 memcpy(new_vars, tape->vars, tape->num_vars * sizeof(SNEPPXVariable*));
87 SNEPPX_free(tape->vars, tape->capacity / 2 * sizeof(SNEPPXVariable*));
88 tape->vars = new_vars;
89 }
90 tape->vars[tape->num_vars++] = var;
91 if (tape->checkpointing && var->backward_ctx && var->recompute_ctx)
92 {
93 var->checkpointed = 1;
94 var->free_ctx(var->backward_ctx);
95 var->backward_ctx = NULL;
96 }
97}

117static void tape_topological_sort(SNEPPXTape* tape, SNEPPXVariable** sorted, size_t* num_sorted)
118{
119 if (!tape || !sorted)
120 return;
121 size_t n = tape->num_vars;
122 if (n == 0)
123 return;
124 int* visited = (int*) SNEPPX_malloc(n * sizeof(int), 64);
125 if (!visited)
126 return;
127 memset(visited, 0, n * sizeof(int));
128 *num_sorted = 0;
129
130 /* DFS-based topological sort: for each node, visit all parents first */
131 for (size_t i = 0; i < n; i++)
132 {
133 if (visited[i])
134 continue;
135 /* Simple stack-based DFS post-order */
136 size_t stack_cap = 64;
137 size_t* stack = (size_t*) SNEPPX_malloc(stack_cap * sizeof(size_t), 64);
138 int* on_stack = (int*) SNEPPX_malloc(n * sizeof(int), 64);
139 if (!stack || !on_stack)
140 {
141 SNEPPX_free(stack, stack_cap * sizeof(size_t));
142 SNEPPX_free(on_stack, n * sizeof(int));
143 break;
144 }
145 memset(on_stack, 0, n * sizeof(int));
146 size_t sp = 0;
147 stack[sp++] = i;
148 on_stack[i] = 1;
149
150 size_t loop_guard = 0;
151 while (sp > 0)
152 {
153 if (++loop_guard > 1000000)
154 {
155 fprintf(stderr, "ERROR: tape_topological_sort infinite loop at i=%zu sp=%zu n=%zu\n", i, sp, n);
156 break;
157 }
158 size_t idx = stack[sp - 1];
159 SNEPPXVariable* var = tape->vars[idx];
160 int all_parents_visited = 1;
161 if (var && var->parents)
162 {
163 for (size_t p = 0; p < var->num_parents; p++)
164 {
165 SNEPPXVariable* parent = var->parents[p];
166 if (!parent)
167 continue;
168 /* Find parent index in tape vars */
169 int found = 0;
170 for (size_t pi = 0; pi < n; pi++)
171 {
172 if (tape->vars[pi] == parent)
173 {
174 found = 1;
175 if (!visited[pi])
176 {
177 if (!on_stack[pi])
178 {
179 /* Grow stack if needed */
180 if (sp >= stack_cap)
181 {
182 stack_cap *= 2;
183 size_t* ns = (size_t*) SNEPPX_malloc(stack_cap * sizeof(size_t), 64);
184 if (!ns)
185 break;
186 memcpy(ns, stack, sp * sizeof(size_t));
187 SNEPPX_free(stack, stack_cap / 2 * sizeof(size_t));
188 stack = ns;
189 }
190 stack[sp++] = pi;
191 on_stack[pi] = 1;
192 all_parents_visited = 0;
193 }
194 }
195 break;
196 }
197 }
198 /* If parent not found on tape, treat as visited (no backward needed) */
199 if (!found)
200 continue;
201 if (!all_parents_visited)
202 break;
203 }
204 }
205 if (all_parents_visited)
206 {
207 visited[idx] = 1;
208 sorted[(*num_sorted)++] = var;
209 on_stack[idx] = 0;
210 sp--;
211 }
212 }
213 SNEPPX_free(on_stack, n * sizeof(int));
214 SNEPPX_free(stack, stack_cap * sizeof(size_t));
215 }
216 SNEPPX_free(visited, n * sizeof(int));
217}

224void SNEPPX_tape_backward(SNEPPXTape* tape, SNEPPXVariable* loss)
225{
226 if (!tape || !loss)
227 return;
228
229 if (loss->grad)
230 SNEPPX_tensor_destroy(loss->grad);
231 loss->grad = SNEPPX_tensor_ones(loss->data->shape, loss->data->ndim, SNEPPX_FLOAT32);
232
233 SNEPPXVariable** sorted = NULL;
234 size_t n = tape->num_vars;
235 size_t num_sorted = 0;
236 if (n > 0)
237 {
238 sorted = (SNEPPXVariable**) SNEPPX_malloc(n * sizeof(SNEPPXVariable*), 64);
239 if (sorted)
240 {
241 tape_topological_sort(tape, sorted, &num_sorted);
242 }
243 }
244
245 for (size_t i = num_sorted; i > 0; i--)
246 {
247 SNEPPXVariable* var = sorted ? sorted[i - 1] : tape->vars[i - 1];
248 if (!var || !var->backward_fn || !var->grad)
249 continue;
250 if (!var->backward_ctx && var->checkpointed && var->recompute_ctx)
251 {
252 var->backward_ctx = var->recompute_ctx(var, var->params, var->param_count);
253 }
254 if (!var->backward_ctx)
255 continue;
256 var->backward_fn(var->backward_ctx, var->grad);
257 if (var->backward_ctx && var->free_ctx)
258 {
259 var->free_ctx(var->backward_ctx);
260 var->backward_ctx = NULL;
261 }
262 for (size_t p = 0; p < var->num_parents; p++)
263 {
264 SNEPPXVariable* parent = var->parents[p];
265 if (!parent)
266 continue;
267 parent->ref_count--;
268 }
269 }
270
271 if (sorted)
272 SNEPPX_free(sorted, n * sizeof(SNEPPXVariable*));
273}

297float SNEPPX_tape_global_norm(SNEPPXTape* tape)
298{
299 if (!tape)
300 return 0.0f;
301 float sum_sq = 0.0f;
302 for (size_t i = 0; i < tape->num_vars; i++)
303 {
304 SNEPPXVariable* var = tape->vars[i];
305 if (!var || !var->grad)
306 continue;
307 float* gd = (float*) var->grad->data;
308 size_t sz = var->grad->size;
309 for (size_t j = 0; j < sz; j++)
310 sum_sq += gd[j] * gd[j];
311 }
312 return sqrtf(sum_sq);
313}

347void SNEPPX_tape_clip_grad_norm(SNEPPXTape* tape, float max_norm)
348{
349 if (!tape || max_norm <= 0.0f)
350 return;
351 float total_norm = SNEPPX_tape_global_norm(tape);
352 if (total_norm <= max_norm)
353 return;
354 float scale = max_norm / (total_norm + 1e-7f);
355 for (size_t i = 0; i < tape->num_vars; i++)
356 {
357 SNEPPXVariable* var = tape->vars[i];
358 if (!var || !var->grad)
359 continue;
360 float* gd = (float*) var->grad->data;
361 for (size_t j = 0; j < var->grad->size; j++)
362 gd[j] *= scale;
363 }
