53int SNEPPX_gptq_compute_hessian(float* hessian, const float* activation, size_t n, size_t dim)
54{
55 if (!hessian || !activation || n == 0 || dim == 0)
56 return -1;
57 memset(hessian, 0, dim * dim * sizeof(float));
58 for (size_t i = 0; i < n; i++)
59 {
60 const float* row = activation + i * dim;
61 for (size_t r = 0; r < dim; r++)
62 {
63 for (size_t c = r; c < dim; c++)
64 {
65 hessian[r * dim + c] += row[r] * row[c];
66 }
67 }
68 }
69 for (size_t r = 0; r < dim; r++)
70 {
71 for (size_t c = 0; c < r; c++)
72 {
73 hessian[r * dim + c] = hessian[c * dim + r];
74 }
75 }
76 float reg = 1e-5f * hessian[0];
77 if (reg < 1e-8f)
78 reg = 1e-5f;
79 for (size_t i = 0; i < dim; i++)

81 hessian[i * dim + i] += reg;
82 }

86static int _cholesky_inv(float* h, size_t dim)
87{
88 float* L = (float*) calloc(dim * dim, sizeof(float));
89 if (!L)
90 return -1;
91 for (size_t j = 0; j < dim; j++)
92 {
93 float s = 0.0f;
94 for (size_t k = 0; k < j; k++)
95 {
96 s += L[j * dim + k] * L[j * dim + k];
97 }
98 float d = h[j * dim + j] - s;
99 if (d < 1e-10f)
100 {
101 free(L);
102 return -1;
103 }
104 L[j * dim + j] = sqrtf(d);
105 for (size_t i = j + 1; i < dim; i++)
106 {
107 s = 0.0f;
108 for (size_t k = 0; k < j; k++)
109 {
110 s += L[i * dim + k] * L[j * dim + k];
111 }
112 L[i * dim + j] = (h[i * dim + j] - s) / L[j * dim + j];
113 }
114 }
115 for (size_t i = 0; i < dim; i++)
116 {
117 for (size_t j = 0; j < dim; j++)
118 {
119 if (j > i)
120 {
121 h[i * dim + j] = 0.0f;
122 }
123 else if (j == i)
124 {
125 h[i * dim + j] = 1.0f / L[i * dim + i];
126 }
127 else
128 {
129 float s = 0.0f;
130 for (size_t k = j; k < dim; k++)
131 {
132 s += L[i * dim + k] * h[k * dim + j];
133 }
134 h[i * dim + j] = -s / L[i * dim + i];
135 }
136 }

156int SNEPPX_gptq_quantize_block(float* w, float* h_inv, float* qw, float* scale, int32_t* zero, size_t block_size,
157 int bits, int sym)
158{
159 if (!w || !h_inv || !qw || !scale)
160 return -1;
161 int qmax = (1 << (bits - 1)) - 1;
162 int qmin = -qmax;
163 if (!sym)
164 {
165 qmax = (1 << bits) - 1;
166 qmin = 0;
167 }
168 size_t cols = block_size;
169 float max_abs = 0.0f;
170 for (size_t j = 0; j < cols; j++)
171 {
172 float a = fabsf(w[j]);
173 if (a > max_abs)
174 max_abs = a;
175 }
176 if (max_abs < FLT_MIN)
177 {
178 *scale = 1.0f;
179 if (zero)
180 *zero = 0;
181 memset(qw, 0, cols * sizeof(float));
182 return 0;
183 }
184 *scale = max_abs / (float) qmax;
185 if (zero)
186 *zero = 0;
187 float inv_scale = 1.0f / (*scale);
188 for (size_t j = 0; j < cols; j++)
189 {
190 int qi = (int) roundf(w[j] * inv_scale);
191 qi = SNEPPX_CLAMP(qi, qmin, qmax);
192 float q = (float) qi * (*scale);
193 float err = q - w[j];
194 qw[j] = q;
195 float h_col = h_inv[j * cols + j];
196 if (fabsf(h_col) > 1e-10f)
197 {
198 float coef = -err / h_col;
199 for (size_t k = j + 1; k < cols; k++)
200 {
201 w[k] += coef * h_inv[k * cols + j];
202 }
203 }
204 }
205 return 0;
206}
