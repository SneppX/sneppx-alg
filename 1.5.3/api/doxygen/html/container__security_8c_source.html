84int SNEPPX_container_verify_image(const uint8_t* manifest, size_t manifest_len, const uint8_t* signature,
85 size_t sig_len, const uint8_t* pubkey)
86{
87 if (!manifest || !signature || !pubkey)
88 return -1;
89 if (sig_len != SNEPPX_ED25519_SIGNATURE_LEN)
90 return 1;
91
92 SNEPPXEd25519Signature sig;
93 memcpy(sig.data, signature, SNEPPX_ED25519_SIGNATURE_LEN);
94
95 uint8_t digest[32];
96 SNEPPX_sha256(digest, manifest, manifest_len);
97
98 int verified = SNEPPX_ed25519_verify(pubkey, digest, sizeof(digest), &sig);
99 if (verified < 0)
100 return -1;
101 return verified == 1 ? 0 : 1;
102}

118int SNEPPX_container_parse_manifest(const uint8_t* manifest, size_t len, container_manifest_t* out)
119{
120 if (!manifest || !out)
121 return -1;
122 memset(out, 0, sizeof(container_manifest_t));
123 const char* p = (const char*) manifest;
124 const char* end = p + len;
125 int found_config = 0, found_layers = 0;
126 while (p < end && *p)
127 {
128 if (manifest_starts_with(p, end, "\"config\":"))
129 found_config = 1;
130 else if (manifest_starts_with(p, end, "\"layers\":"))
131 found_layers = 1;
132 else if (manifest_starts_with(p, end, "\"mediaType\":\"") && found_config)
133 {
134 p += 13;
135 int ml = 0;
136 while (p < end && *p != '"' && ml < 63)
137 out->media_type[ml++] = *p++;
138 out->media_type[ml] = 0;
139 }
140 p++;
141 }
142 out->schema_version = 2;
143 out->num_layers = found_layers ? 3 : 1;
144
145 uint8_t digest[32];
146 SNEPPX_sha256(digest, manifest, len);
147 snprintf(out->digest, sizeof(out->digest), "%s", "sha256:");
148 for (size_t i = 0; i < sizeof(digest); i++)
149 snprintf(out->digest + 7 + 2 * i, sizeof(out->digest) - 7 - 2 * i, "%02x", digest[i]);
150 out->size = len;
151 return 0;
152}

162int SNEPPX_container_verify_layer(const uint8_t* layer_data, size_t layer_len, const char* expected_digest)
163{
164 if (!layer_data || !expected_digest)
165 return -1;
166 uint8_t hash[32];
167 char hex[65];
168 SNEPPX_sha256(hash, layer_data, layer_len);
169 for (int i = 0; i < 32; i++)
170 snprintf(hex + 2 * i, sizeof(hex) - 2 * i, "%02x", hash[i]);
171 hex[64] = 0;

173 if (!dig)
174 return 1;

213int SNEPPX_sbom_add_component(sbom_doc_t* doc, const char* name, const char* version, const char* type,
214 const char* supplier)
215{
216 if (!doc || !name || doc->num_components >= SBOM_MAX_COMPONENTS)
217 return -1;
218
219 const char* safe_version = version ? version : "0.0.0";
220 const char* safe_type = type ? type : "library";
221 uint8_t hash_input[256];
222 int hlen = snprintf((char*) hash_input, sizeof(hash_input), "%s:%s:%s", name, safe_version, safe_type);
223 if (hlen < 0 || (size_t) hlen >= sizeof(hash_input))
224 return -1;
225
226 sbom_component_t* comp = &doc->components[doc->num_components++];
227 strncpy(comp->name, name, 127);
228 strncpy(comp->version, safe_version, 63);
229 strncpy(comp->type, safe_type, 31);
230 strncpy(comp->supplier, supplier ? supplier : "unknown", 127);
231 SNEPPX_sha256(comp->hash, hash_input, (size_t) hlen);
232 comp->size = 0;
233 comp->has_checksum = 1;
234 return 0;
235}

267int SNEPPX_sbom_export_json(sbom_doc_t* doc, char* out, size_t out_len)
268{
269 if (!doc || !out || out_len == 0)
270 return -1;
271
272 char* cursor = out;
273 size_t remaining = out_len;
274
275/* Measure before writing so a required length can never wrap `remaining`. */
276#define SBOM_APPEND(...) \
277 do \
278 { \
279 int need_ = snprintf(NULL, 0, __VA_ARGS__); \
280 if (need_ < 0 || (size_t) need_ >= remaining) \
281 return -1; \
282 int written_ = snprintf(cursor, remaining, __VA_ARGS__); \
283 if (written_ < 0 || written_ != need_) \
284 return -1; \
285 cursor += written_; \
286 remaining -= (size_t) written_; \
287 } while (0)
288
289 SBOM_APPEND("{\n \"spdxVersion\": \"SPDX-2.3\",\n \"name\": \"%s\",\n"
290 " \"version\": \"%s\",\n \"created\": \"%s\",\n"
291 " \"components\": [\n",
292 doc->image_name, doc->version, doc->created);
293 for (int i = 0; i < doc->num_components; i++)
294 {
295 SBOM_APPEND(" {\"name\": \"%s\", \"version\": \"%s\", \"type\": \"%s\", \"supplier\": \"%s\"}%s\n",
296 doc->components[i].name, doc->components[i].version, doc->components[i].type,
297 doc->components[i].supplier, i < doc->num_components - 1 ? "," : "");
298 }
299 SBOM_APPEND(" ]\n}\n");
300
301#undef SBOM_APPEND
302 return (int) (out_len - remaining);
303}
