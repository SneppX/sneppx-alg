76int SNEPPX_poison_detector_train(SNEPPXPoisonDetector* pd, const double* samples, int sample_count)
77{
78 if (!pd || !samples || sample_count < 2)
79 return -1;
80 int f = pd->feature_count;
81 for (int j = 0; j < f; j++)
82 {
83 double sum = 0.0, sum_sq = 0.0;
84 for (int i = 0; i < sample_count; i++)
85 {
86 double v = samples[i * f + j];
87 sum += v;
88 sum_sq += v * v;
89 }
90 pd->feature_means[j] = sum / sample_count;
91 double var = (sum_sq / sample_count) - (pd->feature_means[j] * pd->feature_means[j]);
92 pd->feature_stds[j] = (var > 0) ? sqrt(var) : 1.0;
93 }
94 pd->trained = 1;
95 poison_sample_count = sample_count;
96 if (ensemble_count < SNEPPX_POISON_ENSEMBLE_K)
97 {
98 for (int j = 0; j < f; j++)
99 {
100 ensemble_means[ensemble_count][j] = pd->feature_means[j];
101 ensemble_stds[ensemble_count][j] = pd->feature_stds[j];
102 }
103 ensemble_trained[ensemble_count] = 1;
104 ensemble_count++;
105 }
106 return 0;
107}

159int SNEPPX_poison_detector_update_incremental(SNEPPXPoisonDetector* pd, const double* sample)
160{
161 if (!pd || !sample || !pd->trained)
162 return -1;
163 int f = pd->feature_count;
164 double n = (double) poison_sample_count;
165 for (int j = 0; j < f; j++)
166 {
167 double old_mean = pd->feature_means[j];
168 pd->feature_means[j] = (old_mean * n + sample[j]) / (n + 1.0);
169 double old_var = pd->feature_stds[j] * pd->feature_stds[j];
170 double new_var = (n * (old_var + old_mean * old_mean) + sample[j] * sample[j]) / (n + 1.0) -
171 pd->feature_means[j] * pd->feature_means[j];
172 pd->feature_stds[j] = (new_var > 0) ? sqrt(new_var) : 1.0;
173 }
174 poison_sample_count++;
175 return 0;
176}

217int SNEPPX_poison_detector_ensemble_score(SNEPPXPoisonDetector* pd, const double* sample, int feature_count,

219{
220 if (!pd || !sample || !ensemble_outlier_score)
221 return -1;

227 for (int k = 0; k < ensemble_count; k++)
228 {

232 for (int j = 0; j < feature_count; j++)
233 {

235 if (z > max_per_model)
236 max_per_model = z;
237 }
238 if (max_per_model > max_z)
239 max_z = max_per_model;
240 }
241 *ensemble_outlier_score = max_z;
242 return 0;

252int SNEPPX_poison_detector_incremental_update(SNEPPXPoisonDetector* pd, const double* sample, double learning_rate)
253{
254 if (!pd || !sample || !pd->trained)
255 return -1;
256 if (learning_rate <= 0.0 || learning_rate > 1.0)
257 learning_rate = 0.1;
258 int f = pd->feature_count;
259 for (int j = 0; j < f; j++)
260 {
261 double diff = sample[j] - pd->feature_means[j];
262 pd->feature_means[j] += learning_rate * diff;
263 double var = pd->feature_stds[j] * pd->feature_stds[j];
264 var += learning_rate * (diff * diff - var);
265 pd->feature_stds[j] = (var > 0) ? sqrt(var) : 1.0;
266 }
267 poison_sample_count++;
268 return 0;
269}

498int SNEPPX_poison_detector_cross_validate(const double* samples, int sample_count, int folds, double* mean_accuracy)
499{
500 if (!samples || sample_count < 2 || folds < 2 || !mean_accuracy)
501 return -1;
502 double total_acc = 0.0;
503 int fold_size = sample_count / folds;
504 for (int f = 0; f < folds; f++)
505 {
506 int test_start = f * fold_size;
507 int test_end = (f + 1) * fold_size;
508 if (f == folds - 1)
509 test_end = sample_count;
510 int test_count = test_end - test_start;
511 int train_count = sample_count - test_count;
512 SNEPPXPoisonDetector pd;
513 SNEPPX_poison_detector_init(&pd, 1);
514 double* train = (double*) malloc(train_count * sizeof(double));
515 if (!train)
516 {
517 SNEPPX_poison_detector_destroy(&pd);
518 return -1;
519 }
520 int ti = 0;
521 for (int i = 0; i < sample_count; i++)
522 {
523 if (i < test_start || i >= test_end)
524 {
525 train[ti] = samples[i];
526 ti++;
527 }
528 }
529 SNEPPX_poison_detector_train(&pd, train, train_count);
530 int correct = 0;
531 for (int i = test_start; i < test_end; i++)
532 {
533 double s;
534 SNEPPX_poison_detector_score(&pd, &samples[i], 1, &s);
535 if (s <= pd.outlier_threshold)
536 correct++;
537 }
538 total_acc += (double) correct / (double) test_count;
539 free(train);
540 SNEPPX_poison_detector_destroy(&pd);
541 }
542 *mean_accuracy = total_acc / (double) folds;
543 return 0;
544}

1126 concordant += 0.5;

1298 if (pred == labels[i])
1299 bin_acc[b] += 1.0;
1300 bin_count[b]++;

1304 for (int b = 0; b < bins; b++)
1305 {

1310 total += fabs(acc - conf) * (double) bin_count[b];
1311 total_count += bin_count[b];
