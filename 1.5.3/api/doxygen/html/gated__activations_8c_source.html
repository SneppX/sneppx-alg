84 case SNEPPX_GATED_SWIGLU:

89 break;

97 act_val = 0.5f * xv * (1.0f + t);
98 d_act = 0.5f * (1.0f + t) + 0.5f * xv * c * (1.0f + 3.0f * 0.044715f * xv * xv) * sech2;

104 break;
