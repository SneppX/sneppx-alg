112static int transport_compute_aad_tag(const uint8_t key[SNEPPX_TLS_KEY_LEN], uint64_t sequence,
113 const uint8_t nonce[SNEPPX_TLS_NONCE_LEN], const uint8_t* aad, size_t aad_len,

115{

117 const size_t fixed = sizeof(domain) + 8 + SNEPPX_TLS_NONCE_LEN + 8 + 8;
118 if (aad_len > SIZE_MAX - fixed)
119 return -1;
120 size_t total = fixed + aad_len;

137 offset += 8;
138 transport_encode_u64(data + offset, (uint64_t) len);

714int SNEPPX_transport_decrypt_aead(SNEPPXTransportSecurity* ts, int session_id, const uint8_t* ciphertext, size_t len,
715 const uint8_t* aad, size_t aad_len, const uint8_t nonce[SNEPPX_TLS_NONCE_LEN],
716 const uint8_t tag[SNEPPX_TRANSPORT_AAD_TAG_LEN], uint8_t* plaintext)
717{
718 if (!tag)
719 return -1;
720 if (!ts || !ciphertext || !plaintext || !nonce)
721 return -1;
722 if (!aad && aad_len > 0)
723 return -1;
724 if (session_id < 0 || session_id >= ts->session_count)
725 return -1;
726 SNEPPXTLSSession* s = &ts->sessions[session_id];
727 if (!s->is_active)
728 return -1;
729 if (session_timeouts[session_id] > 0)
730 {
731 uint64_t now = (uint64_t) time(NULL);
732 if (now - s->last_used > session_timeouts[session_id])
733 return -1;
734 }
735 if (use_aes_gcm)
736 return -1;
737
738 uint64_t sequence = transport_decode_sequence(nonce);
739 if (sequence == 0 || sequence != session_rx_sequence[session_id] + 1)
740 return -1;
741
742 uint8_t message_key[SNEPPX_TLS_KEY_LEN];
743 uint8_t aad_nonce[SNEPPX_TLS_NONCE_LEN];
744 transport_derive_message_key(s->session_key, sequence, message_key);
745 if (transport_derive_aad_nonce(nonce, aad, aad_len, aad_nonce) != 0)
746 {
747 memset(message_key, 0, sizeof(message_key));
748 return -1;
749 }
750
751 {
752 SNEPPXChaCha20State state;
753 SNEPPX_chacha20_init(&state, message_key, aad_nonce, 0);
754 memcpy(plaintext, ciphertext, len);
755 SNEPPX_chacha20_encrypt(&state, plaintext, len);
756 memset(&state, 0, sizeof(state));
757 }
758
759 uint8_t expected_tag[SNEPPX_TRANSPORT_AAD_TAG_LEN];
760 int rc = transport_compute_aad_tag(message_key, sequence, nonce, aad, aad_len, ciphertext, len, expected_tag);
761 if (rc == 0 && !transport_tags_equal(tag, expected_tag))
762 rc = -1;
763 memset(expected_tag, 0, sizeof(expected_tag));
764 memset(message_key, 0, sizeof(message_key));
765 memset(aad_nonce, 0, sizeof(aad_nonce));
766 if (rc != 0)
767 {
768 memset(plaintext, 0, len);
769 return -1;
770 }
771
772 session_rx_sequence[session_id] = sequence;
773 s->last_used = (uint64_t) time(NULL);
774 memset(session_aad_keys[session_id], 0, 16);
775 if (aad && aad_len > 0)
776 memcpy(session_aad_keys[session_id], aad, aad_len < 16 ? aad_len : 16);
777 g_total_decrypted += (uint64_t) len;
778 return 0;
779}
