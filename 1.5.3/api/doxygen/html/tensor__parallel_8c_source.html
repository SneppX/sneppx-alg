61// Column-parallel linear: Y = X * W^T (split along K dimension)
62// Input: [M, K], Weight: [N, K] split into [N/tp, K] per rank
63// Output: partial [M, N/tp], all-reduce to get full [M, N]
64int sneppx_tp_linear_forward(SNEPPX_TensorParallel* tp, const float* input, const float* weight, float* output, int m,
65 int n, int k, cudaStream_t stream)
66{
67 if (!tp || !input || !weight || !output)
68 return -1;
69
70 cublasHandle_t handle;
71 cublasCreate(&handle);
72 cublasSetStream(handle, stream);
73
74 // Each rank computes partial output
75 int n_local = n / tp->tp_size;
76 int rank_offset = tp->tp_rank * n_local;
77
78 float alpha = 1.0f, beta = 0.0f;
79 cublasSgemm(handle, CUBLAS_OP_T, CUBLAS_OP_N, n_local, m, k, &alpha, weight, k, // [n_local, k] partial weight
80 input, k, // [m, k] input
81 &beta, output, n_local); // [m, n_local] partial output
82
83 // All-reduce to get full output
84 float* full_output;
85 cudaMalloc(&full_output, m * n * sizeof(float));
86 cudaMemcpy(full_output + (size_t) rank_offset * m, output, (size_t) n_local * m * sizeof(float),
87 cudaMemcpyDeviceToDevice);
88
89 // Real tensor-parallel all-reduce (sum) across TP ranks via NCCL
90 if (tp->pg)
91 {
92 int ret = sneppx_pg_all_reduce(tp->pg, full_output, (size_t) m * n, SNEPPX_NCCL_FLOAT, SNEPPX_NCCL_SUM, stream);
93 if (ret != 0)
94 {
95 cudaFree(full_output);
96 cublasDestroy(handle);
97 return ret;
98 }
99 }
100
101 cudaMemcpy(output, full_output, m * n * sizeof(float), cudaMemcpyDeviceToDevice);
102 cudaFree(full_output);
103
104 cublasDestroy(handle);
105 return 0;
106}
