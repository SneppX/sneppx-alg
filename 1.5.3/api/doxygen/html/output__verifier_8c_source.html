1223int SNEPPX_output_verifier_sanitize_pii(const char* text, size_t len, char* safe, size_t* safe_len)
1224{
1225 if (!text || !safe || !safe_len)
1226 return -1;
1227 size_t cap = *safe_len;
1228 *safe_len = 0;
1229 size_t pos = 0;
1230 const char redacted[] = "[REDACTED]";
1231 size_t rlen = strlen(redacted);
1232 for (size_t i = 0; i < len && pos < cap; i++)
1233 {
1234 if (text[i] == '@')
1235 {
1236 size_t start = i;
1237 while (start > 0 && text[start - 1] != ' ' && text[start - 1] != '\t' && text[start - 1] != '\n' &&
1238 text[start - 1] != '\r')
1239 start--;
1240 size_t end = i;
1241 while (end < len && text[end] != ' ' && text[end] != '\t' && text[end] != '\n' && text[end] != '\r')
1242 end++;
1243 if (pos + rlen <= cap)
1244 {
1245 memcpy(safe + pos, redacted, rlen);
1246 pos += rlen;
1247 }
1248 i = end;
1249 if (i > 0)
1250 i--;
1251 }
1252 else
1253 {
1254 if (pos < cap)
1255 safe[pos++] = text[i];
1256 }
1257 }
1258 *safe_len = pos;
1259 if (pos < cap)
1260 safe[pos] = 0;
1261 return 0;
1262}

1272int SNEPPX_output_verifier_sanitize_profanity(const char* text, size_t len, char* safe, size_t* safe_len)
1273{
1274 if (!text || !safe || !safe_len)
1275 return -1;
1276 size_t cap = *safe_len;

1283 for (size_t i = 0; i < clen; i++)
1284 lower[i] = (char) tolower((unsigned char) text[i]);
1285 lower[clen] = '\0';
1286 for (size_t i = 0; i < len && pos < cap; i++)
1287 {
1288 int matched = 0;
1289 for (int p = 0; profanity_list[p]; p++)
1290 {
1291 size_t plen = strlen(profanity_list[p]);

1304 {
1305 if (pos + rlen <= cap)
1306 {
1307 memcpy(safe + pos, redacted, rlen);
1308 pos += rlen;
1309 }
1310 i += plen - 1;
1311 matched = 1;
1312 break;
1313 }
1314 }
1315 }

1318 if (pos < cap)
1319 safe[pos++] = text[i];
