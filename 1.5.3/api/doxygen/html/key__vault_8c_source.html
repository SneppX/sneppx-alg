72int SNEPPX_key_vault_generate_key(SNEPPXKeyVault* vault, uint8_t* key_id, uint64_t ttl_seconds)
73{
74 if (!vault || vault->is_locked || !key_id || vault->key_count >= SNEPPX_VAULT_MAX_KEYS)
75 return -1;
76 SNEPPXVaultKey* k = &vault->keys[vault->key_count];
77 SNEPPX_random_bytes(k->id, SNEPPX_VAULT_ID_LEN);
78 SNEPPX_random_bytes(k->key_data, SNEPPX_VAULT_KEY_LEN);
79 k->created_at = (uint64_t) time(NULL);
80 k->expires_at = (ttl_seconds > 0) ? k->created_at + ttl_seconds : 0;
81 k->is_active = 1;
82 k->access_count = 0;
83 if (key_id)
84 memcpy(key_id, k->id, SNEPPX_VAULT_ID_LEN);
85 return vault->key_count++;

190int SNEPPX_key_vault_unlock(SNEPPXKeyVault* vault, const uint8_t* master_key)
191{
192 if (!vault)
193 return -1;
194 if (master_key_set && master_key)
195 {
196 uint32_t check = 0;
197 for (int i = 0; i < 32 && i < SNEPPX_VAULT_KEY_LEN; i++)
198 check |= (uint8_t) (master_key[i] ^ master_key_hash[i]);
199 if (check != 0)
200 return -1;
201 }
202 if (master_key && !master_key_set)
203 {
204 for (int i = 0; i < 32 && i < SNEPPX_VAULT_KEY_LEN; i++)
205 master_key_hash[i] = master_key[i];
206 master_key_set = 1;
207 }
208 vault->is_locked = 0;
209 return 0;

553static int vault_compute_checksum(const uint8_t* data, size_t len, uint8_t* sum_out)
554{
555 if (!data || !sum_out || len == 0)
556 return -1;
557 uint32_t crc = 0xFFFFFFFF;
558 for (size_t i = 0; i < len; i++)
559 {
560 crc ^= data[i];
561 for (int b = 0; b < 8; b++)
562 crc = (crc >> 1) ^ (crc & 1 ? 0xEDB88320 : 0);
563 }
564 crc ^= 0xFFFFFFFF;
565 for (int i = 0; i < 4; i++)
566 sum_out[i] = (uint8_t) (crc >> (i * 8));
567 return 0;
568}
