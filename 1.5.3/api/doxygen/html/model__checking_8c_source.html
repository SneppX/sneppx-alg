99SNEPPXModelCheckResult SNEPPX_model_check(SNEPPXFormalModel* model)
100{
101 SNEPPXModelCheckResult result;
102 memset(&result, 0, sizeof(result));
103 if (!model)
104 {
105 result.property_satisfied = 0;
106 return result;
107 }
108
109 result.total_states = model->state_count;
110 int visited[SNEPPX_MODEL_MAX_STATES] = {0};
111 int parent[SNEPPX_MODEL_MAX_STATES];
112 for (int i = 0; i < SNEPPX_MODEL_MAX_STATES; i++)
113 parent[i] = -1;
114 int stack[SNEPPX_MODEL_MAX_STATES];
115 int stack_top = 0;
116 stack[stack_top++] = model->initial_state;
117 visited[model->initial_state] = 1;
118
119 int first_error = -1;
120
121 while (stack_top > 0)
122 {
123 int cur = stack[--stack_top];
124 result.reachable_states++;
125 for (int i = 0; i < model->state_count; i++)
126 {
127 if (model->states[i].state_id != (uint32_t) cur)
128 continue;
129 if (model->states[i].is_error)
130 {
131 result.error_states++;
132 if (first_error < 0)
133 first_error = (int) model->states[i].state_id;
134 }
135 if (model->states[i].next_count == 0)
136 result.deadlock_states++;
137 for (int j = 0; j < model->states[i].next_count; j++)
138 {
139 uint32_t next = model->states[i].next_states[j];
140 int found = 0;
141 for (int k = 0; k < model->state_count; k++)
142 {
143 if (model->states[k].state_id == next)
144 {
145 found = 1;
146 break;
147 }
148 }
149 if (found && !visited[next])
150 {
151 visited[next] = 1;
152 parent[next] = cur;
153 if (stack_top < SNEPPX_MODEL_MAX_STATES)
154 stack[stack_top++] = next;
155 }
156 }
157 }
158 }
159
160 result.property_satisfied = (result.error_states == 0) ? 1 : 0;
161
162 if (result.property_satisfied)
163 {
164 snprintf(result.counterexample, sizeof(result.counterexample), "ok");
165 }
166 else if (first_error >= 0)
167 {
168 uint32_t path[SNEPPX_MODEL_MAX_STATES];
169 int path_len = 0;
170 int node = first_error;
171 while (node >= 0 && path_len < SNEPPX_MODEL_MAX_STATES)
172 {
173 path[path_len++] = (uint32_t) node;
174 node = parent[node];
175 }
176 for (int i = 0; i < path_len / 2; i++)
177 {
178 uint32_t t = path[i];
179 path[i] = path[path_len - 1 - i];
180 path[path_len - 1 - i] = t;
181 }
182 int pos = 0;
183 for (int i = 0; i < path_len; i++)
184 {
185 pos += snprintf(result.counterexample + pos, (size_t) (sizeof(result.counterexample) - pos), "%u", path[i]);
186 if (i < path_len - 1 && (size_t) (pos + 4) < sizeof(result.counterexample))
187 {
188 pos += snprintf(result.counterexample + pos, (size_t) (sizeof(result.counterexample) - pos), " -> ");
189 }
190 }
191 }
192 else
193 {
194 snprintf(result.counterexample, sizeof(result.counterexample), "error state reachable");
195 }
196 return result;
197}
