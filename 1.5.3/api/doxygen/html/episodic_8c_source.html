49static int compare_priority_desc(const void* a, const void* b)
50{
51 const SNEPPXEpisodicExperience* ea = (const SNEPPXEpisodicExperience*) a;
52 const SNEPPXEpisodicExperience* eb = (const SNEPPXEpisodicExperience*) b;
53 if (eb->priority > ea->priority)
54 return 1;
55 if (eb->priority < ea->priority)
56 return -1;
57 return 0;

150 }

174int sneppx_episodic_retrieve(SNEPPXEpisodicBuffer* buf, uint64_t time_start, uint64_t time_end, SNEPPXTensor** output)
175{
176 if (!buf || !output)
177 return -1;
178 size_t match_count = 0;
179 for (size_t i = 0; i < buf->capacity; i++)
180 {
181 if (buf->experiences[i].valid && buf->experiences[i].timestamp >= time_start &&
182 buf->experiences[i].timestamp <= time_end)
183 {
184 match_count++;
185 }
186 }
187 if (match_count == 0)
188 {
189 *output = NULL;
190 return 0;
191 }
192 size_t state_dim = buf->experiences[0].state ? buf->experiences[0].state->size : 1;
193 size_t shape[2] = {match_count, state_dim};
194 SNEPPXTensor* result = SNEPPX_tensor_zeros(shape, 2, SNEPPX_FLOAT32);
195 if (!result)
196 return -1;
197 size_t row = 0;
198 float* data = (float*) result->data;
199 for (size_t i = 0; i < buf->capacity; i++)
200 {
201 if (buf->experiences[i].valid && buf->experiences[i].timestamp >= time_start &&
202 buf->experiences[i].timestamp <= time_end)
203 {
204 if (buf->experiences[i].state)
205 {
206 memcpy(data + row * state_dim, buf->experiences[i].state->data, state_dim * sizeof(float));
207 }
208 row++;
209 }
210 }
211 *output = result;
212 return 0;

215int sneppx_episodic_sample(SNEPPXEpisodicBuffer* buf, size_t batch_size, SNEPPXTensor** states, SNEPPXTensor** actions,
216 float* rewards, SNEPPXTensor** next_states)
217{
218 if (!buf || !states || !rewards || batch_size == 0)
219 return -1;
220 size_t available = 0;
221 for (size_t i = 0; i < buf->capacity; i++)
222 {
223 if (buf->experiences[i].valid)
224 available++;
225 }
226 if (available == 0)
227 return -1;
228 size_t actual_batch = batch_size < available ? batch_size : available;
229 SNEPPXEpisodicExperience* sorted = SNEPPX_malloc(available * sizeof(SNEPPXEpisodicExperience), 64);
230 if (!sorted)
231 return -1;
232 size_t idx = 0;
233 for (size_t i = 0; i < buf->capacity; i++)
234 {
235 if (buf->experiences[i].valid)
236 {
237 sorted[idx++] = buf->experiences[i];
238 }
239 }
240 qsort(sorted, available, sizeof(SNEPPXEpisodicExperience), compare_priority_desc);
241 float total_priority = 0.0f;
242 for (size_t i = 0; i < available; i++)
243 total_priority += sorted[i].priority;
244 if (total_priority < 1e-10f)
245 total_priority = 1.0f;
246 for (size_t i = 0; i < actual_batch; i++)
247 {
248 float r = frand(buf->seed_ptr) * total_priority;
249 float cum = 0.0f;
250 size_t chosen = 0;
251 for (size_t j = 0; j < available; j++)
252 {
253 cum += sorted[j].priority;
254 if (r <= cum)
255 {
256 chosen = j;
257 break;
258 }
259 }
260 SNEPPXEpisodicExperience* exp = &sorted[chosen];
261 if (exp->state)
262 {
263 states[i] = SNEPPX_tensor_copy(exp->state);
264 if (!states[i])
265 {
266 for (size_t k = 0; k < i; k++)
267 SNEPPX_tensor_destroy(states[k]);
268 SNEPPX_free(sorted, available * sizeof(SNEPPXEpisodicExperience));
269 return -1;
270 }
271 }
272 else
273 {
274 states[i] = NULL;
275 }
276 if (actions && exp->action)
277 {
278 actions[i] = SNEPPX_tensor_copy(exp->action);
279 }
280 else if (actions)
281 {
282 actions[i] = NULL;
283 }
284 rewards[i] = exp->reward;
285 if (next_states && exp->next_state)
286 {
287 next_states[i] = SNEPPX_tensor_copy(exp->next_state);
288 }
289 else if (next_states)
290 {
291 next_states[i] = NULL;
292 }
293 sorted[chosen].priority *= 0.9f;
294 }
295 SNEPPX_free(sorted, available * sizeof(SNEPPXEpisodicExperience));
296 return (int) actual_batch;
297}

299int sneppx_episodic_consolidate(SNEPPXEpisodicBuffer* buf)
300{
301 if (!buf)
302 return -1;
303 buf->last_consolidation = buf->next_timestamp;
304 if (buf->count < 2)
305 return 0;
306 float mean_reward = 0.0f;
307 float max_reward = -1e10f;
308 float min_reward = 1e10f;
309 size_t valid_count = 0;
310 for (size_t i = 0; i < buf->capacity; i++)
311 {
312 if (buf->experiences[i].valid)
313 {
314 mean_reward += buf->experiences[i].reward;
315 if (buf->experiences[i].reward > max_reward)
316 max_reward = buf->experiences[i].reward;
317 if (buf->experiences[i].reward < min_reward)
318 min_reward = buf->experiences[i].reward;
319 valid_count++;
320 }
321 }
322 if (valid_count == 0)
323 return 0;
324 mean_reward /= (float) valid_count;
325 for (size_t i = 0; i < buf->capacity; i++)
326 {
327 if (buf->experiences[i].valid)
328 {
329 float age = (float) (buf->next_timestamp - buf->experiences[i].timestamp);
330 if (age > 1000.0f && buf->experiences[i].reward < mean_reward * 0.5f)
331 {
332 buf->experiences[i].priority *= 0.5f;
333 }
334 }
335 }
336 return 0;
