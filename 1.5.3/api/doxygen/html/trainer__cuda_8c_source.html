283int SNEPPX_trainer_cuda_optimizer_step(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n)
284{
285#if defined(SNEPPX_HAS_CUDA)
286 if (!g_trainer_cuda.initialized || !opt || !params || !grads)
287 return -1;
288
289 float lr = opt->learning_rate;
290 float wd = opt->weight_decay;
291 float beta1 = opt->beta1;
292 float beta2 = opt->beta2;
293 float eps = opt->epsilon;
294 int step = (int) opt->step_count + 1;
295
296 size_t m = n < g_trainer_cuda.num_params ? n : g_trainer_cuda.num_params;
297
298 for (size_t i = 0; i < m; i++)
299 {
300 if (!params[i] || !grads[i] || !g_trainer_cuda.d_params[i] || !g_trainer_cuda.d_grads[i])
301 continue;
302 float* grad_data = (float*) grads[i]->data;
303 size_t numel = params[i]->size;
304 size_t bytes = numel * sizeof(float);
305
306 cudaMemcpyAsync(g_trainer_cuda.d_grads[i], grad_data, bytes, cudaMemcpyHostToDevice, g_trainer_cuda.stream);
307
308 if (opt->type == SNEPPX_OPTIMIZER_SGD)
309 {
310 float alpha = -lr;
311 cublasStatus_t cs =
312 cublasSaxpy(g_trainer_cuda.blas, (int) numel, &alpha, (const float*) g_trainer_cuda.d_grads[i], 1,
313 (float*) g_trainer_cuda.d_params[i], 1);
314 if (cs != CUBLAS_STATUS_SUCCESS)
315 return -1;
316
317 if (wd != 0.0f)
318 {
319 float wd_alpha = -lr * wd;
320 cs = cublasSaxpy(g_trainer_cuda.blas, (int) numel, &wd_alpha, (const float*) g_trainer_cuda.d_params[i],
321 1, (float*) g_trainer_cuda.d_params[i], 1);
322 if (cs != CUBLAS_STATUS_SUCCESS)
323 return -1;
324 }
325 }
326 else
327 {
328 /* AdamW via cuBLAS-based element-wise operations */
329 float* d_p = (float*) g_trainer_cuda.d_params[i];
330 const float* d_g = (const float*) g_trainer_cuda.d_grads[i];
331 float* d_m = g_trainer_cuda.d_state1[i];
332 float* d_v = g_trainer_cuda.d_state2[i];
333 int N = (int) numel;
334
335 /* m = beta1*m + (1-beta1)*g */
336 float one_minus_b1 = 1.0f - beta1;
337 cublasSscal(g_trainer_cuda.blas, N, &beta1, d_m, 1);
338 cublasSaxpy(g_trainer_cuda.blas, N, &one_minus_b1, d_g, 1, d_m, 1);
339
340 /* v = beta2*v + (1-beta2)*g*g */
341 float one_minus_b2 = 1.0f - beta2;
342 cublasSscal(g_trainer_cuda.blas, N, &beta2, d_v, 1);
343 /* For g*g, need an element-wise kernel — fall back to host for now */
344 float* h_g = (float*) malloc(bytes);
345 float* h_v = (float*) malloc(bytes);
346 if (!h_g || !h_v)
347 {
348 free(h_g);
349 free(h_v);
350 return -1;
351 }
352 cudaMemcpy(h_g, d_g, bytes, cudaMemcpyDeviceToHost);
353 cudaMemcpy(h_v, d_v, bytes, cudaMemcpyDeviceToHost);
354 for (size_t j = 0; j < numel; j++)
355 {
356 h_v[j] += one_minus_b2 * h_g[j] * h_g[j];
357 }
358 cudaMemcpy(d_v, h_v, bytes, cudaMemcpyHostToDevice);
359 free(h_g);
360 free(h_v);
361
362 /* bias-corrected lr */
363 float lr_corrected = lr * sqrtf(1.0f - powf(beta2, (float) step)) / (1.0f - powf(beta1, (float) step));
364
365 /* p = p - lr_corrected * m / (sqrt(v) + eps) - lr * wd * p */
366 if (wd != 0.0f)
367 {
368 float neg_lr_wd = -lr * wd;
369 cublasSaxpy(g_trainer_cuda.blas, N, &neg_lr_wd, d_p, 1, d_p, 1);
370 }
371
372 /* Need element-wise division: p -= lr_corrected * m / (sqrt(v) + eps) */
373 float* h_p = (float*) malloc(bytes);
374 float* h_m = (float*) malloc(bytes);
375 if (!h_p || !h_m)
376 {
377 free(h_p);
378 free(h_m);
379 return -1;
380 }
381 cudaMemcpy(h_p, d_p, bytes, cudaMemcpyDeviceToHost);
382 cudaMemcpy(h_m, d_m, bytes, cudaMemcpyDeviceToHost);
383 for (size_t j = 0; j < numel; j++)
384 {
385 h_p[j] -= lr_corrected * h_m[j] / (sqrtf(h_v[j]) + eps);
386 }
387 cudaMemcpy(d_p, h_p, bytes, cudaMemcpyHostToDevice);
388 free(h_p);
389 free(h_m);
390 }
391 }
392
393 cudaStreamSynchronize(g_trainer_cuda.stream);
394
395 for (size_t i = 0; i < m; i++)
396 {
397 if (!params[i] || !g_trainer_cuda.d_params[i])
398 continue;
399 cudaMemcpyAsync(params[i]->data, g_trainer_cuda.d_params[i], g_trainer_cuda.sizes[i], cudaMemcpyDeviceToHost,
400 g_trainer_cuda.stream);
401 }
402 cudaStreamSynchronize(g_trainer_cuda.stream);
403
404 return 0;
405#else
406 (void) opt;
407 (void) params;
408 (void) grads;
409 (void) n;
410 return -1;
411#endif
