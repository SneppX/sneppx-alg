60static int16_t fq_reduce(int32_t a)
61{
62 int16_t t = (int16_t) ((a + KYBER_Q * 128) % KYBER_Q);

66/* Reference Montgomery reduction: a * 2^-16 mod q in centered form,
67 * using QINV = -3327 (the minus-formula). The previous code used the
68 * plus-formula with an incorrect QINV (20159), which broke every
69 * NTT-domain multiply. */
70static int16_t mont_reduce(int32_t a)
71{
72 int16_t t = (int16_t) ((int16_t) a * KYBER_QINV);
73 return (int16_t) ((a - (int32_t) t * KYBER_Q) >> 16);
74}

76static int16_t barrett_reduce(int16_t a)
77{
78 int16_t t;
79 const int16_t v = (int16_t) (((1L << 26) + KYBER_Q / 2) / KYBER_Q);
80 t = (int16_t) (((int32_t) v * a + (1L << 25)) >> 26);
81 t = (int16_t) (t * KYBER_Q);
82 return (int16_t) (a - t);
83}

90/* Forward NTT (FIPS 203 / pqcrystals reference layout). Consumes
91 * zetas[1..127], one twiddle per 256/len block, then Barrett-reduces
92 * every coefficient. */
93static void ntt(int16_t r[256])
94{
95 int len = 128, k = 1;
96 while (len >= 2)
97 {
98 int start = 0;
99 while (start < 256)
100 {
101 int16_t zeta = zetas[k++];
102 for (int j = start; j < start + len; j++)
103 {
104 int16_t t = fqmul(zeta, r[j + len]);
105 r[j + len] = (int16_t) (r[j] - t);
106 r[j] = (int16_t) (r[j] + t);
107 }
108 start += len * 2;
109 }
110 len >>= 1;
111 }
112 for (int j = 0; j < 256; j++)
113 r[j] = barrett_reduce(r[j]);
114}

116static void inv_ntt(int16_t r[256])
117{
118 int len = 2, k = 127;
119 while (len <= 128)
120 {
121 int start = 0;
122 while (start < 256)
123 {
124 int16_t zeta = zetas[k--];
125 for (int j = start; j < start + len; j++)
126 {
127 int16_t t = r[j];
128 r[j] = barrett_reduce((int16_t) (t + r[j + len]));
129 r[j + len] = (int16_t) (r[j + len] - t);
130 r[j + len] = fqmul(zeta, r[j + len]);
131 }
132 start += len * 2;
133 }
134 len <<= 1;
135 }
136 /* mont^2/128 -- the reference "tomont" scaling factor (1441). */
137 for (int j = 0; j < 256; j++)
138 r[j] = fqmul(r[j], 1441);
139}

153/* Reference NTT-domain multiply: 2x2 basemul over Z[X]/(X^2 - zeta),
154 * with zetas[64+i] / -zetas[64+i]. A plain pointwise multiply is wrong
155 * for the Montgomery/cyclic layout. */
156static void basemul(int16_t r[2], const int16_t a[2], const int16_t b[2], int16_t zeta)
157{
158 int16_t r0 = fqmul(a[1], b[1]);
159 r0 = fqmul(r0, zeta);
160 r0 = (int16_t) (r0 + fqmul(a[0], b[0]));
161 int16_t r1 = fqmul(a[0], b[1]);
162 r1 = (int16_t) (r1 + fqmul(a[1], b[0]));
163 r[0] = r0;
164 r[1] = r1;
165}

167static void poly_mul(int16_t r[256], const int16_t a[256], const int16_t b[256])
168{
169 int16_t tmp[256], tmp2[256];
170 memcpy(tmp, a, sizeof(tmp));
171 ntt(tmp);
172 memcpy(tmp2, b, sizeof(tmp2));
173 ntt(tmp2);
174 for (int i = 0; i < 64; i++)
175 {
176 basemul(r + 4 * i, tmp + 4 * i, tmp2 + 4 * i, zetas[64 + i]);
177 basemul(r + 4 * i + 2, tmp + 4 * i + 2, tmp2 + 4 * i + 2, (int16_t) (-zetas[64 + i]));
178 }
179 inv_ntt(r);
180}

183{

185 {

204/* LSB-first bit-accumulator packer: round(2^d/q * a) mod 2^d on the
205 * positive representative. The old code used (1<<12)/q == 1 (collapsing
206 * every coefficient to 0/1) and a broken cross-byte shifter that lost
207 * bits whenever shift landed on a byte boundary. */
208static void poly_compress(uint8_t* out, const int16_t a[256], int d)
209{
210 size_t out_idx = 0;
211 uint64_t buf = 0;
212 int bits = 0;
213 memset(out, 0, (size_t) ((256 * d + 7) / 8));
214 for (int i = 0; i < 256; i++)
215 {
216 int32_t t = a[i];
217 t += (t >> 15) & KYBER_Q;
218 t = (((t << d) + KYBER_Q / 2) / KYBER_Q);
219 t &= (1 << d) - 1;
220 buf |= ((uint64_t) t) << bits;
221 bits += d;
222 while (bits >= 8)
223 {
224 out[out_idx++] = (uint8_t) (buf & 0xff);
225 buf >>= 8;
226 bits -= 8;
227 }
228 }
229}

231static void poly_decompress(int16_t r[256], const uint8_t* in, int d)
232{
233 int32_t f = (1 << d) / 2;
234 int in_idx = 0, bits = 0;
235 uint64_t buf = 0;
236 for (int i = 0; i < 256; i++)
237 {
238 while (bits < d)
239 {
240 buf |= ((uint64_t) in[in_idx++] << bits);
241 bits += 8;
242 }
243 int t = (int) (buf & ((1ULL << d) - 1));
244 buf >>= d;
245 bits -= d;
246 r[i] = (int16_t) ((t * KYBER_Q + f) >> d);
247 }
248}

272/* Centered-binomial-distribution noise sampling (FIPS 203 §4.1), eta=2.
273 * Draws 2*eta bits per coefficient (popcount(a) - popcount(b)) so the
274 * error stays small ([-2, 2]) and decapsulation noise stays well below
275 * the compress/decompress rounding threshold. */
276static void poly_getnoise(int16_t r[256], const uint8_t* seed, uint8_t nonce)
277{
278 uint8_t buf[128];
279 SNEPPXDRBG drbg;
280 SNEPPX_drbg_init(&drbg, seed, 32, &nonce, 1);
281 SNEPPX_drbg_generate(&drbg, buf, sizeof(buf));
282 SNEPPX_drbg_destroy(&drbg);
283 int pos = 0;
284 for (int i = 0; i < 256; i++)
285 {
286 int a = 0, b = 0;
287 for (int j = 0; j < KYBER_ETA1; j++)
288 {
289 a += (buf[pos >> 3] >> (pos & 7)) & 1;
290 pos++;
291 }
292 for (int j = 0; j < KYBER_ETA1; j++)
293 {
294 b += (buf[pos >> 3] >> (pos & 7)) & 1;
295 pos++;
296 }
297 r[i] = (int16_t) (a - b);
298 }
