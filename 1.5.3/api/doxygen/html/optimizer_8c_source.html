166static void step_sgd(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n)
167{
168 for (size_t i = 0; i < n; i++)
169 {
170 if (!params[i] || !grads[i])
171 continue;
172 float* pd = (float*) params[i]->data;
173 float* gd = (float*) grads[i]->data;
174 float lr = opt->learning_rate, wd = opt->weight_decay, clip = opt->grad_clip;
175 float mu = opt->momentum, dp = opt->dampening;
176 float* md = opt->momentum_buffers[i] ? (float*) opt->momentum_buffers[i]->data : NULL;
177 size_t sz = params[i]->size;
178
179 if (md && mu > 0)
180 {
181 for (size_t j = 0; j < sz; j++)
182 {
183 float g = clip_grad(gd[j], clip);
184 g += wd * pd[j];
185 md[j] = mu * md[j] - lr * g;
186 pd[j] += md[j];
187 }
188 }
189 else
190 {
191 for (size_t j = 0; j < sz; j++)
192 {
193 float g = clip_grad(gd[j], clip);
194 g += wd * pd[j];
195 pd[j] -= lr * g;
196 }
197 }
198 }

201static void step_adam(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n, int decoupled_wd)
202{
203 float lr = opt->learning_rate, b1 = opt->beta1, b2 = opt->beta2;
204 float eps = opt->epsilon, wd = opt->weight_decay, clip = opt->grad_clip;
205 float bias1 = 1.0f, bias2 = 1.0f;
206 for (size_t t = 0; t < opt->step_count; t++)
207 {
208 bias1 *= b1;
209 bias2 *= b2;
210 }
211 bias1 = 1.0f - bias1;
212 bias2 = 1.0f - bias2;
213
214 for (size_t i = 0; i < n; i++)
215 {
216 if (!params[i] || !grads[i])
217 continue;
218 float* pd = (float*) params[i]->data;
219 float* gd = (float*) grads[i]->data;
220 float* m1 = (float*) opt->momentum_buffers[i]->data;
221 float* m2 = (float*) opt->state_buf2[i]->data;
222 size_t sz = params[i]->size;
223
224 for (size_t j = 0; j < sz; j++)
225 {
226 float g = clip_grad(gd[j], clip);
227 m1[j] = b1 * m1[j] + (1.0f - b1) * g;
228 m2[j] = b2 * m2[j] + (1.0f - b2) * g * g;
229 float m1_hat = m1[j] / bias1;
230 float m2_hat = m2[j] / bias2;
231 float step = lr * m1_hat / (sqrtf(m2_hat) + eps);
232 if (decoupled_wd)
233 step += lr * wd * pd[j];
234 pd[j] -= step;
235 if (!decoupled_wd)
236 pd[j] -= lr * wd * pd[j];
237 }
238 }

241static void step_adamax(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n)
242{
243 float lr = opt->learning_rate, b1 = opt->beta1, b2 = opt->beta2;
244 float eps = opt->epsilon, wd = opt->weight_decay, clip = opt->grad_clip;
245 float bias1 = 1.0f;
246 for (size_t t = 0; t < opt->step_count; t++)
247 bias1 *= b1;
248 bias1 = 1.0f - bias1;
249
250 for (size_t i = 0; i < n; i++)
251 {
252 if (!params[i] || !grads[i])
253 continue;
254 float* pd = (float*) params[i]->data;
255 float* gd = (float*) grads[i]->data;
256 float* m1 = (float*) opt->momentum_buffers[i]->data;
257 float* m2 = (float*) opt->state_buf2[i]->data;
258 size_t sz = params[i]->size;
259
260 for (size_t j = 0; j < sz; j++)
261 {
262 float g = clip_grad(gd[j], clip);
263 m1[j] = b1 * m1[j] + (1.0f - b1) * g;
264 m2[j] = fmaxf(b2 * m2[j], fabsf(g));
265 pd[j] -= lr * m1[j] / (bias1 * (m2[j] + eps)) + lr * wd * pd[j];
266 }
267 }

270static void step_rmsprop(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n)
271{
272 float lr = opt->learning_rate, alpha = opt->beta1, eps = opt->epsilon;
273 float wd = opt->weight_decay, clip = opt->grad_clip, mu = opt->momentum;
274
275 for (size_t i = 0; i < n; i++)
276 {
277 if (!params[i] || !grads[i])
278 continue;
279 float* pd = (float*) params[i]->data;
280 float* gd = (float*) grads[i]->data;
281 float* v = (float*) opt->state_buf2[i]->data;
282 float* m = opt->momentum_buffers[i] ? (float*) opt->momentum_buffers[i]->data : NULL;
283 size_t sz = params[i]->size;
284
285 for (size_t j = 0; j < sz; j++)
286 {
287 float g = clip_grad(gd[j], clip);
288 g += wd * pd[j];
289 v[j] = alpha * v[j] + (1.0f - alpha) * g * g;
290 float step = lr * g / (sqrtf(v[j]) + eps);
291 if (m && mu > 0)
292 {
293 m[j] = mu * m[j] + step;
294 pd[j] -= m[j];
295 }
296 else
297 {
298 pd[j] -= step;
299 }
300 }
301 }

327static void step_adadelta(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n)
328{
329 float rho = opt->rho, eps = opt->epsilon, wd = opt->weight_decay, clip = opt->grad_clip;
330
331 for (size_t i = 0; i < n; i++)
332 {
333 if (!params[i] || !grads[i])
334 continue;
335 float* pd = (float*) params[i]->data;
336 float* gd = (float*) grads[i]->data;
337 float* v = (float*) opt->state_buf2[i]->data;
338 float* d = (float*) opt->state_buf3[i]->data;
339 size_t sz = params[i]->size;
340
341 for (size_t j = 0; j < sz; j++)
342 {
343 float g = clip_grad(gd[j], clip);
344 g += wd * pd[j];
345 v[j] = rho * v[j] + (1.0f - rho) * g * g;
346 float delta = g * sqrtf(d[j] + eps) / sqrtf(v[j] + eps);
347 d[j] = rho * d[j] + (1.0f - rho) * delta * delta;
348 pd[j] -= delta;
349 }
350 }

360void SNEPPX_optimizer_step(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t num_params)
361{
362 if (!opt || !params || !grads || num_params == 0)
363 return;
364 ensure_bufs(opt, params, num_params);
365 if (opt->num_params == 0)
366 return;
367 opt->step_count++;
368
369 switch (opt->type)
370 {
371 case SNEPPX_OPTIMIZER_SGD:
372 step_sgd(opt, params, grads, num_params);
373 break;
374 case SNEPPX_OPTIMIZER_ADAM:
375 step_adam(opt, params, grads, num_params, 0);
376 break;
377 case SNEPPX_OPTIMIZER_ADAMW:
378 step_adam(opt, params, grads, num_params, 1);
379 break;
380 case SNEPPX_OPTIMIZER_ADAMAX:
381 step_adamax(opt, params, grads, num_params);
382 break;
383 case SNEPPX_OPTIMIZER_RMSPROP:
384 step_rmsprop(opt, params, grads, num_params);
385 break;
386 case SNEPPX_OPTIMIZER_ADAGRAD:
387 step_adagrad(opt, params, grads, num_params);
388 break;
389 case SNEPPX_OPTIMIZER_ADADELTA:
390 step_adadelta(opt, params, grads, num_params);
391 break;

608void SNEPPX_lr_scheduler_step(SNEPPXLRScheduler* sched, float current_loss)
609{
610 if (!sched || !sched->lr_ptr)
611 return;
612 sched->last_epoch++;
613
614 switch (sched->type)
615 {
616 case SNEPPX_LR_STEP:
617 if (sched->step_size > 0 && sched->last_epoch % sched->step_size == 0)
618 *sched->lr_ptr *= sched->gamma;
619 break;
620 case SNEPPX_LR_EXPONENTIAL:
621 *sched->lr_ptr *= sched->gamma;
622 break;
623 case SNEPPX_LR_COSINE:
624 {
625 float progress = (float) sched->last_epoch / (float) (sched->total_steps > 0 ? sched->total_steps : 1);
626 if (progress > 1.0f)
627 progress = 1.0f;
628 *sched->lr_ptr =
629 sched->min_lr + 0.5f * (sched->max_lr - sched->min_lr) * (1.0f + cosf((float) 3.14159265f * progress));
630 break;
631 }
632 case SNEPPX_LR_REDUCE_ON_PLATEAU:
633 {
634 int improved = sched->mode_min ? (current_loss < sched->best_loss) : (current_loss > sched->best_loss);
635 if (improved)
636 {
637 sched->best_loss = current_loss;
638 sched->wait_count = 0;
639 }
640 else
641 {
642 sched->wait_count++;
643 if (sched->wait_count >= (int) sched->patience && sched->patience > 0)
644 {
645 *sched->lr_ptr *= sched->factor;
646 sched->wait_count = 0;
647 }
648 }
649 break;
650 }
651 }
