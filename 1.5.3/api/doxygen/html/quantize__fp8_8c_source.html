39uint8_t SNEPPX_float_to_fp8_e4m3(float value)
40{
41 uint32_t b;
42 memcpy(&b, &value, sizeof(b));
43 uint32_t sign = (b >> 31) & 1;
44 uint32_t exp_bits = (b >> 23) & 0xFF;
45
46 /* Handle special float32 values */
47 if (exp_bits == 0xFF)
48 {
49 /* Inf or NaN -> FP8 Inf (0x78/0xF8) */
50 return (uint8_t) ((sign << 7) | 0x78);
51 }
52 if (exp_bits == 0)
53 {
54 /* Subnormal or zero -> FP8 zero */
55 return (uint8_t) (sign << 7);
56 }
57 int32_t exp = (int32_t) exp_bits - 127;
58 uint32_t mant = (b >> 20) & 0x07;
59
60 if (exp < -6)
61 {
62 return (uint8_t) (sign << 7);
63 }
64 if (exp > 8)
65 {
66 return (uint8_t) ((sign << 7) | 0x78); /* Inf */
67 }
68 uint8_t e4m3_exp = (uint8_t) (exp + 7);
69 uint8_t e4m3 = (uint8_t) ((sign << 7) | (e4m3_exp << 3) | mant);
70 return e4m3;
71}

81{
