42static float get_flat(const SNEPPXTensor* t, size_t flat)
43{
44 size_t idx[8];
45 size_t tmp = flat;
46 for (long i = (long) t->ndim - 1; i >= 0; i--)
47 {
48 idx[i] = (t->strides[i] > 0) ? (tmp / t->strides[i]) % t->shape[i] : 0;
49 if (t->strides[i] > 0)
50 tmp %= t->strides[i];
51 }
52 return SNEPPX_tensor_get_f32(t, idx);
53}

131int SNEPPX_grad_opt_step(SNEPPXGradientOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t num_params)
132{
133 if (!opt || !params || !grads)
134 return -1;
135 ensure_buffers(opt, num_params);
136 opt->step_count++;
137 int is_adam = (opt->type == SNEPPX_OPTIMIZER_ADAM || opt->type == SNEPPX_OPTIMIZER_ADAMW ||
138 opt->type == SNEPPX_OPTIMIZER_RADAM || opt->type == SNEPPX_OPTIMIZER_SF_ADAMW);
139 double lr = opt->lr;
140 double wd = opt->weight_decay;
141 for (size_t p = 0; p < num_params; p++)
142 {
143 SNEPPXTensor* param = params[p];
144 SNEPPXTensor* grad = grads[p];
145 if (!param || !grad)
146 return -1;
147 size_t n = param->size < grad->size ? param->size : grad->size;
148 for (size_t i = 0; i < n; i++)
149 {
150 float value = get_flat(param, i);
151 float g = get_flat(grad, i);
152 if (wd != 0.0)
153 g += (float) (wd * value);
154 if (is_adam)
155 {
156 float* m = &opt->buf[p];
157 float* v = &opt->buf2[p];
158 *m = (float) (opt->beta1 * (*m) + (1.0 - opt->beta1) * g);
159 *v = (float) (opt->beta2 * (*v) + (1.0 - opt->beta2) * g * g);
160 double bc1 = 1.0 - pow(opt->beta1, (double) opt->step_count);
161 double bc2 = 1.0 - pow(opt->beta2, (double) opt->step_count);
162 float mhat = (float) (*m / bc1);
163 float vhat = (float) (*v / bc2);
164 float denom = (float) sqrt((double) vhat) + (float) opt->epsilon;
165 float update = mhat / denom;
166 if (opt->type == SNEPPX_OPTIMIZER_ADAMW || opt->type == SNEPPX_OPTIMIZER_SF_ADAMW)
167 value -= (float) (lr * (wd * value + update));
168 else
169 value -= (float) (lr * update);
170 }
171 else
172 {
173 float g2 = g;
174 if (opt->momentum != 0.0)
175 {
176 float prev = opt->buf[p];
177 float cur = (float) (opt->momentum * prev + g2);
178 if (opt->nesterov)
179 g2 = (float) (opt->momentum * prev + g2);
180 else
181 g2 = cur;
182 opt->buf[p] = cur;
183 }
184 value -= (float) (lr * g2);
185 }
186 set_flat(param, i, value);
187 }
188 }
189 return 0;
190}

271int SNEPPX_grad_opt_clip_grad_norm(SNEPPXTensor** grads, size_t num_grads, double max_norm)
272{
273 if (!grads)
274 return -1;
275 double total = 0.0;
276 for (size_t g = 0; g < num_grads; g++)
277 {
278 if (!grads[g])
279 continue;
280 float* d = (float*) grads[g]->data;
281 for (size_t i = 0; i < grads[g]->size; i++)
282 total += (double) d[i] * d[i];
283 }
284 double norm = sqrt(total);
285 if (norm > max_norm && norm > 0.0)
286 {
287 double scale = max_norm / norm;
288 for (size_t g = 0; g < num_grads; g++)
289 {
290 if (!grads[g])
291 continue;
292 float* d = (float*) grads[g]->data;
293 for (size_t i = 0; i < grads[g]->size; i++)
294 d[i] = (float) (d[i] * scale);
295 }
296 }
297 return 0;
298}
