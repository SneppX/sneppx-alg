42{

53static int update_compute_digest(const SNEPPXSignedUpdate* update, uint8_t digest[SNEPPX_UPDATE_HASH_LEN])
54{
55 static const uint8_t domain[] = "SNEPPX-S7-SIGNED-UPDATE-v2";
56 if (!update || !digest || update->chunk_count <= 0 || update->chunk_count > SNEPPX_MAX_UPDATE_CHUNKS ||
57 update->timestamp == 0 || (update->is_delta != 0 && update->is_delta != 1))
58 return -1;
59 size_t total_size = 0;
60 for (int i = 0; i < update->chunk_count; i++)
61 {
62 const SNEPPXUpdateChunk* chunk = &update->chunks[i];
63 if (chunk->chunk_index != (uint32_t) i || chunk->chunk_size == 0 ||
64 (size_t) chunk->chunk_size > SIZE_MAX - total_size)
65 return -1;
66 total_size += (size_t) chunk->chunk_size;
67 }
68 if (total_size == 0)
69 return -1;
70 SNEPPXBlake3State ctx;
71 SNEPPX_blake3_init(&ctx);
72 SNEPPX_blake3_update(&ctx, domain, sizeof(domain));
73 update_hash_u32(&ctx, update->version_major);
74 update_hash_u32(&ctx, update->version_minor);
75 update_hash_u32(&ctx, update->version_patch);
76 update_hash_u64(&ctx, update->timestamp);
77 update_hash_u32(&ctx, (uint32_t) update->is_delta);
78 update_hash_u32(&ctx, (uint32_t) update->chunk_count);
79 for (int i = 0; i < update->chunk_count; i++)
80 {
81 const SNEPPXUpdateChunk* chunk = &update->chunks[i];
82 update_hash_u32(&ctx, chunk->chunk_index);
83 update_hash_u32(&ctx, chunk->chunk_size);
84 SNEPPX_blake3_update(&ctx, chunk->chunk_hash, SNEPPX_UPDATE_HASH_LEN);
85 }
86 SNEPPX_blake3_finish(&ctx, digest);
87 return 0;
88}
89static int update_payload_matches(const SNEPPXSignedUpdate* update, const uint8_t* data, size_t data_len)
90{
91 if (!update || !data || data_len == 0 || update->chunk_count <= 0 || update->chunk_count > SNEPPX_MAX_UPDATE_CHUNKS)
92 return -1;
93 size_t offset = 0;
94 for (int i = 0; i < update->chunk_count; i++)
95 {
96 size_t chunk_size = (size_t) update->chunks[i].chunk_size;
97 if (update->chunks[i].chunk_index != (uint32_t) i || chunk_size == 0 || offset > data_len ||
98 chunk_size > data_len - offset)
99 return -1;
100 SNEPPXBlake3State ctx;
101 uint8_t actual_hash[SNEPPX_UPDATE_HASH_LEN];
102 SNEPPX_blake3_init(&ctx);
103 SNEPPX_blake3_update(&ctx, data + offset, chunk_size);
104 SNEPPX_blake3_finish(&ctx, actual_hash);
105 int matches = SNEPPX_ct_equal(actual_hash, update->chunks[i].chunk_hash, SNEPPX_UPDATE_HASH_LEN);
106 memset(actual_hash, 0, sizeof(actual_hash));
107 if (!matches)
108 return -1;
109 offset += chunk_size;
110 }
111 return offset == data_len ? 0 : -1;
112}

199int SNEPPX_update_verifier_apply(SNEPPXUpdateVerifier* uv, const SNEPPXSignedUpdate* update, const uint8_t* update_data,
200 size_t data_len)
201{
202 if (!uv || !update || !update_data || data_len == 0)
203 return -1;
204 /* A valid signature does not make an already-applied version applicable again. */
205 if (update->version_major == uv->current_version[0] && update->version_minor == uv->current_version[1] &&
206 update->version_patch == uv->current_version[2])
207 return -1;
208 if (!uv->verification_enabled || SNEPPX_update_verifier_check(uv, update) != 0)
209 return -1;
210 if (update_payload_matches(update, update_data, data_len) != 0)
211 return -1;

340int SNEPPX_update_verifier_sign_update(SNEPPXSignedUpdate* update, const uint8_t* signing_key, size_t key_len)
341{
342 if (!update || !signing_key || key_len != SNEPPX_ED25519_SEED_LEN)
343 return -1;
344 uint8_t digest[SNEPPX_UPDATE_HASH_LEN];
345 if (update_compute_digest(update, digest) != 0)
346 return -1;
347 SNEPPXEd25519Keypair kp;
348 SNEPPXEd25519Signature sig;
349 memset(&kp, 0, sizeof(kp));
350 memset(&sig, 0, sizeof(sig));
351 int rc = SNEPPX_ed25519_secret_key_expand(kp.private_key, signing_key);
352 if (rc == 0)
353 rc = SNEPPX_ed25519_publickey_from_seed(signing_key, kp.public_key);
354 if (rc == 0)
355 rc = SNEPPX_ed25519_sign(&kp, digest, sizeof(digest), &sig);
356 if (rc == 0)
357 {
358 memcpy(update->update_hash, digest, sizeof(digest));
359 memcpy(update->signature, sig.data, sizeof(sig.data));
360 }
361 memset(digest, 0, sizeof(digest));
362 memset(&kp, 0, sizeof(kp));
363 memset(&sig, 0, sizeof(sig));
364 return rc == 0 ? 0 : -1;
365}

384int SNEPPX_update_verifier_verify_signature(const SNEPPXSignedUpdate* update, const uint8_t* public_key, size_t key_len)
385{
386 if (!update || !public_key || key_len != SNEPPX_ED25519_PUBLIC_KEY_LEN)
387 return -1;
388 uint8_t digest[SNEPPX_UPDATE_HASH_LEN];
389 if (update_compute_digest(update, digest) != 0)
390 return -1;
391 if (!SNEPPX_ct_equal(digest, update->update_hash, sizeof(digest)))
392 {
393 memset(digest, 0, sizeof(digest));
394 return -1;
395 }
396 SNEPPXEd25519Signature sig;
397 memcpy(sig.data, update->signature, sizeof(sig.data));
398 int rc = SNEPPX_ed25519_verify(public_key, digest, sizeof(digest), &sig);
399 memset(digest, 0, sizeof(digest));
400 memset(&sig, 0, sizeof(sig));
401 return rc == 1 ? 0 : -1;
402}
