50static int update_compute_digest(const SNEPPXSignedUpdate* update, uint8_t digest[SNEPPX_UPDATE_HASH_LEN]) {
51 static const uint8_t domain[] = "SNEPPX-S7-SIGNED-UPDATE-v2";
52 if (!update || !digest || update->chunk_count <= 0 ||
53 update->chunk_count > SNEPPX_MAX_UPDATE_CHUNKS ||
54 update->timestamp == 0 || (update->is_delta != 0 && update->is_delta != 1)) return -1;
55 size_t total_size = 0;
56 for (int i = 0; i < update->chunk_count; i++) {
57 const SNEPPXUpdateChunk* chunk = &update->chunks[i];
58 if (chunk->chunk_index != (uint32_t)i || chunk->chunk_size == 0 ||
59 (size_t)chunk->chunk_size > SIZE_MAX - total_size) return -1;
60 total_size += (size_t)chunk->chunk_size;
61 }
62 if (total_size == 0) return -1;
63 SNEPPXBlake3State ctx;
64 SNEPPX_blake3_init(&ctx);
65 SNEPPX_blake3_update(&ctx, domain, sizeof(domain));
66 update_hash_u32(&ctx, update->version_major);
67 update_hash_u32(&ctx, update->version_minor);
68 update_hash_u32(&ctx, update->version_patch);
69 update_hash_u64(&ctx, update->timestamp);
70 update_hash_u32(&ctx, (uint32_t)update->is_delta);
71 update_hash_u32(&ctx, (uint32_t)update->chunk_count);
72 for (int i = 0; i < update->chunk_count; i++) {
73 const SNEPPXUpdateChunk* chunk = &update->chunks[i];
74 update_hash_u32(&ctx, chunk->chunk_index);
75 update_hash_u32(&ctx, chunk->chunk_size);
76 SNEPPX_blake3_update(&ctx, chunk->chunk_hash, SNEPPX_UPDATE_HASH_LEN);
77 }
78 SNEPPX_blake3_finish(&ctx, digest);
79 return 0;

81static int update_payload_matches(const SNEPPXSignedUpdate* update, const uint8_t* data, size_t data_len) {
82 if (!update || !data || data_len == 0 || update->chunk_count <= 0 ||
83 update->chunk_count > SNEPPX_MAX_UPDATE_CHUNKS) return -1;
84 size_t offset = 0;
85 for (int i = 0; i < update->chunk_count; i++) {
86 size_t chunk_size = (size_t)update->chunks[i].chunk_size;
87 if (update->chunks[i].chunk_index != (uint32_t)i || chunk_size == 0 ||
88 offset > data_len || chunk_size > data_len - offset) return -1;
89 SNEPPXBlake3State ctx;
90 uint8_t actual_hash[SNEPPX_UPDATE_HASH_LEN];
91 SNEPPX_blake3_init(&ctx);
92 SNEPPX_blake3_update(&ctx, data + offset, chunk_size);
93 SNEPPX_blake3_finish(&ctx, actual_hash);
94 int matches = SNEPPX_ct_equal(actual_hash, update->chunks[i].chunk_hash, SNEPPX_UPDATE_HASH_LEN);
95 memset(actual_hash, 0, sizeof(actual_hash));
96 if (!matches) return -1;
97 offset += chunk_size;
98 }
99 return offset == data_len ? 0 : -1;

177int SNEPPX_update_verifier_apply(SNEPPXUpdateVerifier* uv, const SNEPPXSignedUpdate* update,
178 const uint8_t* update_data, size_t data_len) {
179 if (!uv || !update || !update_data || data_len == 0) return -1;
180 /* A valid signature does not make an already-applied version applicable again. */
181 if (update->version_major == uv->current_version[0] &&
182 update->version_minor == uv->current_version[1] &&
183 update->version_patch == uv->current_version[2]) return -1;
184 if (!uv->verification_enabled || SNEPPX_update_verifier_check(uv, update) != 1) return -1;
185 if (update_payload_matches(update, update_data, data_len) != 0) return -1;
186 uint32_t target[3] = {update->version_major, update->version_minor, update->version_patch};
187 if (SNEPPX_update_verifier_rollback_check(uv, target) != 0) return -1;
188 uv->current_version[0] = update->version_major;
189 uv->current_version[1] = update->version_minor;
190 uv->current_version[2] = update->version_patch;

193 update_history[update_history_count].minor = update->version_minor;
194 update_history[update_history_count].patch = update->version_patch;

292int SNEPPX_update_verifier_sign_update(SNEPPXSignedUpdate* update, const uint8_t* signing_key, size_t key_len) {
293 if (!update || !signing_key || key_len != SNEPPX_ED25519_SEED_LEN) return -1;
294 uint8_t digest[SNEPPX_UPDATE_HASH_LEN];
295 if (update_compute_digest(update, digest) != 0) return -1;
296 SNEPPXEd25519Keypair kp;
297 SNEPPXEd25519Signature sig;
298 memset(&kp, 0, sizeof(kp));
299 memset(&sig, 0, sizeof(sig));
300 int rc = SNEPPX_ed25519_secret_key_expand(kp.private_key, signing_key);
301 if (rc == 0) rc = SNEPPX_ed25519_publickey_from_seed(signing_key, kp.public_key);
302 if (rc == 0) rc = SNEPPX_ed25519_sign(&kp, digest, sizeof(digest), &sig);
303 if (rc == 0) {
304 memcpy(update->update_hash, digest, sizeof(digest));
305 memcpy(update->signature, sig.data, sizeof(sig.data));
306 }
307 memset(digest, 0, sizeof(digest));
308 memset(&kp, 0, sizeof(kp));
309 memset(&sig, 0, sizeof(sig));
310 return rc == 0 ? 0 : -1;
311}

328int SNEPPX_update_verifier_verify_signature(const SNEPPXSignedUpdate* update, const uint8_t* public_key, size_t key_len) {
329 if (!update || !public_key || key_len != SNEPPX_ED25519_PUBLIC_KEY_LEN) return -1;
330 uint8_t digest[SNEPPX_UPDATE_HASH_LEN];
331 if (update_compute_digest(update, digest) != 0) return -1;
332 if (!SNEPPX_ct_equal(digest, update->update_hash, sizeof(digest))) {
333 memset(digest, 0, sizeof(digest));
334 return -1;
335 }
336 SNEPPXEd25519Signature sig;
337 memcpy(sig.data, update->signature, sizeof(sig.data));
338 int rc = SNEPPX_ed25519_verify(public_key, digest, sizeof(digest), &sig);
339 memset(digest, 0, sizeof(digest));
340 memset(&sig, 0, sizeof(sig));
341 return rc == 1 ? 0 : -1;
342}
