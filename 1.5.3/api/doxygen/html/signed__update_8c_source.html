96int SNEPPX_update_verifier_check(SNEPPXUpdateVerifier* uv, const SNEPPXSignedUpdate* update) {
97 if (!uv || !update) return 0;
98 if (!uv->verification_enabled) return 0;
99
100 if (uv->rollback_protection_enabled) {
101 uint32_t target[3] = {update->version_major, update->version_minor, update->version_patch};
102 if (SNEPPX_update_verifier_rollback_check(uv, target) != 0) return 0;
103 if (target[0] < uv->min_allowed_version[0]) return 0;
104 if (target[0] == uv->min_allowed_version[0] && target[1] < uv->min_allowed_version[1]) return 0;
105 if (target[0] == uv->min_allowed_version[0] && target[1] == uv->min_allowed_version[1] && target[2] < uv->min_allowed_version[2]) return 0;
106 }
107
108 SNEPPXBlake3State ctx;
109 uint8_t computed_hash[SNEPPX_UPDATE_HASH_LEN];
110 SNEPPX_blake3_init(&ctx);
111 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->version_major, sizeof(uint32_t) * 3);
112 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->chunk_count, sizeof(int));
113 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->timestamp, sizeof(uint64_t));
114 SNEPPX_blake3_finish(&ctx, computed_hash);
115
116 /* Fail-closed: a trusted public key must be configured to verify. */
117 if (!uv->has_trusted_key) return 0;
118
119 /* Verify Ed25519 signature over BLAKE3(update) hash against trusted key. */
120 SNEPPXEd25519Signature sig;
121 memcpy(sig.data, update->signature, SNEPPX_ED25519_SIGNATURE_LEN);
122 int rc = SNEPPX_ed25519_verify(uv->trusted_pubkey, computed_hash, SNEPPX_UPDATE_HASH_LEN, &sig);
123 return (rc == 1) ? 1 : 0;
124}

247int SNEPPX_update_verifier_sign_update(SNEPPXSignedUpdate* update, const uint8_t* signing_key, size_t key_len) {
248 if (!update || !signing_key || key_len < SNEPPX_ED25519_SEED_LEN) return -1;
249 SNEPPXBlake3State ctx;
250 uint8_t hash[SNEPPX_UPDATE_HASH_LEN];
251 SNEPPX_blake3_init(&ctx);
252 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->version_major, sizeof(uint32_t) * 3);
253 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->chunk_count, sizeof(int));
254 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->timestamp, sizeof(uint64_t));
255 SNEPPX_blake3_finish(&ctx, hash);
256 memcpy(update->update_hash, hash, SNEPPX_UPDATE_HASH_LEN);
257 /* Real Ed25519 signature over the update hash. signing_key is the 32-byte seed. */
258 SNEPPXEd25519Keypair kp;
259 if (SNEPPX_ed25519_secret_key_expand(kp.private_key, signing_key) != 0) return -1;
260 if (SNEPPX_ed25519_publickey_from_seed(signing_key, kp.public_key) != 0) return -1;
261 SNEPPXEd25519Signature sig;
262 if (SNEPPX_ed25519_sign(&kp, hash, SNEPPX_UPDATE_HASH_LEN, &sig) != 0) return -1;
263 memcpy(update->signature, sig.data, SNEPPX_UPDATE_SIG_LEN);
264 return 0;
265}
