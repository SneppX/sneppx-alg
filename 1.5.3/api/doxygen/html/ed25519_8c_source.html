139static void fe_mul(field* r, const field* a, const field* b)
140{
141 uint128 t[9] = {0};
142 for (int i = 0; i < 5; i++)
143 for (int j = 0; j < 5; j++)
144 {
145 int k = i + j;
146 if (k < 9)
147 t[k] = add128(t[k], mul64_64(a->v[i], b->v[j]));
148 }
149 /* Fold high terms (×19) back into lower 5 */
150 for (int i = 8; i >= 5; i--)
151 {
152 uint64_t low51 = t[i].lo & 0x7ffffffffffffULL;
153 uint128 high = rshift128(t[i], 51);
154 t[i - 5] = add128(t[i - 5], mul64_64(19, low51));
155 t[i - 4] = add128(t[i - 4], mul64_64(19, high.lo));
156 }
157 /* Two rounds of carry propagation */
158 for (int round = 0; round < 2; round++)
159 {
160 uint128 carry;
161 carry = rshift128(t[0], 51);
162 t[0].lo = mask51(t[0].lo);
163 t[0].hi = 0;
164 t[1] = add128(t[1], carry);
165 carry = rshift128(t[1], 51);
166 t[1].lo = mask51(t[1].lo);
167 t[1].hi = 0;
168 t[2] = add128(t[2], carry);
169 carry = rshift128(t[2], 51);
170 t[2].lo = mask51(t[2].lo);
171 t[2].hi = 0;
172 t[3] = add128(t[3], carry);
173 carry = rshift128(t[3], 51);
174 t[3].lo = mask51(t[3].lo);
175 t[3].hi = 0;
176 t[4] = add128(t[4], carry);
177 carry = rshift128(t[4], 51);
178 t[4].lo = mask51(t[4].lo);
179 t[4].hi = 0;
180 if (carry.lo || carry.hi)
181 {
182 t[0] = add128(t[0], mul64_64(19, carry.lo));
183 t[0] = add128(t[0], mul64_64(19, carry.hi));
184 }
185 }
186 for (int i = 0; i < 5; i++)
187 r->v[i] = t[i].lo;
188}

366/* Decode Edwards point from 32 bytes: Y in bytes, X parity in top bit of byte[31] */
367static int point_from_bytes(point* p, const uint8_t b[32])
368{
369 /* Reject non-canonical y encodings instead of silently reducing mod p. */
370 uint8_t encoded_y[32], canonical_y[32];
371 memcpy(encoded_y, b, sizeof(encoded_y));
372 encoded_y[31] &= 0x7f;
373 field decoded_y;
374 fe_from_bytes(&decoded_y, encoded_y);
375 fe_to_bytes(canonical_y, &decoded_y);
376 if (!SNEPPX_ct_equal(encoded_y, canonical_y, sizeof(encoded_y)))
377 return -1;
378 fe_from_bytes(&p->Y, b);
379 memset(&p->Z, 0, sizeof(field));
380 p->Z.v[0] = 1;
381 field u, v, v3, vxx, check;
382 fe_sq(&u, &p->Y);
383 fe_sub(&u, &u, &p->Z); /* u = Y^2 - 1 */
384 fe_sq(&v, &p->Y);
385 fe_mul(&v, &v, &D);
386 fe_add(&v, &v, &p->Z); /* v = d*Y^2 + 1 */
387 fe_sq(&v3, &v);
388 fe_mul(&v3, &v3, &v); /* v3 = v^3 */
389 fe_sq(&p->X, &v3);
390 fe_mul(&p->X, &p->X, &v);
391 fe_mul(&p->X, &p->X, &u); /* X = u * v^7 */
392 fe_pow22523(&p->X, &p->X); /* X = (u * v^7)^((p-5)/8) */
393 fe_mul(&p->X, &p->X, &v3);
394 fe_mul(&p->X, &p->X, &u); /* X = u * v^3 * (u*v^7)^((p-5)/8) */
395 fe_sq(&vxx, &p->X);
396 fe_mul(&vxx, &vxx, &v);
397 fe_sub(&check, &vxx, &u);
398 uint8_t ck[32];
399 fe_to_bytes(ck, &check);
400 if (!SNEPPX_ct_is_zero(ck, 32))
401 {
402 fe_add(&check, &vxx, &u);
403 fe_to_bytes(ck, &check);
404 if (!SNEPPX_ct_is_zero(ck, 32))
405 {
406 return -1;
407 }
408 fe_mul(&p->X, &p->X, &SQRTM1);
409 }
410 /* RFC 8032 forbids the negative-zero encoding (x = 0, sign bit = 1). */
411 uint8_t x_encoding[32];
412 fe_to_bytes(x_encoding, &p->X);
413 if (SNEPPX_ct_is_zero(x_encoding, sizeof(x_encoding)) && (b[31] & 0x80))
414 return -1;
415 if ((p->X.v[0] & 1) != (b[31] >> 7))
416 fe_neg(&p->X, &p->X);
417 fe_mul(&p->T, &p->X, &p->Y);
418 return 0;
419}

497static void point_double(point* r, const point* p)
498{
499 field a, b, c, d, e, f, g, h, t;
500 fe_sub(&a, &p->Y, &p->X);
501 fe_sq(&a, &a); /* A = (Y-X)^2 */
502 fe_add(&b, &p->Y, &p->X);
503 fe_sq(&b, &b); /* B = (Y+X)^2 */
504 fe_sq(&c, &p->T);
505 fe_mul(&c, &c, &D2); /* C = 2*d*T^2 */
506 fe_sq(&d, &p->Z);
507 fe_add(&d, &d, &d); /* D = 2*Z^2 */
508 fe_sub(&e, &b, &a); /* E = B-A = 4*X*Y */
509 fe_sub(&f, &d, &c); /* F = D-C = 2*(Z^2 - d*T^2) */
510 fe_add(&g, &d, &c); /* G = D+C = 2*(Z^2 + d*T^2) */
511 fe_add(&h, &b, &a); /* H = B+A = 2*(Y^2+X^2) */
512 fe_mul(&r->X, &e, &f);
513 fe_mul(&r->Y, &g, &h);
514 fe_mul(&r->T, &e, &h);
515 fe_mul(&r->Z, &f, &g);
516 /* Identity guard: if X == 0 and Y == Z, force output to (0,1,1,0) */
517 {
518 uint8_t xb[32], yzb[32];
519 fe_to_bytes(xb, &p->X);
520 fe_sub(&t, &p->Y, &p->Z);
521 fe_to_bytes(yzb, &t);
522 uint64_t idm = (uint64_t) (-(int) (SNEPPX_ct_is_zero(xb, 32) && SNEPPX_ct_is_zero(yzb, 32)));
523 for (int i = 0; i < 5; i++)
524 {
525 r->X.v[i] &= ~idm;
526 r->Y.v[i] = (r->Y.v[i] & ~idm) | (idm & 1ULL);
527 r->Z.v[i] = (r->Z.v[i] & ~idm) | (idm & 1ULL);
528 r->T.v[i] &= ~idm;
529 }
530 }
531}

563 r->Y.v[j] ^= mask & (r->Y.v[j] ^ t.Y.v[j]);
564 r->Z.v[j] ^= mask & (r->Z.v[j] ^ t.Z.v[j]);
565 r->T.v[j] ^= mask & (r->T.v[j] ^ t.T.v[j]);
566 }
567 }

570static void point_normalize(point* p)
571{
572 field zinv;
573 fe_invert(&zinv, &p->Z);
574 fe_mul(&p->X, &p->X, &zinv);

578 fe_mul(&p->T, &p->X, &p->Y);

592/* RFC 8032 requires S to be the canonical little-endian scalar in [0, L). */
593static int scalar_is_canonical(const uint8_t scalar[32])
594{
595 for (int i = 31; i >= 0; i--)
596 {
597 if (scalar[i] < L_BYTES[i])
598 return 1;
599 if (scalar[i] > L_BYTES[i])
600 return 0;
601 }
602 return 0;
603}

605/* Strict verification requires A and R to be non-identity points in the
606 * prime-order subgroup. Checking [L]P = identity rejects torsion components;
607 * checking [8]P != identity also rejects the identity/small-order points. */
608static int point_is_identity(point* p)
609{
610 point_normalize(p);
611 uint8_t x[32], y[32], zero[32] = {0}, one[32] = {1};
612 fe_to_bytes(x, &p->X);
613 fe_to_bytes(y, &p->Y);
614 return SNEPPX_ct_equal(x, zero, sizeof(x)) && SNEPPX_ct_equal(y, one, sizeof(y));
615}

617static int point_is_valid_verification_point(const point* p)
618{
619 static const uint8_t eight[32] = {8};
620 point order_check, cofactor_check;
621 point_scalar_mult(&order_check, L_BYTES, sizeof(L_BYTES), p);
622 if (!point_is_identity(&order_check))
623 return 0;
624 point_scalar_mult(&cofactor_check, eight, sizeof(eight), p);
625 return !point_is_identity(&cofactor_check);
626}

628/* Subtract L << (8*shift) from t[64]; return 0 if result >= 0, 1 if borrow */
629static int sub_sc_shifted(uint8_t t[64], int shift)
630{
631 uint16_t borrow = 0;
632 for (int i = 0; i < 64; i++)
633 {
634 uint16_t sub = (i >= shift && i < shift + 32) ? (uint16_t) L_BYTES[i - shift] : 0;
635 uint16_t diff = (uint16_t) t[i] - sub - borrow;
636 t[i] = (uint8_t) (diff & 0xFF);
637 borrow = (diff >> 8) & 1;
638 }
639 return (int) borrow;
640}

642/* Reduce a 64-byte value mod L to 32 bytes by repeatedly subtracting L << (8*k) */
643static void sc_reduce64(uint8_t r[32], const uint8_t a[64])
644{
645 uint8_t t[64];
646 memcpy(t, a, 64);
647 /* Find highest non-zero byte in t */
648 int hi = 63;
649 while (hi >= 0 && t[hi] == 0)
650 hi--;
651 if (hi < 0)
652 {
653 memset(r, 0, 32);
654 return;
655 }
656 /* Subtract L << (8*k) for k from (hi-31) down to 0 */
657 for (int k = hi - 31; k >= 0; k--)
658 {
659 while (1)
660 {
661 /* Check if t >= L << (8*k) by comparing bytes from hi down to k */
662 int ge = 0;
663 for (int i = hi; i >= k + 32; i--)
664 {
665 if (t[i])
666 {
667 ge = 1;
668 break;
669 }
670 }
671 if (!ge)
672 {
673 /* Compare the overlapping 32-byte window */
674 int cmp = 0;
675 for (int i = k + 31; i >= k && cmp == 0; i--)
676 {
677 uint8_t tv = t[i];
678 uint8_t lv = L_BYTES[i - k];
679 if (tv > lv)
680 {
681 cmp = 1;
682 }
683 else if (tv < lv)
684 {
685 cmp = -1;
686 }
687 }
688 if (cmp < 0)
689 break; /* t < L << (8*k) */
690 }
691 if (sub_sc_shifted(t, k))
692 break; /* borrow = overshot */
693 }
694 }
695 /* Final subtraction of L (shift 0) up to 4 times */
696 for (int iter = 0; iter < 4; iter++)
697 {
698 int cmp = 0;
699 for (int i = 31; i >= 0 && cmp == 0; i--)
700 {
701 if (t[i] > L_BYTES[i])
702 cmp = 1;
703 else if (t[i] < L_BYTES[i])
704 cmp = -1;
705 }
706 if (cmp < 0)
707 break;
708 sub_sc_shifted(t, 0);
709 }
710 memcpy(r, t, 32);
711}

713/* Multiply two 32-byte scalars -> 64-byte product (schoolbook, 16-bit limbs) */
714static void sc_mul256(uint8_t p[64], const uint8_t a[32], const uint8_t b[32])
715{
716 uint32_t t[64] = {0};
717 for (int i = 0; i < 32; i++)
718 for (int j = 0; j < 32; j++)
719 t[i + j] += (uint32_t) a[i] * (uint32_t) b[j];
720 uint64_t carry = 0;
721 for (int i = 0; i < 64; i++)
722 {
723 carry += t[i];
724 p[i] = (uint8_t) (carry & 0xFF);
725 carry >>= 8;
726 }
727}

801int SNEPPX_ed25519_sign(const SNEPPXEd25519Keypair* kp, const uint8_t* message, size_t msg_len,
802 SNEPPXEd25519Signature* sig)
803{
804 if (!kp || !message || !sig)
805 return -1;
806 init_base_point();
807 uint8_t r_seed[64], r_scalar[32], h_scalar[32], hram[64], product[64], sum[64];
808 /* r = SHA-512(nonce_seed || message) */
809 SNEPPXSHA512Context ctx;
810 SNEPPX_sha512_init(&ctx);
811 SNEPPX_sha512_update(&ctx, kp->private_key + 32, 32);
812 SNEPPX_sha512_update(&ctx, message, msg_len);
813 SNEPPX_sha512_finish(&ctx, r_seed);
814 sc_reduce64(r_scalar, r_seed);
815 /* R = r_scalar * B */
816 point R;
817 point_scalar_mult(&R, r_scalar, 32, &B);
818 point_normalize(&R);
819 fe_to_bytes(sig->data, &R.Y);
820 sig->data[31] |= (uint8_t) ((R.X.v[0] & 1) << 7);
821 /* h = SHA-512(R || A || message) */
822 SNEPPX_sha512_init(&ctx);
823 SNEPPX_sha512_update(&ctx, sig->data, 32);
824 SNEPPX_sha512_update(&ctx, kp->public_key, 32);
825 SNEPPX_sha512_update(&ctx, message, msg_len);
826 SNEPPX_sha512_finish(&ctx, hram);
827 /* Reduce full 64-byte challenge hash mod L */
828 sc_reduce64(h_scalar, hram);
829 /* S = (r + h*a) mod L */
830 /* a is the clamped private key (kp->private_key[0:32]) */
831 sc_mul256(product, h_scalar, kp->private_key);
832 /* product + r_scalar as 64-byte sum */
833 uint32_t carry = 0;
834 for (int i = 0; i < 32; i++)
835 {
836 carry = (uint32_t) product[i] + (uint32_t) r_scalar[i] + carry;
837 sum[i] = (uint8_t) (carry & 0xFF);
838 carry >>= 8;
839 }
840 for (int i = 32; i < 64; i++)
841 {
842 carry += product[i];
843 sum[i] = (uint8_t) (carry & 0xFF);
844 carry >>= 8;
845 }
846 /* Reduce sum mod L to get S */
847 sc_reduce64(sig->data + 32, sum);
848 return 0;
849}

862static int ed25519_verify_core(const uint8_t* public_key, const uint8_t* message, size_t msg_len,
863 const SNEPPXEd25519Signature* sig, int legacy)
864{
865 if (!public_key || !message || !sig)
866 return -1;
867 /* Non-canonical S values make Ed25519 signatures malleable. */
868 if (!scalar_is_canonical(sig->data + 32))
869 return 0;
870 init_base_point();
871 uint8_t hram[64];
872 SNEPPXSHA512Context ctx;
873 SNEPPX_sha512_init(&ctx);
874 SNEPPX_sha512_update(&ctx, sig->data, 32);
875 SNEPPX_sha512_update(&ctx, public_key, 32);
876 SNEPPX_sha512_update(&ctx, message, msg_len);
877 if (legacy)
878 SNEPPX_sha512_legacy_finish(&ctx, hram);
879 else
880 SNEPPX_sha512_finish(&ctx, hram);
881 point A;
882 int ret_A = point_from_bytes(&A, public_key);
883 if (ret_A != 0)
884 return -1;
885 point R;
886 int ret_R = point_from_bytes(&R, sig->data);
887 if (ret_R != 0)
888 return -1;
889 if (!point_is_valid_verification_point(&A) || !point_is_valid_verification_point(&R))
890 return 0;
891
892 /* Verify: [S]B = R + [h]A */
893 uint8_t h_scalar[32];
894 sc_reduce64(h_scalar, hram);
895 point S_point;
896 point_scalar_mult(&S_point, sig->data + 32, 32, &B);
897
898 point hA;
899 point_scalar_mult(&hA, h_scalar, 32, &A);
900
901 point_add(&hA, &R, &hA);
902
903 point_normalize(&S_point);
904 point_normalize(&hA);
905
906 /* Compare in affine coordinates (canonical bytes) */
907 uint8_t bx[32], cx[32], by[32], cy[32];
908 fe_to_bytes(bx, &S_point.X);
909 fe_to_bytes(cx, &hA.X);
910 fe_to_bytes(by, &S_point.Y);
911 fe_to_bytes(cy, &hA.Y);
912 return SNEPPX_ct_equal(bx, cx, 32) && SNEPPX_ct_equal(by, cy, 32);
913}

925 const SNEPPXEd25519Signature* sig)
926{
927 return ed25519_verify_core(public_key, message, msg_len, sig, 0);

947int SNEPPX_ed25519_verify_compat(const uint8_t* sig, const uint8_t* m, size_t mlen, const uint8_t* pk, int allow_legacy)
948{
949 if (!sig || !m || !pk)
950 return -1;
951 SNEPPXEd25519Signature sig_struct;
952 memcpy(sig_struct.data, sig, SNEPPX_ED25519_SIGNATURE_LEN);
953
954 /* RFC-correct verify first */
955 int ret = ed25519_verify_core(pk, m, mlen, &sig_struct, 0);
956 if (ret == 1)
957 return 1;

959 if (ret == -1 || !allow_legacy)
960 return ret;
961 /* Legacy fallback: re-derive challenge with old H1 digest */
962 return ed25519_verify_core(pk, m, mlen, &sig_struct, 1);
963}
