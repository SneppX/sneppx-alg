74int SNEPPX_leak_track_alloc(void* ptr, size_t size, const char* file, int line, const char* func)
75{
76 if (!ptr || !tracking_enabled)
77 return -1;
78 if (alloc_count >= LEAK_MAX_ALLOCATIONS)
79 return -1;
80 allocations[alloc_count].ptr = ptr;
81 allocations[alloc_count].size = size;
82 allocations[alloc_count].file = file;
83 allocations[alloc_count].line = line;
84 allocations[alloc_count].func = func;
85 allocations[alloc_count].timestamp = (uint64_t) clock();
86 allocations[alloc_count].freed = 0;
87 allocations[alloc_count].bt_depth = 0;
88 alloc_count++;
89 total_allocated += size;
90 current_usage += size;
91 if (current_usage > peak_usage)
92 peak_usage = current_usage;
93 return 0;
94}

101int SNEPPX_leak_track_free(void* ptr)
102{
103 if (!ptr || !tracking_enabled)
104 return -1;
105 for (int i = 0; i < alloc_count; i++)
106 {
107 if (allocations[i].ptr == ptr && !allocations[i].freed)
108 {
109 allocations[i].freed = 1;
110 current_usage -= allocations[i].size;
111 total_freed += allocations[i].size;
112 return 0;
113 }
114 }
115 return 1;
116}

125int SNEPPX_leak_check(leak_report_t* reports, int max_reports)
